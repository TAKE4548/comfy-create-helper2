### 2.2 RescanModal.tsx の変更対象箇所
```typescript
interface RescanModalProps {
  workflowId: number
  workflowJson: Record<string, unknown>
  existingMappings?: { master_id: number; node_id: string }[] | WorkflowMapping[]
  onClose: () => void
  onSaved: () => void
}

// useEffect 内
workflowsApi.scan(workflowJson)
  .then((result) => {
    if (!cancelled) {
      const res = result as ScanResult
      setScanResult(res)
      initFromScan(res, existingMappings, workflowJson)
      setPhase('editing')
    }
  })
```
