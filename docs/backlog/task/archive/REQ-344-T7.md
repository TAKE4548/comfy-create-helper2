        fs.chmodSync(depsDir, 0o777)
      } catch {
        // 権限エラー等は安全に無視
      }

      const metadataFile = path.join(depsDir, '_metadata.json')
      if (!fs.existsSync(metadataFile)) {

## Branches and Output

- BASELINE_NATIVE_EPERM＋同runtime/cwd/hash/env解除: T8比較候補として親へ。
- native全PASS: 再現状況の変化として返す。T8/testsを自動実行しない。
- partial success/別error/version変化/link異常/hash不一致/env残存/期限/証拠不足:
  INCONCLUSIVEで親へ、追加probe/runtime/path/context切替なし。
- cache deps/deps_tempがあるならCONFIG_MUTATION_RISK。変更せずfocused禁止。
- 既存枝が発火しなくても通常cache書込みゼロの証拠なしならCACHE_SAFETY_UNPROVEN。
  厳密なcache変更禁止の下でtestsを許可しない。例外は人のreviewへ。
