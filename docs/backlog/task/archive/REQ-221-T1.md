### 2.1 必須の型定義およびユーティリティ (`useChipParser.ts`)
```typescript
export type TagChip = {
  id: string;
  text: string;
  enabled: boolean;
};
// TagChipEditor 内では TagChipType として import される別名
export type TagChipType = TagChip;

// 有効タグのみをカンマ区切り文字列に変換
export const chipsToString = (chips: TagChip[]): string => {
  return chips
    .filter((chip) => chip.enabled)
    .map((chip) => chip.text)
    .join(', ');
};

// 文字列からタグチップ配列をパース
export const parseToChips = (text: string): TagChip[] => {
  if (!text) return [];
  return text
    .split(',')
    .map((s) => s.trim())
    .filter(Boolean)
    .map((t, idx) => ({ id: `${Date.now()}-${idx}`, text: t, enabled: true }));
};
```

## 3. 実装要件 (Implementation Logic)
1. **`frontend/src/store/slices/multiSetSlice.ts`**:
   - チップ配列の等価性判定関数 `areChipsEqual` をエクスポート定義する：
     ```typescript
     export const areChipsEqual = (a?: TagChip[], b?: TagChip[]): boolean => {
       if (a === b) return true
       if (!a || !b) return false
       if (a.length !== b.length) return false
       return a.every((chip, idx) => chip.id === b[idx].id && chip.enabled === b[idx].enabled && chip.text === b[idx].text)
     }
     ```
   - `setPromptChips: (role, chips) => { ... }` 内で、既存の `get().promptChips?.[role]` と `chips` を `areChipsEqual` で比較し、同一であれば `set()` をスキップする。

2. **`frontend/src/components/TagChipEditor/index.tsx`**:
   - `onChipsChange` と `onChange` を `useRef` でラップする（レンダーごとに最新関数インスタンスを保持）：
     ```typescript
     const onChipsChangeRef = useRef(onChipsChange);
     onChipsChangeRef.current = onChipsChange;
     const onChangeRef = useRef(onChange);
     onChangeRef.current = onChange;
     const prevEmittedChipsRef = useRef<TagChipType[] | undefined>(chipsProp);
     ```
   - 外部同期 Effect（`chipsProp !== undefined`）実行時に、受け取った `chipsProp` を `prevEmittedChipsRef.current = chipsProp` に保存する。
   - 内部状態通知 Effect を以下のように改修する：
     - 依存配列から `onChipsChange` および `onChange` を除外し `[state, value]` とする。
     - `state` と `prevEmittedChipsRef.current` を比較（件数および各要素の `id`, `text`, `enabled`）。
     - 内容に変化がある場合のみ `prevEmittedChipsRef.current = state; onChipsChangeRef.current?.(state)` を実行する。
