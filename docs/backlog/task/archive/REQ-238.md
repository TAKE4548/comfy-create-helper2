```typescript
/// <reference types="vitest" />
import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'
import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'

const __filename = fileURLToPath(import.meta.url)
const __dirname = path.dirname(__filename)

/**
 * Windows環境で発生する .vite/deps ディレクトリの Read-Only 属性および
 * _metadata.json 欠落時の EPERM rmdir クラッシュを自律解消する事前サニタイズ処理。
 */
function cleanBrokenViteDeps(cacheDir: string): void {
  try {
    if (!fs.existsSync(cacheDir)) {
      return
    }

    const depsDir = path.join(cacheDir, 'deps')
    if (fs.existsSync(depsDir)) {
      try {
        // Windows NTFS の Read-Only (attrib +r) 属性を解除
        fs.chmodSync(depsDir, 0o777)
      } catch {
        // 権限エラー等は安全に無視
      }

      const metadataFile = path.join(depsDir, '_metadata.json')
      if (!fs.existsSync(metadataFile)) {
        // キャッシュメタデータが存在しない空洞化・破損フォルダは事前に削除し、
        // Vite 内部の fsp.rm による EPERM 例外を回避
        try {
          fs.rmSync(depsDir, { recursive: true, force: true })
        } catch {
          // 削除失敗時も起動処理自体はブロックしない
        }
      }
    }

    // 過去の破損リネームで残留した deps_temp_* 一時フォルダ群を一掃
    const entries = fs.readdirSync(cacheDir, { withFileTypes: true })
    for (const entry of entries) {
      if (entry.isDirectory() && entry.name.startsWith('deps_temp_')) {
        const tempPath = path.join(cacheDir, entry.name)
        try {
          fs.chmodSync(tempPath, 0o777)
          fs.rmSync(tempPath, { recursive: true, force: true })
        } catch {}
      }
    }
  } catch {
    // ディレクトリアクセス例外等で Vite 本体の起動を阻害しないようハンドリング
  }
}

// 設定評価前にキャッシュチェック・自律修復を実行
cleanBrokenViteDeps(path.resolve(__dirname, 'node_modules/.vite'))

// https://vite.dev/config/
export default defineConfig({
  plugins: [react()],
  test: {
    globals: true,
    environment: 'jsdom',
    setupFiles: './src/test/setup.ts',
    pool: 'forks',
  },
  server: {
    host: true, // Listen on all local IP addresses

      '/api': {
        target: 'http://127.0.0.1:8000',
        changeOrigin: true,
```

### 2.2 既存および改修対象コード (`tools/server-manager/server-manager.ps1`)

```powershell
# 関連する既存関数
function Test-PortOpen ([int]$port) {
    $conn = Get-NetTCPConnection -LocalPort $port -ErrorAction SilentlyContinue | Where-Object State -eq "Listen"
    return ($null -ne $conn)
}

function Get-PidsFromPort ([int]$port) {
    $conns = Get-NetTCPConnection -LocalPort $port -ErrorAction SilentlyContinue | Where-Object State -eq "Listen"
    if ($conns) {
        $pids = @($conns | Select-Object -ExpandProperty OwningProcess -Unique)
        return ,@($pids)
    }
    return ,@()
}

# 改修対象関数 1: Stop-PortProcesses
function Stop-PortProcesses ([int]$port, [string]$name) {
    $pids = Get-PidsFromPort $port
    foreach ($p in $pids) {
        try {
            # taskkill /T /F によりプロセスおよび配下の子プロセスツリー（cmd, node, esbuild等）を一括強制終了
            & taskkill.exe /PID $p /T /F 2>$null
            Stop-Process -Id $p -Force -ErrorAction SilentlyContinue
        } catch {}
    }
}

# 改修対象関数 2: Start-FrontendProcess
function Start-FrontendProcess ([bool]$asBackground) {
    if (Test-PortOpen $cfg.frontend.port) {
        Write-Host "Frontend is already running on port $($cfg.frontend.port)." -ForegroundColor Yellow
        return
    }
    $feDir = (Resolve-Path (Join-Path $SCRIPT_DIR $cfg.frontend.relative_dir)).Path

    # 起動前に .vite ディレクトリの ReadOnly 属性ビットを解除し、空洞化 deps を先行クリーンアップ
    $viteDir = Join-Path $feDir "node_modules\.vite"
    if (Test-Path $viteDir) {
        try {
            Get-ChildItem -Path $viteDir -Recurse -Force -ErrorAction SilentlyContinue | ForEach-Object {
                if ($_.Attributes -band [System.IO.FileAttributes]::ReadOnly) {
                    $_.Attributes = $_.Attributes -band (-bnot [System.IO.FileAttributes]::ReadOnly)
                }
            }
            $depsDir = Join-Path $viteDir "deps"
            $metaFile = Join-Path $depsDir "_metadata.json"
            if ((Test-Path $depsDir) -and (-not (Test-Path $metaFile))) {
                Remove-Item -Path $depsDir -Recurse -Force -ErrorAction SilentlyContinue
            }
        } catch {}
    }

    if ($asBackground) {
        $logFile = Join-Path $LOGS_DIR "frontend.log"
        Start-Process -FilePath "cmd.exe" -ArgumentList "/c $($cfg.frontend.start_cmd) > `"$logFile`" 2>&1" -WorkingDirectory $feDir -WindowStyle Hidden
    } else {
        Start-Process -FilePath "cmd.exe" -ArgumentList "/k title PromptSurgeon Frontend && $($cfg.frontend.start_cmd)" -WorkingDirectory $feDir
    }
}
```

## 3. 実装要件 (Implementation Logic)

### 3.1 `frontend/vite.config.ts`
1. `node:fs`, `node:path`, `node:url` から必要な関数をインポートする。
2. ESModule 形式で `__dirname` を取得する（`const __filename = fileURLToPath(import.meta.url); const __dirname = path.dirname(__filename)`）。
3. `cleanBrokenViteDeps(cacheDir: string)` 関数を定義し、設定オブジェクト評価前に `cleanBrokenViteDeps(path.resolve(__dirname, 'node_modules/.vite'))` を呼び出す。
   - `deps` が存在し、`_metadata.json` がない場合は `fs.chmodSync` 後に `fs.rmSync` で削除。
   - `deps_temp_*` が存在する場合は属性解除後に削除。
   - すべてのファイル操作は try-catch で保護し、例外で Vite 起動を阻害しない。

### 3.2 `tools/server-manager/server-manager.ps1`
1. `Stop-PortProcesses`:
   - 単発の `Stop-Process -Force` 前に `& taskkill.exe /PID $p /T /F 2>$null` を呼び出し、cmd や node、esbuild 等の全子プロセスツリーを終了させる。
2. `Start-FrontendProcess`:
   - `npm run dev` 起動の直前に、`$feDir/node_modules\.vite` 配下の ReadOnly ビットクリアおよび破損 `deps` の先行削除処理を追加する。

### 3.3 `docs/specs/server_manager_guide.md`
- 第3章の利用方法およびトラブルシューティング関連の項目に、プロセスツリー終了（`taskkill /T /F`）およびキャッシュ自律回復の仕組みに関する解説を反映する。
