### 2.1 リトライ付き即時アンロード仕様 (CRIT-05)
```python
async def unload_vlm_model_with_retry(max_retries: int = 1) -> bool:
    """LM Studio API を呼び出してGPU VRAMを即時解放 (1回リトライ付き)"""
    for attempt in range(max_retries + 1):
        try:
            async with httpx.AsyncClient(timeout=4.0) as client:
                res = await client.post(
                    f"{settings.llm_base_url}/models/unload",
                    json={},
                    headers={"Content-Type": "application/json"}
                )
                if res.status_code in (200, 204):
                    logger.info("VLM model successfully unloaded from GPU VRAM.")
                    return True
        except Exception as e:
            logger.warning(f"Unload attempt {attempt + 1} failed: {e}")
    logger.error("CRITICAL: Failed to unload VLM model. GPU VRAM might cause CUDA OOM in ComfyUI!")
    return False
```

### 2.2 エンドポイント仕様 (`POST /api/speech/scene-draft`)
```python
# backend/app/routers/speech_scene.py
@router.post("/scene-draft", response_model=schemas.SceneDraftResponse)
async def generate_scene_draft(
    payload: schemas.SceneDraftRequest,
    db: Session = Depends(get_db)
):
    interpretation, bubbles, vram_unloaded, tone_char, scene_prompt = await speech_scene_service.generate_scene_script(
        db, payload.snapshot_id, payload.target_tag_set_id, payload.explicit_preset_id
    )
    return schemas.SceneDraftResponse(
        scene_interpretation=interpretation,
        reference_tone_char=tone_char,
        reference_scene_prompt=scene_prompt,
        vram_unloaded=vram_unloaded,
        bubbles=bubbles
    )
```
