### 2.2 既存の対象コード
```python
def detect_bubble(image_path: str, roi: dict | None = None) -> dict | None:
    """
    Returns {"x": int, "y": int, "w": int, "h": int, "confidence": float} or None.
    roi: {"x": int, "y": int, "w": int, "h": int} or None (全画像検索)
    """
    img = cv2.imread(str(image_path))
    if img is None:
        return None
    if roi:
        x, y, w, h = roi["x"], roi["y"], roi["w"], roi["h"]
        img = img[y:y+h, x:x+w]
        offset = (x, y)
    else:
        offset = (0, 0)
    
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    blurred = cv2.GaussianBlur(gray, (5, 5), 0)
    edges = cv2.Canny(blurred, 50, 150)
    contours, _ = cv2.findContours(edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
    
    if not contours:
        return None
    
    # 白っぽい領域の輪郭から最大のものを選択
    best = None
    best_area = 0
    for cnt in contours:
        area = cv2.contourArea(cnt)
        if area < 1000:  # ノイズ除去
            continue
        x2, y2, w2, h2 = cv2.boundingRect(cnt)
        # 矩形らしさチェック (aspect ratio)
        aspect = w2 / max(h2, 1)
        if aspect < 0.3 or aspect > 10:
            continue
        # 内部の平均輝度チェック (吹き出しは白い)
        roi_region = gray[y2:y2+h2, x2:x2+w2]
        mean_bright = roi_region.mean()
        if mean_bright < 150:
            continue
        if area > best_area:
            best_area = area
            best = (x2 + offset[0], y2 + offset[1], w2, h2)
    
    if best is None:
        return None
    x3, y3, w3, h3 = best
    img_area = img.shape[0] * img.shape[1]
    
    if roi:
        denom_area = roi["w"] * roi["h"]
    else:
        denom_area = img.shape[0] * img.shape[1]
        
    confidence = min(best_area / denom_area * 10, 1.0)
    return {"x": x3, "y": y3, "w": w3, "h": h3, "confidence": round(confidence, 2)}
```

## 3. 実装要件 (Implementation Logic)
- [x] 画像の事前クロップ処理 (`img = img[y:y+h, x:x+w]`) を削除し、常に画像全体に対してグレースケール変換・エッジ検出・輪郭抽出を実行するようにする。
- [x] ROI指定がある場合、検出された輪郭のバウンディングボックスの中心座標 `(x2 + w2/2, y2 + h2/2)` が、指定された `roi` 範囲 `[x, x+w]`, `[y, y+h]` に含まれているか判定し、含まれていない場合はスキップする処理を追加する。
- [x] 確信度（`confidence`）の計算における分母（`denom_area`）を、ROI指定時は `roi_w * roi_h`、未指定時は画像全体の面積（`img.shape[0] * img.shape[1]`）に切り替える処理を追加する。
- [x] 検出に成功した場合、座標 `x2, y2, w2, h2` は画像全体の座標となるため、従来の offset を加算する処理は不要。
