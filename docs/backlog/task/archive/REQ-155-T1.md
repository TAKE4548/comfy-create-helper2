### 2.1 関連型定義・パーサー関数 (`useChipParser.ts`)
```typescript
export interface ParsedTagWeight {
  baseText: string;
  weight: number | null;
}

export function parseTagWeight(text: string): ParsedTagWeight {
  const trimmed = text.trim();
  const match = trimmed.match(/^\((.+):([0-9.]+)\)$/);
  if (match) {
    const baseText = match[1].trim();
    const weightNum = parseFloat(match[2]);
    if (!isNaN(weightNum)) {
      return { baseText, weight: weightNum };
    }
  }
  return { baseText: trimmed, weight: null };
}

export function formatTagWeight(baseText: string, weight: number | null): string {
  const trimmedBase = baseText.trim();
  if (weight === null || isNaN(weight) || weight === 1) {
    return trimmedBase;
  }
  const rounded = parseFloat(weight.toFixed(2));
  if (rounded === 1) {
    return trimmedBase;
  }
  return `(${trimmedBase}:${rounded})`;
}
```

#### 2. 単語確定・キー操作ハンドラの実装
```typescript
  const handleWordSubmit = () => {
    const trimmed = wordInput.trim();
    if (!trimmed) {
      // 空文字の場合は元の単語にロールバック
      setWordInput(baseText);
      return;
    }
    if (trimmed !== baseText) {
      const { weight: currentWeight } = parseTagWeight(text);
      const newFormatted = formatTagWeight(trimmed, currentWeight);
      onEdit(id, newFormatted);
    }
  };

  const handleWordKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
    e.stopPropagation();
    if (e.key === 'Enter') {
      e.preventDefault();
      handleWordSubmit();
    } else if (e.key === 'Escape') {
      e.preventDefault();
      setWordInput(baseText);
      setIsMenuOpen(false);
    }
  };
```
