```typescript
// ファイルサイズ・構文バリデーション (最大10MB)
const MAX_FILE_SIZE = 10 * 1024 * 1024 // 10MB
const handleFileUpload = (file: File) => {
  if (file.size > MAX_FILE_SIZE) {
    setError('ファイルサイズが上限（10MB）を超えています')
    return
  }
  const reader = new FileReader()
  reader.onload = (e) => {
    try {
      const parsed = JSON.parse(e.target?.result as string)
      if (!parsed || typeof parsed !== 'object') throw new Error('Invalid JSON structure')
      setParsedWorkflow(parsed)
      setError(null)
    } catch {
      setError('ワークフローJSONの構文または構造が無効です')
    }
  }
  reader.readAsText(file)
}

// 逆引き選択時の非破壊ガード
const handleNodeSelect = (classType: string, title: string, paramName: string) => {
  if (hasExistingFormValues) {
    if (!window.confirm('既存の入力値を上書きしますか？')) return
  }
  onSelectNode({ classType, title, paramName })
  onClose()
}
```
