```typescript
    queuePrompt: async (
      workflow: Record<string, unknown>,
      metadata?: {
        paramValues: Record<string, string>;
        appliedAssets: Record<string, AppliedAsset[]>;
      }
    ) => {
      if (!comfyUrl) return null
      const activeUrl = resolveComfyUrl(comfyUrl)
      try {
        const res = await fetch(`${activeUrl}/prompt`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ prompt: workflow, client_id: get().clientId }),
        })
        if (!res.ok) throw new Error(`ComfyUI returned ${res.status}`)
        const data = await res.json()
        const promptId = data.prompt_id ?? null
        set({ isGenerating: true, previewImage: null, progress: null, currentPromptId: promptId, activeClientId: get().clientId })
        if (promptId) {
          localStorage.setItem(LS_KEYS.CURRENT_PROMPT_ID, promptId)
          // キャッシュ保存
          const metaKey = `prompt-surgeon-meta-${promptId}`
          const promptMetadata = {
            paramValues: metadata?.paramValues ?? get().paramValues,
            appliedAssets: metadata?.appliedAssets ?? get().appliedAssets,
            timestamp: Date.now()
          }
          localStorage.setItem(metaKey, JSON.stringify(promptMetadata))

          // 古いキャッシュの自動削除 (最大200件制限)
          try {
            const keys = Object.keys(localStorage).filter(k => k.startsWith('prompt-surgeon-meta-'))
            if (keys.length > 200) {
              const items = keys.map(k => {
                try {
                  const val = JSON.parse(localStorage.getItem(k) || '{}')
                  return { key: k, timestamp: val.timestamp || 0 }
                } catch {
                  return { key: k, timestamp: 0 }
                }
              })
              items.sort((a, b) => a.timestamp - b.timestamp)
              const toDelete = items.slice(0, items.length - 200)
              toDelete.forEach(item => localStorage.removeItem(item.key))
            }
          } catch (e) {
            console.error('Failed to prune prompt metadata:', e)
          }
        }
        await get().fetchQueue()
        return promptId
      } catch (e) {
        console.error('Failed to queue prompt:', e)
        return null
      }
    }
```

## 3. 実装要件 (Implementation Logic)
- [x] `useGenerationStore.ts` で `queuePrompt` の引数を拡張し、パラメータと適用アセット（`metadata`）を受け取り `localStorage` にキャッシュするロジック（自動プルーニング付き）を実装。
- [x] `useGenerationStore.ts` の `reconcileState` (L269) と WebSocket の `executed` (L442) の画像URL生成ロジックで、`Date.now()` の代わりに `promptId` (または `prompt_id`) をクエリパラメータ `t` に指定するように変更。
- [x] `useGenerateAction.ts` から `queuePrompt` を呼び出す際、現在適用されているパラメータ・アセット（単一生成の場合はUIの状態、Multi-Set生成の場合は各Setの状態）を第2引数として渡すように変更。
- [x] `useSnapshotActions.ts` でスナップショット保存時、画像URLのパラメータ `t` (promptId) を解析し、キャッシュメタデータからアセット情報を復元して保存用の API リクエストを組み立てるように修正（キャッシュが無い場合はUI状態へフォールバック）。
