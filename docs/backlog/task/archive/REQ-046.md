### 2.1 HistoryModalPreview.tsx
```typescript
import React, { useState, useEffect } from 'react'
import { X, Loader2, Info, Maximize2 } from 'lucide-react'

export interface HistoryModalPreviewProps {
  isOpen: boolean
  onClose: () => void
  images: string[]
  currentIndex: number
  onChangeIndex: (index: number) => void
  snapshotIds?: (number | null)[]
}

export function HistoryModalPreview({
  isOpen,
  onClose,
  images,
  currentIndex,
  onChangeIndex,
  snapshotIds,
}: HistoryModalPreviewProps) {
  const [viewMode, setViewMode] = useState<'full' | 'info'>('full')
  
  useEffect(() => {
    if (isOpen) setViewMode('full')
  }, [isOpen])

  const activeSnapshotId = snapshotIds && currentIndex >= 0 && currentIndex < snapshotIds.length
    ? snapshotIds[currentIndex]
    : null

  const activeImage = images[currentIndex]

  return (
    <div className="fixed inset-0 z-50 bg-black/95 flex flex-col" onClick={onClose}>
      <div className="relative h-12 flex items-center justify-end px-4" onClick={(e) => e.stopPropagation()}>
        <button onClick={onClose} className="absolute top-4 right-4 z-50">
          <X size={20} />
        </button>
      </div>

      <div className={viewMode === 'full' ? "w-full flex-1 flex items-center justify-center bg-black/40 relative cursor-pointer" : "w-full flex items-center justify-center bg-black/40 relative cursor-pointer"} style={viewMode === 'info' ? { height: '60vh' } : undefined}>
        <img src={activeImage} className="max-w-full max-h-full object-contain" />
        {activeSnapshotId && (
          viewMode === 'full' ? (
            <button onClick={(e) => { e.stopPropagation(); setViewMode('info') }} className="absolute bottom-4 right-4 z-10 p-3 rounded-full bg-black/60 text-white">
              <Info size={20} />
            </button>
          ) : (
            <button onClick={(e) => { e.stopPropagation(); setViewMode('full') }} className="absolute bottom-4 right-4 z-10 p-3 rounded-full bg-black/60 text-white">
              <Maximize2 size={20} />
            </button>
          )
        )}
      </div>

      {viewMode === 'info' && (
        <div className="flex-1 overflow-y-auto p-5 bg-surface border-t">
          {/* Metadata Display */}
        </div>
      )}
    </div>
  )
}
```

### 2.2 GeneratePage.tsx
```typescript
import React, { useState, useMemo } from 'react'

export function GeneratePage() {
  const [isModalPreviewOpen, setIsModalPreviewOpen] = useState(false)
  const [modalPreviewIndex, setModalPreviewIndex] = useState(0)

  // Zustand Store / Hooks Mock
  const previewImage = "some_url"
  const selectedImg = "some_url"
  const sortedDisplayImages: string[] = ["img1", "img2"]
  const snapshotIdsForDisplay: (number | null)[] = [1, null]

  const handleOpenModalPreview = () => {
    const currentImg = previewImage || selectedImg
    if (!currentImg) return
    const idx = sortedDisplayImages.indexOf(currentImg)
    setModalPreviewIndex(idx >= 0 ? idx : 0)
    setIsModalPreviewOpen(true)
  }

  return (
    <div>
      {/* Desktop Preview */}
      <div onClick={handleOpenModalPreview} className="cursor-pointer">
        <img src={previewImage || selectedImg} className="w-full h-full object-contain" />
        <a href="#" download onClick={(e) => e.stopPropagation()}>Download</a>
      </div>

      {/* Mobile Preview */}
      <div onClick={handleOpenModalPreview} className="cursor-pointer">
        <img src={previewImage || selectedImg} className="w-full h-full object-contain" />
        <a href="#" download onClick={(e) => e.stopPropagation()}>Download</a>
      </div>

      <HistoryModalPreview
        isOpen={isModalPreviewOpen}
        onClose={() => setIsModalPreviewOpen(false)}
        images={sortedDisplayImages}
        currentIndex={modalPreviewIndex}
        onChangeIndex={setModalPreviewIndex}
        snapshotIds={snapshotIdsForDisplay}
      />
    </div>
  )
}
```

## 3. 実装要件 (Implementation Logic)
- [x] **HistoryModalPreview.tsx の変更**:
  - `viewMode` 状態 (`'full' | 'info'`) を `useState` で追加します。
  - `isOpen` が `true` に変化したとき、`viewMode` を `'full'` にリセットする `useEffect` を追加します。
  - `viewMode === 'full'` の場合、下部のメタ情報パネルを非表示にし、画像表示エリアを `flex-1 w-full` に設定します。画像アスペクト比維持のため `object-contain` クラスが維持されていることを確認します。
  - `activeSnapshotId` が存在する場合、画像表示エリアの右下に絶対配置でトグルボタンを配置します。
    - `viewMode === 'full'` 時：`<Info size={20} />` を表示し、クリックで `'info'` に切り替える。
    - `viewMode === 'info'` 時：`<Maximize2 size={20} />` を表示し、クリックで `'full'` に切り替える。
- [x] **GeneratePage.tsx の変更**:
  - メインプレビュー画像（デスクトップ、およびモバイル）コンテナに対して、クリック時にモーダルプレビューを開くようにします。
  - 現在表示中の画像 (`previewImage || selectedImg`) が `sortedDisplayImages` に含まれるインデックスを特定し、`modalPreviewIndex` に設定した上で `setIsModalPreviewOpen(true)` を呼び出します。
  - 子要素のボタン（ダウンロード、左右送り）に `e.stopPropagation()` を適用し、クリック時のイベントバブリングによる予期しないモーダルオープンを防ぎます。
