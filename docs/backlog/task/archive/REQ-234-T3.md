### 3.3 `frontend/src/test/setup.ts` 変更仕様
既存の `import '@testing-library/jest-dom';` に加えて MSW ライフサイクルを設定する:
```typescript
import '@testing-library/jest-dom';
import { server } from './mocks/server';

beforeAll(() => {
  // onUnhandledRequest: 'bypass' により未定義リクエストはスルーし、既存テストのベタモックと安全に共存
  server.listen({ onUnhandledRequest: 'bypass' });
});

afterEach(() => {
  server.resetHandlers();
});

afterAll(() => {
  server.close();
});
```
