### 2.1 課題と移行仕様
`BaseImageSelectModal.tsx` では以下の生 `useEffect` が存在し、他コンポーネントの React Query キャッシュと分断されている：
- `workflowsApi.list()`
- `lorasApi.list()`
- `tagSetsApi.list()`
- `albumsApi.list()`
- `albumsApi.get(selectedAlbumId)`
- `snapshotsApi.list(...)`

これを以下のように React Query の `useQuery` に移行する：
```typescript
import { useQuery } from '@tanstack/react-query'
import { STALE_TIMES } from '../constants/staleTimes'

// Workflows, Loras, TagSets
const { data: workflows = [] } = useQuery({
  queryKey: ['workflows'],
  queryFn: workflowsApi.list,
  staleTime: STALE_TIMES.WORKFLOWS,
})
const { data: loras = [] } = useQuery({
  queryKey: ['loras'],
  queryFn: lorasApi.list,
  staleTime: STALE_TIMES.LORAS,
})
const { data: tagSets = [] } = useQuery({
  queryKey: ['tag-sets'],
  queryFn: tagSetsApi.list,
  staleTime: STALE_TIMES.TAG_SETS,
})

// Snapshots
const { data: snapshots = [], isLoading: isLoadingSnapshots } = useQuery({
  queryKey: ['snapshots', { workflow_id: selectedWorkflowId, lora_id: selectedLoraId, tag_set_id: selectedTagSetId }],
  queryFn: () => snapshotsApi.list({
    workflow_id: selectedWorkflowId ?? undefined,
    lora_id: selectedLoraId ?? undefined,
    tag_set_id: selectedTagSetId ?? undefined,
  }),
  staleTime: STALE_TIMES.SNAPSHOTS,
})

// Albums
const { data: albums = [], isLoading: isLoadingAlbums } = useQuery({
  queryKey: ['albums'],
  queryFn: () => albumsApi.list() as Promise<Album[]>,
  staleTime: STALE_TIMES.ALBUMS,
  enabled: activeTab === 'albums',
})

// Album Detail
const { data: currentAlbum = null, isLoading: isLoadingAlbumDetail } = useQuery({
  queryKey: ['album', selectedAlbumId],
  queryFn: () => albumsApi.get(selectedAlbumId!),
  staleTime: STALE_TIMES.SNAPSHOT_DETAIL,
  enabled: activeTab === 'albums' && selectedAlbumId != null,
})
```
