```python
import logging
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session
from app.database import get_db
from app.schemas.speech import SceneDraftRequest, SceneDraftResponse
from app.services import speech_scene_service
router = APIRouter(prefix="/api/speech", tags=["speech-scene"])
```

```python
@router.post("/scene-draft", response_model=SceneDraftResponse)
async def generate_scene_draft(payload: SceneDraftRequest, db: Session = Depends(get_db)):
    try:
        (interpretation, bubbles, vram_unloaded, tone_samples,
         selected_similar_scene_id, candidate_similar_scenes
        ) = await speech_scene_service.generate_scene_script(
            db, payload.snapshot_id, payload.target_tag_set_id,
            payload.explicit_preset_id, payload.scene_hint,
            override_similar_scene_snapshot_id=payload.override_similar_scene_snapshot_id,
        )
        return SceneDraftResponse(
            scene_interpretation=interpretation, vram_unloaded=vram_unloaded,
            bubbles=bubbles, tone_samples=tone_samples,
            selected_similar_scene_id=selected_similar_scene_id,
            candidate_similar_scenes=candidate_similar_scenes,
        )
```

```python
from fastapi.testclient import TestClient
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
from app.main import app
from app.database import Base, get_db
engine = create_engine('sqlite:///:memory:', connect_args={'check_same_thread': False})
connection = engine.connect()
TestingSessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=connection)
def override_get_db():
    db = TestingSessionLocal()
    try:
        yield db
    finally:
        db.close()
app.dependency_overrides[get_db] = override_get_db
Base.metadata.create_all(bind=connection)
client = TestClient(app)
# テスト後: Base.metadata.drop_all(bind=connection)、overrideを元に戻す
```
