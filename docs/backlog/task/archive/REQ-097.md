```python
class Workflow(Base):
    __tablename__ = "workflows"

    id = Column(Integer, primary_key=True, index=True)
    name = Column(String, unique=True, index=True, nullable=False)
    json_body = Column(Text, nullable=False, default="{}")
    default_applied_assets = Column(Text, nullable=True)

    mappings = relationship("WorkflowMapping", back_populates="workflow", cascade="all, delete-orphan")
    presets = relationship("WorkflowPreset", back_populates="workflow", cascade="all, delete-orphan")
    snapshots = relationship("Snapshot", back_populates="workflow", cascade="all, delete-orphan")
```
