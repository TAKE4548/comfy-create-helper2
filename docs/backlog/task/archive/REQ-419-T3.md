## 6. `variantDraft.ts` の公開 API
```ts
export type DefinitionKind = 'ruleSet' | 'selector'
export interface EntryDraft { class_type: string; param_name: string; priority: number }
export interface PatternDraft { pattern: string; match_type: 'exact' | 'contains'; priority: number }
export interface RuleSetDraft { key: string; id: number | null; name: string; entries: EntryDraft[]; saved: { name: string; entries: EntryDraft[] } | null; referencedRoles: string[] }
export interface SelectorDraft { key: string; id: number | null; name: string; patterns: PatternDraft[]; saved: { name: string; patterns: PatternDraft[] } | null; referencedRoles: string[] }
export interface VariantDraft { rowKey: string; id: number | null; stageId: number; stageKey: string; stageName: string; stageSortOrder: number; role: string; isEnabled: boolean; ruleSetKey: string | null; selectorKey: string | null; savedRole: string | null }
export type SharedChoice = 'update_shared' | 'duplicate_for_variant'
export interface MasterDraftState { variants: VariantDraft[]; ruleSets: Record<string, RuleSetDraft>; selectors: Record<string, SelectorDraft>; editOrigin: Record<string, string>; choices: Record<string, SharedChoice> }

export function buildMasterDraft(master: MappingMasterV2, stages: StageMaster[], ruleSets: BindingRuleSet[], selectors: NodeSelector[]): MasterDraftState
export function sortVariantDrafts(v: VariantDraft[]): VariantDraft[]
export function sortedEntries(e: EntryDraft[]): EntryDraft[]
export function sortedPatterns(p: PatternDraft[]): PatternDraft[]
export function isDefinitionDirty(d: RuleSetDraft | SelectorDraft): boolean
export function findOtherReferrers(state: MasterDraftState, kind: DefinitionKind, defKey: string): string[]
export function pendingSharedChoices(state: MasterDraftState): Array<{ kind: DefinitionKind; defKey: string; originRowKey: string; otherRoles: string[] }>
export function validateVariantDrafts(state: MasterDraftState, knownRuleSetNames: string[], knownSelectorNames: string[]): Record<string, string[]>
export function generateCopyName(baseName: string, role: string, existingNames: string[]): string
export function buildAtomicSaveRequest(state: MasterDraftState, master: { name: string; input_type: string; group_id: number | null; default_enabled: boolean }, existingNames: { ruleSet: string[]; selector: string[] }): MasterAtomicSaveRequest
export function buildScanPreviewRequest(state: MasterDraftState, workflowId: number): ScanPreviewRequest
export function mapServerValidationErrors(err: unknown): { rows: Record<string, string[]>; general: string[] } | null
export const STAGE_LABELS: Record<'exact' | 'contains' | 'no_selector', string>  // 完全一致 / 部分一致 / セレクタなし
```
規則:
- `buildMasterDraft`: `rowKey="v:<id>"`、`role = variant.role ?? variant.computed_role`、`savedRole` も同値、stage 情報は `stages`（無ければ variant の stage_* と sort_order=0）。定義キー `rs:<id>`/`ns:<id>`。新規行は呼び出し側が `new:<n>`、新規定義は `rs:new:<n>`/`ns:new:<n>`。
- `sortVariantDrafts`: stageSortOrder→stageId→rowKey 昇順。`sortedEntries`/`sortedPatterns`: priority 昇順（安定）。
- `isDefinitionDirty`: `saved===null` は false。name と内容（順序込み）を比較。
- `findOtherReferrers`: `(referencedRoles − この Master の保存済み Variant の savedRole) ∪ { v.role | v.rowKey ≠ editOrigin[defKey] かつ v が defKey を参照 }`、重複除去・昇順。
- `pendingSharedChoices`: `id≠null` かつ dirty かつ他参照元1件以上かつ `choices[defKey]` 未設定。
- `validateVariantDrafts`: 同一 stageId の全行、同一の非 null selectorKey の全行にエラー（null は対象外）。ruleSetKey null はエラー。参照される下書き定義の name が空、または既存名（自分の saved.name を除く）・他下書きと重複ならその定義を参照する全行にエラー。
- `generateCopyName`: `${baseName} (${role})`、既存なら ` 2`, ` 3`…。
- `buildAtomicSaveRequest`:
  - 参照される新規定義 → `{ref: defKey, op: 'create', id: null, name, entries|patterns}`。
  - choice=`duplicate_for_variant` → `{ref: `${defKey}:copy`, op: 'create', name: generateCopyName(...)}`（内容は下書き）。編集元行だけ `{ref: `${defKey}:copy`}` を参照し、他行は `{id}` のまま。元定義の update は出さない。
  - choice=`update_shared`、または他参照元なしで dirty → `{ref: defKey, op: 'update', id, name, ...}`。
  - 未参照の新規定義は出さない。variants は全行（順序は `sortVariantDrafts`）。`rule_set`/`selector` は永続化済みなら `{id}`、新規・コピーなら `{ref}`。selectorKey null は `selector: null`。entries/patterns は priority 順。
- `buildScanPreviewRequest`: 全行、`client_key=rowKey`、entries/patterns は下書き（priority 順）、selectorKey null は `patterns: null`。
- `mapServerValidationErrors`: `Error.message` の `API error 422: {...}` から `detail.code==='MASTER_SAVE_VALIDATION_FAILED'` の `errors[]` を、`row_key` があれば `rows[row_key]` へ、無ければ `general` へ。形が違えば null。
