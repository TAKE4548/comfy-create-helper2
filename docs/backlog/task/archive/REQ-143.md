### 2.8 `BaseImageSelectModal.tsx` の完全な実装仕様
```tsx
import React, { useState, useEffect } from 'react'
import { X, Image as ImageIcon, Loader2 } from 'lucide-react'
import { snapshotsApi } from '../api/client'
import type { SnapshotSummary } from '../api/types'

export type CloseCallback = () => void
export type SelectSnapshotCallback = (snapshot: SnapshotSummary) => void

export interface BaseImageSelectModalProps {
  workflowId: number
  currentSnapshotId: number
  onClose: CloseCallback
  onSelect: SelectSnapshotCallback
}

export function BaseImageSelectModal({
  workflowId,
  currentSnapshotId,
  onClose,
  onSelect,
}: BaseImageSelectModalProps) {
  const [snapshots, setSnapshots] = useState<SnapshotSummary[]>([])
  const [isLoading, setIsLoading] = useState<boolean>(false)

  useEffect(() => {
    setIsLoading(true)
    snapshotsApi.list({ workflow_id: workflowId })
      .then((data: SnapshotSummary[]) => {
        setSnapshots(data || [])
      })
      .catch((err: unknown) => {
        console.error('Failed to list snapshots:', err)
      })
      .finally(() => {
        setIsLoading(false)
      })
  }, [workflowId])

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/75 backdrop-blur-sm p-4" onClick={onClose}>
      <div
        className="bg-[#18181C] border border-[rgba(255,255,255,0.08)] rounded-xl w-full max-w-3xl max-h-[80vh] flex flex-col overflow-hidden shadow-2xl"
        onClick={(e) => e.stopPropagation()}
      >
        {/* Header */}
        <div className="flex items-center justify-between px-5 py-3.5 border-b border-[rgba(255,255,255,0.08)]">
          <div className="flex items-center gap-2">
            <ImageIcon size={18} className="text-accent-green" />
            <h3 className="text-base font-semibold text-white">生成履歴からベース画像を差し替え</h3>
          </div>
          <button onClick={onClose} className="text-zinc-400 hover:text-white p-1 rounded-lg hover:bg-white/5">
            <X size={18} />
          </button>
        </div>

        {/* Body */}
        <div className="p-4 overflow-y-auto flex-1 min-h-[300px]">
          {isLoading ? (
            <div className="flex items-center justify-center h-48 text-zinc-400 gap-2">
              <Loader2 className="animate-spin" size={24} />
              <span>生成履歴を読み込み中...</span>
            </div>
          ) : snapshots.length === 0 ? (
            <div className="text-center py-12 text-zinc-500 text-xs">
              利用可能なスナップショットが見つかりませんでした
            </div>
          ) : (
            <div className="grid grid-cols-3 sm:grid-cols-4 md:grid-cols-5 gap-3">
              {snapshots.map((s) => (
                <button
                  key={s.id}
                  type="button"
                  onClick={() => onSelect(s)}
                  className={`group relative rounded-lg overflow-hidden border transition-all text-left aspect-square bg-zinc-900 flex flex-col ${
                    s.id === currentSnapshotId
                      ? 'border-accent-green ring-2 ring-accent-green/30'
                      : 'border-white/10 hover:border-white/30'
                  }`}
                >
                  <img
                    src={s.thumbnail_path || s.image_path || ''}
                    alt={`Snapshot ${s.id}`}
                    className="w-full h-full object-cover group-hover:scale-105 transition-transform"
                  />
                  {s.id === currentSnapshotId && (
                    <span className="absolute top-1 right-1 px-1.5 py-0.5 rounded bg-accent-green text-black font-bold text-[9px]">
                      現在
                    </span>
                  )}
                  <span className="absolute bottom-0 inset-x-0 bg-black/70 px-1.5 py-0.5 text-[10px] text-zinc-300 truncate">
                    #{s.id}
                  </span>
                </button>
              ))}
            </div>
          )}
        </div>
      </div>
    </div>
  )
}
```

### 2.9 `SpeechBubbleEditorModal.tsx` での画像ロード・再編集復元・差し替え・保存仕様
```typescript
// A. 画像ロード（currentImageUrl 変化時にトリガー）
useEffect(() => {
  if (!currentImageUrl) return
  const img = new Image()
  img.src = currentImageUrl
  img.crossOrigin = 'anonymous'
  img.onload = () => {
    setImageObj(img)
    const maxW = typeof window !== 'undefined' ? Math.min(500, window.innerWidth - 48) : 500
    const maxH = 500
    let w = img.naturalWidth
    let h = img.naturalHeight
    if (w > maxW || h > maxH) {
      const ratio = Math.min(maxW / w, maxH / h)
      w = Math.round(w * ratio)
      h = Math.round(h * ratio)
    }
    setCanvasDimensions({ width: w, height: h })
  }
}, [currentImageUrl])

// B. スナップショット初期化と再編集復元
useEffect(() => {
  let active = true
  const initEditor = async () => {
    try {
      const fullSnapshot = await snapshotsApi.get(snapshot.id)
      const params = fullSnapshot.params as Record<string, any> | undefined
      const savedBubbles = params?.speech_bubbles as SpeechBubble[] | undefined
      const baseSnapId = (params?.base_snapshot_id as number) || fullSnapshot.source_snapshot_id || fullSnapshot.id
      const baseImgPath = (params?.base_image_path as string) || fullSnapshot.image_path

      if (!active) return

      if (savedBubbles && Array.isArray(savedBubbles) && savedBubbles.length > 0) {
        setBubbles(savedBubbles)
        setSelectedBubbleId(savedBubbles[0]?.id || null)
        setCurrentSourceSnapshotId(baseSnapId)
        setCurrentImageUrl(baseImgPath || fullSnapshot.image_path || '')
      } else {
        setCurrentSourceSnapshotId(fullSnapshot.id)
        setCurrentImageUrl(fullSnapshot.image_path || '')
        // 初回自動検出
        setIsDrafting(true)
        try {
          const res = await snapshotsApi.detectBubble(fullSnapshot.id)
          if (active && res.bounding_box) {
            const newBubble: SpeechBubble = {
              id: `bubble-${Date.now()}`,
              x: res.bounding_box.x,
              y: res.bounding_box.y,
              w: res.bounding_box.w,
              h: res.bounding_box.h,
              text: '',
              shape: 'oval',
              orientation: 'vertical',
              fontFamily: selectedFont || 'sans-serif',
              fontSize: 0,
              textColor: 'rgba(0, 0, 0, 1)',
              backgroundColor: 'rgba(255, 255, 255, 0.95)',
              borderColor: 'rgba(0, 0, 0, 1)',
              borderWidth: 2,
              tail: {
                x: res.bounding_box.x + res.bounding_box.w / 2 - 20,
                y: res.bounding_box.y + res.bounding_box.h + 30
              }
            }
            setBubbles([newBubble])
            setSelectedBubbleId(newBubble.id)
          }
        } catch (e) {
          console.error(e)
        } finally {
          if (active) setIsDrafting(false)
        }
      }
    } catch (err) {
      console.error('Failed to init editor:', err)
      setCurrentImageUrl(snapshot.image_path || '')
    }
  }
  initEditor()
  return () => { active = false }
}, [snapshot.id])

// C. ベース画像差し替えハンドラー
// ユーザーがスナップショットを選択した際、対象のIDと画像パスをセットしてモーダルを閉じる
const handleSelectNewBaseImage = (selectedSnapshot: SnapshotSummary) => {
  if (selectedSnapshot.image_path) {
    setCurrentSourceSnapshotId(selectedSnapshot.id)
    setCurrentImageUrl(selectedSnapshot.image_path)
    setIsImageSelectOpen(false)
  }
}

// D. メタデータ付き新規保存
const handleSave = async () => {
  if (!imageObj || bubbles.length === 0) return

  setIsSaving(true)
  try {
    const offCanvas = document.createElement('canvas')
    offCanvas.width = imageObj.naturalWidth
    offCanvas.height = imageObj.naturalHeight
    const offCtx = offCanvas.getContext('2d')
    if (!offCtx) throw new Error('Offscreen canvas context generation failed')

    offCtx.drawImage(imageObj, 0, 0, offCanvas.width, offCanvas.height)
    bubbles.forEach((bubble) => {
      drawSpeechBubble(offCtx, bubble, false, 1, 1, activeFontFamily)
    })

    const blob = await new Promise<Blob | null>((resolve) => offCanvas.toBlob(resolve, 'image/png'))
    if (!blob) throw new Error('Canvas blob generation failed')

    const uploadRes = await snapshotsApi.upload(blob)
    const parentSnapshot = await snapshotsApi.get(currentSourceSnapshotId)

    const updatedParams = {
      ...(parentSnapshot.params || {}),
      speech_bubbles: bubbles,
      base_snapshot_id: currentSourceSnapshotId,
      base_image_path: currentImageUrl,
    }

    await snapshotsApi.create({
      workflow_id: parentSnapshot.workflow_id,
      image_path: uploadRes.image_path,
      params: updatedParams,
      source_snapshot_id: parentSnapshot.id,
    })

    onSaved()
    onClose()
  } catch (err: any) {
    console.error(err)
    alert('画像の保存に失敗しました: ' + err.message)
  } finally {
    setIsSaving(false)
  }
}
```
