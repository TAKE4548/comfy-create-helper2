```tsx
              value={textValue}
              onChange={(e) => {
```

### 2.3 変更後の修正方針
1. `isTextMode` が `true` の場合、外部からの `chipsProp` の参照変化があっても、ユーザー自身が編集中の場合は `setTextValue(chipsToString(chipsProp))` を呼び出さない（テキストエリアの入力を上書き消去しない）。
   - `isTextMode` 中は、`prevChipsProp.current = chipsProp` は更新するが、`setTextValue` の強制上書きをガードする。
2. `textarea.onChange` において、`setTextValue(e.target.value)` に加え、親コンポーネントへ生のテキスト入力を即座に伝達するため `onChange?.(e.target.value)` を呼び出し、同時にパースしたチップを `onChipsChange?.(newChips)` で親に通知する。
3. チップモードからテキストモードへの切り替え時は、現在のチップ状態からテキストを正しく復元し、テキストモードからチップモードへの切り替え時は `parseToChips(textValue)` でチップを確定する。

### 2.4 テスト追加ケース (`TagChipEditor.test.tsx`)

```tsx
  it('does not overwrite or revert textarea input when chips prop is provided and re-renders occur in text mode', () => {
    let currentChips = parseToChips('apple, banana');
    const mockOnChipsChange = vi.fn((chips) => { currentChips = chips; });
    const { rerender } = render(
      <TagChipEditor
        value="apple, banana"
        chips={currentChips}
        onChipsChange={mockOnChipsChange}
        onChange={mockOnChange}
      />
    );

    // テキストモードに切り替え
    const switchBtn = screen.getByRole('button', { name: 'テキスト入力モード切替' });
    fireEvent.click(switchBtn);

    const textarea = screen.getByRole('textbox') as HTMLTextAreaElement;
    expect(textarea.value).toBe('apple, banana');

    // 末尾にカンマとスペースを入力
    fireEvent.change(textarea, { target: { value: 'apple, banana, ' } });

    // 親の chips 更新を模倣して rerender
    rerender(
      <TagChipEditor
        value="apple, banana, "
        chips={[...currentChips]}
        onChipsChange={mockOnChipsChange}
        onChange={mockOnChange}
      />
    );

    // 末尾のカンマとスペースが巻き戻されずに維持されること
    expect(textarea.value).toBe('apple, banana, ');
  });
```
