function Wait-For-Port ([int]$port, [int]$timeoutSec = 10) {
    $elapsed = 0
    while ($elapsed -lt $timeoutSec) {
        if (Test-PortOpen $port) { return $true }
        Start-Sleep -Seconds 1
        $elapsed++
    }
    return (Test-PortOpen $port)
}
