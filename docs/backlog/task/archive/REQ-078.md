### 2.2 既存の対象コード

```typescript
export function HistoryModalPreview({
  isOpen,
  onClose,
  images,
  currentIndex,
  onChangeIndex,
  snapshotIds,
}: HistoryModalPreviewProps) {
  const [viewMode, setViewMode] = useState<'full' | 'info'>('full')
  const touchStartX = useRef<number | null>(null)
  
  const activeImage = images[currentIndex]

  // キーボードイベント監視 (ArrowLeft, ArrowRight)
  useEffect(() => {
    if (!isOpen) return
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'ArrowLeft') {
        if (currentIndex > 0) onChangeIndex(currentIndex - 1)
      } else if (e.key === 'ArrowRight') {
        if (currentIndex < images.length - 1) onChangeIndex(currentIndex + 1)
      }
    }
    window.addEventListener('keydown', handleKeyDown)
    return () => window.removeEventListener('keydown', handleKeyDown)
  }, [isOpen, currentIndex, images.length, onChangeIndex])

  return (
    <div className="fixed inset-0 z-[100] bg-black/95 flex flex-col" onClick={onClose}>
      {/* Header with Close Button */}
      <div className="relative h-12 flex items-center justify-end px-4" onClick={(e) => e.stopPropagation()}>
        <button
          onClick={onClose}
          className="absolute top-4 right-4 z-50 p-2 rounded-full bg-white/10 hover:bg-white/20 text-white/80 transition-colors"
          aria-label="Close preview"
        >
          <X size={20} />
        </button>
      </div>

      {/* Image Display Area */}
      <div
        className={`w-full flex items-center justify-center bg-black/40 relative cursor-pointer min-h-0 ${
          viewMode === 'full' ? 'flex-1' : ''
        }`}
        style={viewMode === 'info' ? { height: '60vh' } : undefined}
        onTouchStart={handleTouchStart}
        onTouchEnd={handleTouchEnd}
        onClick={(e) => {
          if (e.target === e.currentTarget) {
            onClose()
          }
        }}
      >
        {currentIndex > 0 && (
          <button
            onClick={(e) => { e.stopPropagation(); onChangeIndex(currentIndex - 1) }}
            className="absolute left-4 z-10 p-3 rounded-full bg-black/60 hover:bg-black/80 text-white transition-colors"
          >
            &lt;
          </button>
        )}
        <img
          src={activeImage}
          alt={`Preview ${currentIndex + 1}`}
          className="max-w-full max-h-full object-contain pointer-events-none select-none"
        />
        {currentIndex < images.length - 1 && (
          <button
            onClick={(e) => { e.stopPropagation(); onChangeIndex(currentIndex + 1) }}
            className="absolute right-4 z-10 p-3 rounded-full bg-black/60 hover:bg-black/80 text-white transition-colors"
          >
            &gt;
          </button>
        )}
      </div>
    </div>
  )
}
```

- [x] **コンテナの参照の追加**: モーダルの最外枠 `div` 要素に対して `useRef<HTMLDivElement>(null)` (例: `containerRef`) を設定し、全画面化のターゲットにする。
- [x] **全画面状態の管理**: `isFullscreen` ステート (`useState<boolean>(false)`) を定義する。
- [x] **トグル機能の実装**: 画像領域（`img` タグのラッパーまたは `img` 自体）のクリック時に全画面表示をトグルする `handleToggleFullscreen` 関数を定義する。
  ```typescript
  const handleToggleFullscreen = (e: React.MouseEvent) => {
    e.stopPropagation()
    const element = containerRef.current as HTMLElementWithFullscreen
    const doc = document as DocumentWithFullscreen
    if (!element) return

    if (!doc.fullscreenElement && !doc.webkitFullscreenElement) {
      if (element.requestFullscreen) {
        element.requestFullscreen()
      } else if (element.webkitRequestFullscreen) {
        element.webkitRequestFullscreen()
      }
    } else {
      if (doc.exitFullscreen) {
        doc.exitFullscreen()
      } else if (doc.webkitExitFullscreen) {
        doc.webkitExitFullscreen()
      }
    }
  }
  ```
- [x] **解除検知とクローズの同期 (Bパターン)**: `useEffect` を用い、`document` に `fullscreenchange` および `webkitfullscreenchange` イベントリスナーを登録する。
  - 全画面が解除された（`document.fullscreenElement` および `document.webkitFullscreenElement` が共に `null` である）ことを検知した際、プレビューモーダルが開いている状態 (`isOpen`) ならば `onClose()` を呼び出してプレビューモーダルも同時に閉じる。
- [x] **閉じるボタンの動作変更**: 全画面中に閉じるボタンが押された場合、全画面を終了する処理 (`document.exitFullscreen()` 等) を実行する。これにより `fullscreenchange` の副作用を通じて自動的にモーダルも閉じられるようにする。
- [x] **スタイル調整**: 全画面表示時には、画像が `object-fit: contain` となり、アスペクト比を保ったまま画面内に最大限大きく表示されるよう、コンテナと `img`要素の Tailwind クラスを調整する。
