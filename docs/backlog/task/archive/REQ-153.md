### 2.4 ハイブリッドカードラッパー実装例 (`SortableTagSetCard`, `SortableLoraCard`, `SortableItem`)
```tsx
function SortableTagSetCard({ id, disabled, children }: SortableTagSetCardProps) {
  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id, disabled });

  const mouseListeners: React.DOMAttributes<HTMLDivElement> = {
    onMouseDown: listeners?.onMouseDown,
    onKeyDown: listeners?.onKeyDown,
  };

  const touchHandleListeners: DragHandleListeners = {
    onTouchStart: listeners?.onTouchStart,
    onMouseDown: listeners?.onMouseDown,
  };

  const style: React.CSSProperties = {
    transform: CSS.Transform.toString(transform),
    transition,
    opacity: isDragging ? 0.5 : 1,
    zIndex: isDragging ? 50 : 1,
  };

  return (
    <div ref={setNodeRef} style={style} {...attributes} {...mouseListeners}>
      {children({ attributes, listeners: touchHandleListeners, isDragging })}
    </div>
  );
}
```
