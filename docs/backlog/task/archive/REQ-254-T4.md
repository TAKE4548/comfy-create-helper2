### `frontend/src/api/types.ts`
```typescript
export interface NodeTopologySlot {
  slot: string
  source_node_id?: string
  source_class?: string
  source_title?: string
  target_node_id?: string
  target_class?: string
  target_title?: string
}

export interface NodeTopology {
  node_id: string
  title: string
  class_type: string
  inputs: NodeTopologySlot[]
  outputs: NodeTopologySlot[]
}

export interface NodeCandidate {
  node_id: string
  title: string
  class_type: string
  score: number
}

export interface RoleMigration {
  master_id: number
  role: string
  display_name: string
  old_node_id: string
  new_node_id: string | null
  confidence: number
  reason: string
  status: 'ok' | 'renamed' | 'reindexed' | 'substituted' | 'false_reuse' | 'not_found'
  candidates: NodeCandidate[]
  old_topology?: NodeTopology | null
  new_topology?: NodeTopology | null
}

export interface DiffAnalyzeResponse {
  summary: {
    added: number
    removed: number
    renamed: number
    reindexed: number
    false_reuse: number
  }
  role_migrations: RoleMigration[]
  bypass_warnings: Array<{ profile_name: string; missing_node_id: string; suggested_node_id?: string }>
  unmapped_new_nodes: Array<{ node_id: string; title: string; class_type: string }>
}
```

### `frontend/src/api/client.ts`
```typescript
export const workflowsApi = {
  // 既存メソッド...
  diffAnalyze: (workflowId: number, newJsonBody: Record<string, unknown>) =>
    request<DiffAnalyzeResponse>(`/workflows/${workflowId}/diff-analyze`, {
      method: 'POST',
      body: JSON.stringify({ new_json_body: newJsonBody }),
    }),
  rollback: (workflowId: number) =>
    request<{ message: string }>(`/workflows/${workflowId}/rollback`, { method: 'POST' }),
  getHistoryStatus: (workflowId: number) =>
    request<{ has_rollback: boolean; history_count: number; latest_backup_at?: string }>(
      `/workflows/${workflowId}/history-status`
    ),
}
```
