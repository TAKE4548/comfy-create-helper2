// frontend/src/utils/promptUtils.ts
export function addAssetToPrompt(
  prompt: string,
  tags: string[],
  loraPromptString?: string,
): string {
  const currentVal = prompt || '';
  const tagsToAdd = [...tags];
  if (loraPromptString) {
    tagsToAdd.push(loraPromptString);
  }
  const toAddStr = tagsToAdd.join(', ');
  if (!currentVal) return toAddStr;
  return `${currentVal}, ${toAddStr}`;
}

// frontend/src/api/client.ts
export const workflowPresetsApi = {
  update: (id: number, data: WorkflowPresetUpdate): Promise<WorkflowPreset> =>
    fetchJson<WorkflowPreset>(`/api/workflow-presets/${id}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    }),
};

// frontend/src/pages/GeneratePage/hooks/useGeneratePageState.ts (既存の handleBulkEditApply)
const handleBulkEditApply = async (payload: BulkEditPayload) => {
  if (!presets || bulkTargetIds.length === 0) return
  setIsBulkApplying(true)
  try {
    const selectedPresets = presets.filter(p => bulkTargetIds.includes(p.id))
    await Promise.all(selectedPresets.map(async preset => {
      let newParamValues = { ...preset.param_values }
      let newAppliedAssets = { ...preset.applied_assets }
      let newBatchSize = preset.batch_size
      
      // ... 既存の promptOps, paramPatch, batchSize, loraOps ...
      
      await workflowPresetsApi.update(preset.id, {
        param_values: newParamValues,
        applied_assets: newAppliedAssets,
        batch_size: newBatchSize,
      })
    }))
  } finally {
    setIsBulkApplying(false)
  }
}
