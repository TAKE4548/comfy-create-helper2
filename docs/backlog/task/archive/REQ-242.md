### 背景と重要課題
現状、Service Manager からサービス群を起動した際、一瞬のフラッシュにとどまらず **3 つのターミナルウィンドウ（Backend, Frontend, ComfyUI）が前面に立ち上がって残り続ける**。
原因は `server-manager.ps1` 内で `conhost.exe ... -WindowStyle Hidden` を使用していることにあり、Windows 上で `conhost.exe` はウィンドウ生成を抑制できず `MainWindowHandle` が作成され、内部の常駐プロセスによって閉じずに残り続ける。
本タスクでは `conhost.exe` を全廃し、.NET の `ProcessStartInfo`（`CreateNoWindow = $true`, `UseShellExecute = $false`）を用いた `Start-ProcessNoWindow` 関数により、ウィンドウ生成数 0（`MainWindowHandle == 0`）での完全バックグラウンド常駐を実現する。

# [新規関数] 完全非表示プロセス起動ヘルパー
function Start-ProcessNoWindow ([string]$filePath, [string]$arguments, [string]$workingDirectory = "") {
    $psi = New-Object System.Diagnostics.ProcessStartInfo
    $psi.FileName = $filePath
    $psi.Arguments = $arguments
    if ($workingDirectory) {
        $psi.WorkingDirectory = $workingDirectory
    }
    $psi.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Hidden
    $psi.CreateNoWindow = $true
    $psi.UseShellExecute = $false
    return [System.Diagnostics.Process]::Start($psi)
}

# [改修] Start-ComfyUIProcess: バックグラウンド起動時の conhost 廃止および comfy.log へのリダイレクト
function Start-ComfyUIProcess ([bool]$asBackground) {
    if (Test-PortOpen $cfg.wsl.port) {
        Write-Host "ComfyUI is already running on port $($cfg.wsl.port)." -ForegroundColor Yellow
        return
    }

    if (-not (Test-WslAvailability)) {
        return
    }

    $wslPrefix = Get-WslDistroPrefix
    $cmdStr = "cd $($cfg.wsl.project_dir) && if [ -f '$($cfg.wsl.venv_activate)' ]; then source $($cfg.wsl.venv_activate); fi && cd $($cfg.wsl.comfy_dir) && export PYTORCH_HIP_ALLOC_CONF='$($cfg.wsl.env_vars.PYTORCH_HIP_ALLOC_CONF)' && $($cfg.wsl.start_command)"

    if ($asBackground) {
        $logFile = Join-Path $LOGS_DIR "comfy.log"
        Start-ProcessNoWindow "cmd.exe" "/c wsl $($wslPrefix)bash -c `"$cmdStr`" > `"$logFile`" 2>&1"
    } else {
        Start-Process "cmd.exe" -ArgumentList "/k title ComfyUI (WSL2) && wsl $($wslPrefix)bash -c `"$cmdStr`""
    }
}

# [改修] Start-BackendProcess: バックグラウンド起動時の conhost 廃止
function Start-BackendProcess ([bool]$asBackground) {
    if (Test-PortOpen $cfg.backend.port) {
        Write-Host "Backend is already running on port $($cfg.backend.port)." -ForegroundColor Yellow
        return
    }
    $beDir = (Resolve-Path (Join-Path $SCRIPT_DIR $cfg.backend.relative_dir)).Path

    if ($asBackground) {
        $logFile = Join-Path $LOGS_DIR "backend.log"
        Start-ProcessNoWindow "cmd.exe" "/c run_server.bat > `"$logFile`" 2>&1" $beDir
    } else {
        Start-Process -FilePath "cmd.exe" -ArgumentList "/k title PromptSurgeon Backend && run_server.bat" -WorkingDirectory $beDir
    }
}

# [改修] Start-FrontendProcess: バックグラウンド起動時の conhost 廃止
function Start-FrontendProcess ([bool]$asBackground) {
    if (Test-PortOpen $cfg.frontend.port) {
        Write-Host "Frontend is already running on port $($cfg.frontend.port)." -ForegroundColor Yellow
        return
    }
    $feDir = (Resolve-Path (Join-Path $SCRIPT_DIR $cfg.frontend.relative_dir)).Path

    # 起動前に .vite ディレクトリの ReadOnly 属性ビットを解除し、空洞化 deps を先行クリーンアップ
    $viteDir = Join-Path $feDir "node_modules\.vite"
    if (Test-Path $viteDir) {
        try {
            Get-ChildItem -Path $viteDir -Recurse -Force -ErrorAction SilentlyContinue | ForEach-Object {
                if ($_.Attributes -band [System.IO.FileAttributes]::ReadOnly) {
                    $_.Attributes = $_.Attributes -band (-bnot [System.IO.FileAttributes]::ReadOnly)
                }
            }
            $depsDir = Join-Path $viteDir "deps"
            $metaFile = Join-Path $depsDir "_metadata.json"
            if ((Test-Path $depsDir) -and (-not (Test-Path $metaFile))) {
                Remove-Item -Path $depsDir -Recurse -Force -ErrorAction SilentlyContinue
            }
        } catch {}
    }

    if ($asBackground) {
        $logFile = Join-Path $LOGS_DIR "frontend.log"
        Start-ProcessNoWindow "cmd.exe" "/c $($cfg.frontend.start_cmd) > `"$logFile`" 2>&1" $feDir
    } else {
        Start-Process -FilePath "cmd.exe" -ArgumentList "/k title PromptSurgeon Frontend && $($cfg.frontend.start_cmd)" -WorkingDirectory $feDir
    }
}

# [改修] Show-Logs: comfy.log への優先フォールバック
function Show-Logs ([string]$name) {
    switch ($name.ToLower()) {
        "comfy" {
            $cLog = Join-Path $LOGS_DIR "comfy.log"
            if (Test-Path $cLog) {
                Get-Content $cLog -Tail 50
            } else {
                $wslPrefix = Get-WslDistroPrefix
                Invoke-Expression "wsl $($wslPrefix)tail -n 50 ~/comfyui.log"
            }
        }
        "backend" {
            $logFile = Join-Path $LOGS_DIR "backend.log"
            if (Test-Path $logFile) { Get-Content $logFile -Tail 50 } else { Write-Host "No backend.log found" }
        }
        "frontend" {
            $logFile = Join-Path $LOGS_DIR "frontend.log"
            if (Test-Path $logFile) { Get-Content $logFile -Tail 50 } else { Write-Host "No frontend.log found" }
        }
        default {
            Write-Host "=== ComfyUI Log ===" -ForegroundColor Cyan
            $cLog = Join-Path $LOGS_DIR "comfy.log"
            if (Test-Path $cLog) { Get-Content $cLog -Tail 20 }
            Write-Host "=== Backend Log ===" -ForegroundColor Cyan
            $bLog = Join-Path $LOGS_DIR "backend.log"
            if (Test-Path $bLog) { Get-Content $bLog -Tail 20 }
            Write-Host "=== Frontend Log ===" -ForegroundColor Cyan
            $fLog = Join-Path $LOGS_DIR "frontend.log"
            if (Test-Path $fLog) { Get-Content $fLog -Tail 20 }
        }
    }
}

def read_log_file(filename: str, tail_lines: int = 200) -> tuple[bool, str, int]:
    target = LOGS_DIR / filename
    if not target.exists():
        return False, "ログファイルはまだ生成されていません (サービス未起動)", 0
    try:
        # UTF-8 で試行、ダメなら CP932 でフォールバック
        try:
            with open(target, "r", encoding="utf-8", errors="replace") as f:
                lines = f.readlines()
        except UnicodeDecodeError:
            with open(target, "r", encoding="cp932", errors="replace") as f:
                lines = f.readlines()
        
        tail = lines[-tail_lines:] if len(lines) > tail_lines else lines
        return True, "".join(tail), len(lines)
    except Exception as e:
        return False, f"ログ読み込みエラー: {e}", 0

# do_GET 内のログルーティング
        elif self.path.startswith("/api/logs"):
            parsed = urllib.parse.urlparse(self.path)
            params = urllib.parse.parse_qs(parsed.query)
            service = params.get("service", ["backend"])[0].lower()
            tail_str = params.get("tail", ["200"])[0]
            try:
                tail = max(10, min(1000, int(tail_str)))
            except ValueError:
                tail = 200

            service_file_map = {
                "backend": "backend.log",
                "be": "backend.log",
                "frontend": "frontend.log",
                "fe": "frontend.log",
                "comfy": "comfy.log",

            self._send_json({
                "service": service,
                "filename": filename,
                "exists": exists,
                "lines": total_lines,
                "content": content

// JavaScript タブおよびログ更新ロジック
let activeService = 'backend';
let actionLogContent = '待機中...';

const logOutput = document.getElementById('log-output');
const autoRefreshLog = document.getElementById('auto-refresh-log');
const autoScrollLog = document.getElementById('auto-scroll-log');
const tailSelect = document.getElementById('tail-lines-select');
const logStatus = document.getElementById('log-status');
const logUpdated = document.getElementById('log-updated');

function selectTab(service) {
  activeService = service;
  document.querySelectorAll('.tab-btn').forEach(btn => {
    btn.classList.toggle('active', btn.getAttribute('data-service') === service);
  });
  fetchLogs();
}

async function fetchLogs() {
  if (activeService === 'manager') {
    logOutput.textContent = actionLogContent;
    logStatus.textContent = "Manager 実行ログ表示中";
    logUpdated.textContent = "最終更新: " + new Date().toLocaleTimeString();
    if (autoScrollLog.checked) logOutput.scrollTop = logOutput.scrollHeight;
    return;
  }

  const tail = tailSelect.value;
  try {
    const res = await fetch(`/api/logs?service=${activeService}&tail=${tail}`);
    const data = await res.json();
    if (data.error) {
      logStatus.textContent = data.error;
      return;
    }
    const wasAtBottom = logOutput.scrollHeight - logOutput.scrollTop - logOutput.clientHeight < 40;
    logOutput.textContent = data.content || "(出力なし)";
    logStatus.textContent = `${data.filename} (${data.lines}行中 末尾表示)`;
    logUpdated.textContent = "最終更新: " + new Date().toLocaleTimeString();
    if (autoScrollLog.checked && wasAtBottom) {
      logOutput.scrollTop = logOutput.scrollHeight;
    }
  } catch (err) {
    logStatus.textContent = "ログ取得エラー: " + err;
  }
}

// 2秒間隔でアクティブタブのログを自動更新
setInterval(() => {
  if (autoRefreshLog && autoRefreshLog.checked && activeService !== 'manager') {
    fetchLogs();
  }
}, 2000);

## 3. 実装要件 (Implementation Logic)

### 3.1 `tools/server-manager/server-manager.ps1`
1. `Start-ProcessNoWindow` を定義し、`System.Diagnostics.ProcessStartInfo` により `CreateNoWindow = $true`、`UseShellExecute = $false`、`WindowStyle = Hidden` で外部ウィンドウ生成を抑止する。
2. `Start-ComfyUIProcess`、`Start-BackendProcess`、`Start-FrontendProcess` のバックグラウンド実行分岐をすべて `Start-ProcessNoWindow` に置換し、`conhost.exe` を全廃する。
3. `Start-ComfyUIProcess` において、`$logFile = Join-Path $LOGS_DIR "comfy.log"` を出力先として `> "$logFile" 2>&1` を付与し、ComfyUI の実行ログを確実にキャプチャする。
4. `Show-Logs` の `comfy` 分岐において、`$LOGS_DIR\comfy.log` が存在する場合は優先してコンソール表示する。

### 3.2 `tools/server-manager/server_api.py`
1. Windows 環境下で `CREATE_NO_WINDOW = 0x08000000` を定義し、すべての `subprocess.run` 呼び出しに `creationflags=CREATE_NO_WINDOW` を指定する。
2. `read_log_file` 関数を実装し、指定ファイル（ホワイトリスト制限）の末尾 N 行を UTF-8（CP932 フォールバック）で安全に読み出す。
3. `do_GET` に `/api/logs` ルーティングを追加し、`service` および `tail` パラメータに応じて JSON レスポンスを返却する。
4. `HTML_TEMPLATE` 内の実行ログエリアを「タブ別コンソール出力コンポーネント」に刷新する。
   - [Win BE] [React FE] [WSL ComfyUI] [Manager 操作] の 4 タブ。
   - 2 秒ごとの自動更新、最新追従（自動スクロール）、行数選択、今すぐ更新。
   - アクション実行時のログ（`data.output`）は `actionLogContent` に格納し、Manager 操作タブで閲覧可能にする。
