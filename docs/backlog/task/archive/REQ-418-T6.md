const handleCreateGroup = async (e: React.FormEvent) => {

const handleSlaveToggle = async (slaveRole: string) => {
  if (!activeGroup) return;
  const newSlaves = activeGroup.slave_roles.includes(slaveRole)
    ? activeGroup.slave_roles.filter((r) => r !== slaveRole) : [...activeGroup.slave_roles, slaveRole];
  await updateParamSyncGroup(activeGroup.id, { slave_roles: newSlaves });

const handleMasterChange = async (newMaster: string) => {
  if (!activeGroup) return;

### 4. Implementation logic

- [ ] **Types (`frontend/src/api/types.ts`)**:
  - Add `export interface VariantRef { id: number; role: string; display_name?: string | null }`.
  - On `SnapshotSummary`: add `params?: Record<string, unknown> | null`, `variant_params?: Record<string, unknown>`, `variants?: VariantRef[]`.
  - On `Snapshot`: add `variant_params?: Record<string, unknown>`, `variants?: VariantRef[]`.
  - On `WorkflowPreset`: add `variant_param_values?: Record<string, string>`, `variants?: VariantRef[]`.
  - On `ParamSyncGroup`: keep `master_role: string` and `slave_roles: string[]`; add `master_variant_id?: number`, `slave_variant_ids?: number[]`, `variants?: VariantRef[]`.
  - On `ParamSyncGroupCreate`: make `master_role?: string` and `slave_roles?: string[]` optional; add `master_variant_id?: number` and `slave_variant_ids?: number[]`.
  - On `ParamSyncGroupUpdate`: add `master_variant_id?: number` and `slave_variant_ids?: number[]`.
- [ ] **`ParamSyncModal.tsx` payload construction**:
  - Build `roleToVariantId = new Map<string, number>()` from `activeWorkflow?.mappings ?? []` (where `m.role && typeof m.variant_id === 'number'`, plus `activeGroup?.variants`).
  - Helper `resolveSlaveIds(roles: string[]): number[] | null`: if every role in `roles` has a `number` entry in `roleToVariantId`, return `roles.map((r) => roleToVariantId.get(r)!)` (for `roles.length === 0`, return `[]` when `roleToVariantId.size > 0`, else `null`); otherwise return `null`.
  - `handleCreateGroup`: if `masterVid = roleToVariantId.get(newMasterRole)` is defined, send `{ name: newGroupName.trim(), master_variant_id: masterVid, slave_variant_ids: [], master_role: newMasterRole, slave_roles: [], is_enabled: true }`; else send `{ name: newGroupName.trim(), master_role: newMasterRole, slave_roles: [], is_enabled: true }`.
  - `handleSlaveToggle`: compute `newSlaves` and `slaveIds = resolveSlaveIds(newSlaves)`. If `slaveIds !== null`, pass `{ slave_variant_ids: slaveIds, slave_roles: newSlaves }`; else pass `{ slave_roles: newSlaves }`.
  - `handleMasterChange`: compute `validSlaves`, `masterVid = roleToVariantId.get(newMaster)`, and `slaveIds = resolveSlaveIds(validSlaves)`. If `masterVid !== undefined && slaveIds !== null`, pass `{ master_variant_id: masterVid, slave_variant_ids: slaveIds, master_role: newMaster, slave_roles: validSlaves }`; else pass `{ master_role: newMaster, slave_roles: validSlaves }`.
  - Do NOT alter any JSX structure, CSS class names, DOM attributes, or candidate filtering logic (D7 UI guardrail).
