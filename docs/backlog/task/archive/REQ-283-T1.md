export function parseTagWeight(tag: string): { base: string; weight: number | null; isParenthesized: boolean } {
  const trimmed = tag.trim();
  const weightedMatch = trimmed.match(/^\((.+?):([0-9.]+)\)$/);
  if (weightedMatch) {
    return { base: weightedMatch[1].trim(), weight: parseFloat(weightedMatch[2]), isParenthesized: true };
  }
  const parenMatch = trimmed.match(/^\((.+?)\)$/);
  if (parenMatch) {
    return { base: parenMatch[1].trim(), weight: null, isParenthesized: true };
  }
  return { base: trimmed, weight: null, isParenthesized: false };
}

export function formatTagWithWeight(base: string, weight: number | null): string {
  if (weight === null || isNaN(weight) || weight === 1.0) return base;
  return `(${base}:${weight})`;
}
