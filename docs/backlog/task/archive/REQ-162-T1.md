### REQ-162-T1: ComfyUI プロキシルーター拡張とノードスキーマキャッシュの実装

#### 目的
ComfyUIの全ノードスキーマ（`/object_info`）をバックエンド側で安全にインメモリキャッシュし、モバイル端末等の回線遅延による3秒タイムアウトを根絶する。また、ジョブ投入（`/prompt`）、キュー取得・削除（`/queue`）、中断（`/interrupt`）、履歴（`/history`）のプロキシエンドポイントを新設する。

#### 対象ファイル
1. `backend/app/routers/comfy.py`
2. `backend/tests/test_comfy_proxy.py`

#### 依存関係・設定仕様 (Dependencies & Config Context)
- **FastAPI / HTTPX**: `from fastapi import APIRouter, Response, HTTPException, status`
- **Settings**: `from app.config import settings` (`settings.comfy_url: str = "http://localhost:8188"`)
- **Pydantic**: `from pydantic import BaseModel`

#### バックエンド実装詳細 (Backend Complete Source Code)

```python
# backend/app/routers/comfy.py
import logging
import httpx
from fastapi import APIRouter, Response, HTTPException, status
from pydantic import BaseModel
from typing import Any, Dict, List, Optional
from app.config import settings

logger = logging.getLogger(__name__)
router = APIRouter(prefix="/api/comfy", tags=["comfy"])

# インメモリキャッシュ
_object_info_cache: Optional[Dict[str, Any]] = None

@router.get("/object-info/{class_type}", response_model=dict)
async def get_comfy_object_info(class_type: str):
    """
    ComfyUI の /object_info/{class_type} をプロキシして返す。
    ComfyUI未起動時は {} を返す（エラーにしない）。
    """
    async with httpx.AsyncClient() as client:
        try:
            response = await client.get(
                f"{settings.comfy_url}/object_info/{class_type}",
                timeout=5.0
            )
            response.raise_for_status()
            return response.json()
        except (httpx.HTTPError, httpx.RequestError) as e:
            logger.warning(f"Failed to fetch object_info from ComfyUI for '{class_type}': {e}")
            return {}

@router.get("/object_info", response_model=dict)
async def get_comfy_object_info_all():
    """ComfyUI の全ノード /object_info をプロキシしインメモリキャッシュして返す。"""
    global _object_info_cache
    if _object_info_cache is not None:
        return _object_info_cache
    async with httpx.AsyncClient() as client:
        try:
            response = await client.get(f"{settings.comfy_url}/object_info", timeout=15.0)
            response.raise_for_status()
            _object_info_cache = response.json()
            return _object_info_cache
        except Exception as e:
            logger.warning(f"Failed to fetch all object_info from ComfyUI: {e}")
            return {}

class PromptRequest(BaseModel):
    prompt: Dict[str, Any]
    client_id: Optional[str] = None

@router.post("/prompt")
async def queue_prompt(payload: PromptRequest):
    """ComfyUI の /prompt へジョブを投入する。"""
    async with httpx.AsyncClient() as client:
        try:
            response = await client.post(
                f"{settings.comfy_url}/prompt",
                json=payload.model_dump(exclude_none=True),
                timeout=10.0
            )
            return Response(
                content=response.content,
                status_code=response.status_code,
                media_type="application/json"
            )
        except Exception as e:
            logger.error(f"Failed to proxy prompt to ComfyUI: {e}")
            raise HTTPException(status_code=502, detail=f"ComfyUI prompt error: {e}")

@router.get("/queue")
async def get_queue():
    """ComfyUI の /queue を取得する。"""
    async with httpx.AsyncClient() as client:
        try:
            response = await client.get(f"{settings.comfy_url}/queue", timeout=5.0)
            return Response(
                content=response.content,
                status_code=response.status_code,
                media_type="application/json"
            )
        except Exception as e:
            return {"queue_running": [], "queue_pending": []}

class QueueActionRequest(BaseModel):
    delete: Optional[List[str]] = None
    clear: Optional[bool] = None

@router.post("/queue")
async def manage_queue(payload: QueueActionRequest):
    """ComfyUI の /queue に対する削除・クリアを実行する。"""
    async with httpx.AsyncClient() as client:
        try:
            response = await client.post(
                f"{settings.comfy_url}/queue",
                json=payload.model_dump(exclude_none=True),
                timeout=5.0
            )
            return Response(
                content=response.content,
                status_code=response.status_code,
                media_type="application/json"
            )
        except Exception as e:
            logger.error(f"Failed to proxy queue delete to ComfyUI: {e}")
            return {"status": "error", "detail": str(e)}

@router.post("/interrupt")
async def interrupt_execution():
    """ComfyUI の /interrupt を呼び出して実行を中断する。"""
    async with httpx.AsyncClient() as client:
        try:
            response = await client.post(f"{settings.comfy_url}/interrupt", timeout=5.0)
            return Response(
                content=response.content,
                status_code=response.status_code,
                media_type="application/json"
            )
        except Exception as e:
            return {"status": "error", "detail": str(e)}

@router.get("/history")
async def get_history():
    """ComfyUI の /history を取得する。"""
    async with httpx.AsyncClient() as client:
        try:
            response = await client.get(f"{settings.comfy_url}/history", timeout=5.0)
            return Response(
                content=response.content,
                status_code=response.status_code,
                media_type="application/json"
            )
        except Exception as e:
            return {}

class HistoryActionRequest(BaseModel):
    delete: Optional[List[str]] = None
    clear: Optional[bool] = None

@router.post("/history")
async def manage_history(payload: HistoryActionRequest):
    """ComfyUI の /history 削除を実行する。"""
    async with httpx.AsyncClient() as client:
        try:
            response = await client.post(
                f"{settings.comfy_url}/history",
                json=payload.model_dump(exclude_none=True),
                timeout=5.0
            )
            return Response(
                content=response.content,
                status_code=response.status_code,
                media_type="application/json"
            )
        except Exception as e:
            return {"status": "error", "detail": str(e)}
```

```python
import unittest
from fastapi.testclient import TestClient
from unittest.mock import patch, MagicMock
import httpx
from app.main import app
import app.routers.comfy as comfy_router

class TestComfyProxyAPI(unittest.TestCase):
    def setUp(self):
        self.client = TestClient(app)
        comfy_router._object_info_cache = None

    @patch("httpx.AsyncClient.get")
    def test_get_object_info_cached(self, mock_get):
        mock_response = MagicMock()
        mock_response.status_code = 200
        mock_response.json.return_value = {"KSampler": {"input": {}}}
        mock_response.raise_for_status = MagicMock()
        mock_get.return_value = mock_response

        # 1回目 (fetch)
        res1 = self.client.get("/api/comfy/object_info")
        self.assertEqual(res1.status_code, 200)
        self.assertEqual(res1.json(), {"KSampler": {"input": {}}})
        self.assertEqual(mock_get.call_count, 1)

        # 2回目 (キャッシュヒット)
        res2 = self.client.get("/api/comfy/object_info")
        self.assertEqual(res2.status_code, 200)
        self.assertEqual(res2.json(), {"KSampler": {"input": {}}})
        self.assertEqual(mock_get.call_count, 1)

    @patch("httpx.AsyncClient.post")
    def test_post_prompt_proxy_success(self, mock_post):
        mock_response = MagicMock()
        mock_response.status_code = 200
        mock_response.content = b'{"prompt_id": "p-123", "number": 1}'
        mock_post.return_value = mock_response

        res = self.client.post("/api/comfy/prompt", json={"prompt": {"3": {}}, "client_id": "c-1"})
        self.assertEqual(res.status_code, 200)
        self.assertEqual(res.json(), {"prompt_id": "p-123", "number": 1})

    @patch("httpx.AsyncClient.get")
    def test_get_queue_proxy(self, mock_get):
        mock_response = MagicMock()
        mock_response.status_code = 200
        mock_response.content = b'{"queue_running": [], "queue_pending": []}'
        mock_get.return_value = mock_response

        res = self.client.get("/api/comfy/queue")
        self.assertEqual(res.status_code, 200)
        self.assertEqual(res.json(), {"queue_running": [], "queue_pending": []})

    @patch("httpx.AsyncClient.post")
    def test_post_queue_clear_proxy(self, mock_post):
        mock_response = MagicMock()

        mock_response.content = b'{"status": "ok"}'
        mock_post.return_value = mock_response

        res = self.client.post("/api/comfy/queue", json={"clear": True})
        self.assertEqual(res.status_code, 200)

    @patch("httpx.AsyncClient.post")
    def test_post_interrupt_proxy(self, mock_post):

        mock_response.content = b'{"status": "interrupted"}'
        mock_post.return_value = mock_response

        res = self.client.post("/api/comfy/interrupt")
        self.assertEqual(res.status_code, 200)
```
