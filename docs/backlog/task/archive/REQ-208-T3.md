### 3.1 `useGenerateDisplayState.ts` 実装仕様
```typescript
import { useState, useMemo, useEffect } from 'react'
import { useQuery } from '@tanstack/react-query'
import { snapshotsApi } from '../../../api/client'
import { useGenerationStore } from '../../../store/useGenerationStore'
import { useImageNavigation } from './useImageNavigation'
import { useMultiSelect } from '../../../hooks/useMultiSelect'
import { promptToSetMapCache } from '../../../utils/promptToSetMapCache'

export interface SnapshotSummaryWithParams {
  id: number
  workflow_id: number
  image_path: string | null
  thumbnail_path?: string | null
  created_at: string
  params?: Record<string, any>
}

export interface UseGenerateDisplayStateProps {
  selectedWorkflowId: number | null
}

export function useGenerateDisplayState({
  selectedWorkflowId,
}: UseGenerateDisplayStateProps) {
  const store = useGenerationStore()
  const {
    clientId,
    batchImages,
    selectedImg,
    setSelectedImg,
    comfyHistoryImages,
    filterOwnSession,
    setFilterOwnSession,
    activeClientId,
    isGenerating,
    activeBatch,
    queueRunning,
    queuePending,
    previewImage,
  } = store

  const [sortOrder, setSortOrder] = useState<'desc' | 'asc'>('desc')
  const [isModalPreviewOpen, setIsModalPreviewOpen] = useState(false)
  const [modalPreviewIndex, setModalPreviewIndex] = useState(0)

  // ── Snapshots Query ──
  const { data: snapshots } = useQuery({
    queryKey: ['snapshots', selectedWorkflowId],
    queryFn: () => snapshotsApi.list(selectedWorkflowId!) as Promise<SnapshotSummaryWithParams[]>,
    enabled: !!selectedWorkflowId,
  })

  const displayImages = filterOwnSession ? batchImages : comfyHistoryImages

  const sortedDisplayImages = useMemo(() => {
    return sortOrder === 'desc' ? displayImages : [...displayImages].reverse()
  }, [displayImages, sortOrder])

  const getSetNameForImage = (imgUrl: string) => {
    return promptToSetMapCache.getSetNameForUrl(imgUrl)
  }

  const sortedSnapshots = useMemo(() => {
    if (!snapshots) return []
    const copy = [...snapshots] as SnapshotSummaryWithParams[]
    return sortOrder === 'desc'
      ? copy.sort((a, b) => new Date(b.created_at).getTime() - new Date(a.created_at).getTime())
      : copy.sort((a, b) => new Date(a.created_at).getTime() - new Date(b.created_at).getTime())
  }, [snapshots, sortOrder])

  const { handlePrevImg, handleNextImg } = useImageNavigation(sortedDisplayImages)

  const historyItems = useMemo(() => {
    return sortedDisplayImages.map((url) => ({ id: url, url }))
  }, [sortedDisplayImages])

  const {
    isMultiSelectMode,
    selectedIds,
    handleItemClick,
    getLongPressHandlers,
    isSelected,
    exitMultiSelectMode,
    selectAll,
    clearSelection,
  } = useMultiSelect<string>({ items: historyItems })

  const snapshotIdsForDisplay = useMemo(() => {
    return sortedDisplayImages.map((img) => {
      const found = sortedSnapshots?.find((s) => s.image_path === img)
      return found ? found.id : null
    })
  }, [sortedDisplayImages, sortedSnapshots])

  const showStrip =
    batchImages.length > 0 || comfyHistoryImages.length > 0 || (!!sortedSnapshots && sortedSnapshots.length > 0)
  const hasHistory = !!selectedWorkflowId && !!sortedSnapshots && sortedSnapshots.length > 0

  const isGeneratingHere =
    isGenerating &&
    (activeClientId === clientId ||
      (activeBatch !== null && activeBatch.some((j) => j.status === 'generating' || j.status === 'pending')))

  const isOtherSessionGenerating = isGenerating && !isGeneratingHere

  const allQueueCount = useMemo(() => {
    return queueRunning.length + queuePending.length
  }, [queueRunning, queuePending])

  const totalQueueCount = useMemo(() => {
    if (!filterOwnSession) return allQueueCount
    const filterJob = (job: any) => job[3]?.client_id === clientId
    return queueRunning.filter(filterJob).length + queuePending.filter(filterJob).length
  }, [queueRunning, queuePending, filterOwnSession, clientId, allQueueCount])

  // MultiSelect selection tracking
  useEffect(() => {
    if (isMultiSelectMode && selectedIds.size > 0) {
      const idsArray = Array.from(selectedIds)
      const lastUrl = idsArray[idsArray.length - 1]
      if (lastUrl && sortedDisplayImages.includes(lastUrl)) {
        if (lastUrl !== selectedImg) {
          setSelectedImg(lastUrl)
        }
      }
    }
  }, [selectedIds, isMultiSelectMode, sortedDisplayImages, selectedImg, setSelectedImg])

  // Visibility / focus refresh
  useEffect(() => {
    const handleVisibility = () => {
      if (document.visibilityState === 'visible') {
        store.fetchQueue()
        store.fetchHistory()
      }
    }
    document.addEventListener('visibilitychange', handleVisibility)
    window.addEventListener('focus', handleVisibility)
    return () => {
      document.removeEventListener('visibilitychange', handleVisibility)
      window.removeEventListener('focus', handleVisibility)
    }
  }, [store])

  const handleOpenModalPreview = () => {
    const currentImg = previewImage || selectedImg
    if (!currentImg) return
    const idx = sortedDisplayImages.indexOf(currentImg)
    setModalPreviewIndex(idx >= 0 ? idx : 0)
    setIsModalPreviewOpen(true)
  }

  return {
    snapshots,
    sortOrder,
    setSortOrder,
    displayImages,
    sortedDisplayImages,
    getSetNameForImage,
    sortedSnapshots,
    handlePrevImg,
    handleNextImg,
    isMultiSelectMode,
    selectedIds,
    handleItemClick,
    getLongPressHandlers,
    isSelected,
    exitMultiSelectMode,
    selectAll,
    clearSelection,
    snapshotIdsForDisplay,
    showStrip,
    hasHistory,
    isGeneratingHere,
    isOtherSessionGenerating,
    allQueueCount,
    totalQueueCount,
    isModalPreviewOpen,
    setIsModalPreviewOpen,
    modalPreviewIndex,
    setModalPreviewIndex,
    handleOpenModalPreview,
    filterOwnSession,
    setFilterOwnSession,
  }
}
```

### 3.2 `useGeneratePageModals.ts` 実装仕様
```typescript
import { useState, useEffect } from 'react'
import { useGenerationStore } from '../../../store/useGenerationStore'

export function useGeneratePageModals() {
  const { comfyUrl } = useGenerationStore()
  const [isMobile, setIsMobile] = useState(() =>
    window.location.pathname.startsWith('/mobile') || window.innerWidth < 768
  )

  useEffect(() => {
    const handleResize = () =>
      setIsMobile(window.location.pathname.startsWith('/mobile') || window.innerWidth < 768)
    window.addEventListener('resize', handleResize)
    return () => window.removeEventListener('resize', handleResize)
  }, [])

  const [rescanOpen, setRescanOpen] = useState(false)
  const [isSettingsOpen, setIsSettingsOpen] = useState(false)
  const [localComfyUrl, setLocalComfyUrl] = useState('')
  const [thumbnailTab, setThumbnailTab] = useState<'history' | 'snapshots'>('history')
  const [isJobModalOpen, setIsJobModalOpen] = useState(false)
  const [isSceneGeneratorOpen, setIsSceneGeneratorOpen] = useState(false)

  useEffect(() => {
    if (comfyUrl) setLocalComfyUrl(comfyUrl)
  }, [comfyUrl])

  return {
    isMobile,
    rescanOpen,
    setRescanOpen,
    isSettingsOpen,
    setIsSettingsOpen,
    localComfyUrl,
    setLocalComfyUrl,
    thumbnailTab,
    setThumbnailTab,
    isJobModalOpen,
    setIsJobModalOpen,
    isSceneGeneratorOpen,
    setIsSceneGeneratorOpen,
  }
}
```

```typescript
import { useEffect } from 'react'
import { useGenerationStore } from '../../../store/useGenerationStore'
import { useWorkflowParams } from './useWorkflowParams'
import { useGenerateAction } from './useGenerateAction'
import { useSnapshotActions } from './useSnapshotActions'
import { useAuditAction } from './useAuditAction'
import { useWorkflowUpload } from './useWorkflowUpload'
import { usePromptTranslation } from './usePromptTranslation'
import { useWorkflowManagement } from './useWorkflowManagement'
import { usePresetState } from './usePresetState'
import { useAssetSelectionState } from './useAssetSelectionState'
import { useGenerateDisplayState } from './useGenerateDisplayState'
import { useGeneratePageModals } from './useGeneratePageModals'

export type { SnapshotSummaryWithParams } from './useGenerateDisplayState'

export function useGeneratePageState() {
  const store = useGenerationStore()
  const {
    clientId,
    selectedWorkflowId,
    setSelectedWorkflowId,
    paramValues,
    setParamValues,
    batchSize,
    setBatchSize,
    seedMode,
    setSeedMode,
    seed,
    setSeed,
    batchImages,
    selectedImg,
    setSelectedImg,
    comfyHistoryImages,
    appliedAssets,
    addAppliedAsset,
    removeAppliedAsset,
    toggleAppliedAsset,
    setAppliedAssets,
    comfyUrl,
    setComfyUrl,
    sets,
    updateActiveSet,
    isConnected,
    isGenerating,
    progress,
    previewImage,
    activeClientId,
    activeBatch,
    activeSetId,
    addSet,
    removeSet,
    renameSet,
    toggleSetChecked,
    setAllSetsChecked,
    setActiveSetId,
    openPresetIds,
    openPreset,
    closePreset,
    fetchParamSyncGroups,
  } = store

  useEffect(() => {
    if (selectedWorkflowId) {
      fetchParamSyncGroups(selectedWorkflowId)
    }
  }, [selectedWorkflowId, fetchParamSyncGroups])

  // 1. Workflow Management
  const workflowMgmt = useWorkflowManagement({
    selectedWorkflowId,
    setSelectedWorkflowId,
  })

  // 2. Workflow Params & Actions
  const { loadSnapshotParams, handleSaveDefaultParams, handleResetDefaultParams, saveDefaultParamsMutation } =
    useWorkflowParams(workflowMgmt.workflow, selectedWorkflowId)

  const { handleGenerateSets } = useGenerateAction(workflowMgmt.workflow, workflowMgmt.comfyObjectInfo)

  const promptTranslation = usePromptTranslation({ paramValues, setParamValues })

  const snapshotActions = useSnapshotActions(selectedWorkflowId)
  const auditAction = useAuditAction()
  const uploadAction = useWorkflowUpload()

  // 3. Preset State & Operations
  const presetState = usePresetState({
    selectedWorkflowId,
    workflow: workflowMgmt.workflow,
  })

  // 4. Asset Selection & Snapshot Rebuild
  const assetSelection = useAssetSelectionState({
    setRoleCollapsed: workflowMgmt.setRoleCollapsed,
    loadSnapshotParams,
  })

  // 5. Display & Gallery State
  const displayState = useGenerateDisplayState({
    selectedWorkflowId,
  })

  // 6. Modals & Responsive State
  const modals = useGeneratePageModals()

  return {
    ...modals,
    ...workflowMgmt,
    ...presetState,
    ...assetSelection,
    ...displayState,
    ...snapshotActions,
    ...auditAction,
    ...uploadAction,

    // Store properties
    store,
    clientId,
    selectedWorkflowId,
    setSelectedWorkflowId,
    paramValues,
    setParamValues,
    batchSize,
    setBatchSize,
    seedMode,
    setSeedMode,
    seed,
    setSeed,
    batchImages,
    selectedImg,
    setSelectedImg,
    comfyHistoryImages,
    appliedAssets,
    addAppliedAsset,
    removeAppliedAsset,
    toggleAppliedAsset,
    setAppliedAssets,
    comfyUrl,
    setComfyUrl,
    sets,
    updateActiveSet,
    isConnected,
    isGenerating,
    progress,
    previewImage,
    activeClientId,
    activeBatch,
    activeSetId,
    addSet,
    removeSet,
    renameSet,
    toggleSetChecked,
    setAllSetsChecked,
    setActiveSetId,
    openPresetIds,
    openPreset,
    closePreset,

    // Action handlers
    promptTranslation,
    saveDefaultParamsMutation,
    handleSaveDefaultParams,
    handleResetDefaultParams,
    handleGenerateSets,
  }
}
```
