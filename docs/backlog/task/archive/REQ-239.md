function Test-PortOpen ([int]$port) {
    $conn = Get-NetTCPConnection -LocalPort $port -ErrorAction SilentlyContinue | Where-Object State -eq "Listen"
    return ($null -ne $conn)
}

function Get-PidsFromPort ([int]$port) {
    $conns = Get-NetTCPConnection -LocalPort $port -ErrorAction SilentlyContinue | Where-Object State -eq "Listen"
    if ($conns) {
        $pids = @($conns | Select-Object -ExpandProperty OwningProcess -Unique)
        return ,@($pids)
    }
    return ,@()
}
```

function Get-StatusObject {
    $fePids = Get-PidsFromPort $cfg.frontend.port
    $bePids = Get-PidsFromPort $cfg.backend.port
    $comfyPids = Get-PidsFromPort $cfg.wsl.port

    $feRunning = ($fePids.Count -gt 0)
    $beRunning = ($bePids.Count -gt 0)
    $comfyRunning = ($comfyPids.Count -gt 0)

    return [PSCustomObject]@{
        fe = [PSCustomObject]@{
            name = "React FE"
            port = $cfg.frontend.port
            running = $feRunning
            status = if ($feRunning) { "LISTEN" } else { "STOPPED" }
            pids = $fePids
        }
        be = [PSCustomObject]@{
            name = "Win BE"
            port = $cfg.backend.port
            running = $beRunning
            status = if ($beRunning) { "LISTEN" } else { "STOPPED" }
            pids = $bePids
        }
        comfy = [PSCustomObject]@{
            name = "WSL ComfyUI"
            port = $cfg.wsl.port
            running = $comfyRunning
            status = if ($comfyRunning) { "LISTEN" } else { "STOPPED" }
            pids = $comfyPids
        }
    }
}

function Stop-ComfyUIProcess {
    $wslPrefix = Get-WslDistroPrefix

        Invoke-Expression "wsl $($wslPrefix)bash -c `"pkill -f 'python.*main.py.*8188' || pkill -f 'python.*main.py'`" 2>`$null"

### 2.4 既存の個別再起動コード (`tools/server-manager/server-manager.ps1`: L337-L365)

function Restart-ServiceByName ([string]$name, [bool]$asBackground) {
    switch ($name.ToLower()) {
        "comfy" {
            Write-Host "Restarting WSL2 ComfyUI..." -ForegroundColor Yellow
            Stop-ComfyUIProcess
            Start-Sleep -Seconds 1
            Start-ComfyUIProcess $asBackground
            Wait-For-Service "comfy" "WSL ComfyUI" | Out-Null
        }
        "backend" {
            Write-Host "Restarting Backend..." -ForegroundColor Yellow
            Stop-BackendProcess
            Start-Sleep -Seconds 1
            Start-BackendProcess $asBackground
            Wait-For-Service "backend" "Backend" | Out-Null
        }
        "frontend" {
            Write-Host "Restarting Frontend..." -ForegroundColor Yellow
            Stop-FrontendProcess
            Start-Sleep -Seconds 1
            Start-FrontendProcess $asBackground
            Wait-For-Service "frontend" "Frontend" | Out-Null
        }
        default {
            Write-Host "Unknown service: $name (choose: comfy, backend, frontend)" -ForegroundColor Red
        }
    }
    Show-Status
}
```

### 2.5 WSL ディストリビューション Prefix 取得関数 (`tools/server-manager/server-manager.ps1`: L200-L210)

function Get-WslDistroPrefix {
    [CmdletBinding()]
    [OutputType([string])]
    param()

    # 空文字または未定義の場合は "" を返し、-d を付与せずWSLの既定ディストリビューションを使用する
    if ($cfg.wsl.distro -and ($cfg.wsl.distro.ToString().Trim() -ne "")) {
        return "-d $($cfg.wsl.distro.ToString().Trim()) "
    }
    return ""
}

## 3. 実装要件 (Implementation Logic)
- [x] `Test-PortConnect ([string]$targetHost = "127.0.0.1", [int]$port, [int]$timeoutMs = 500)` を新設する。
  - `System.Net.Sockets.TcpClient` を使用し、指定ポートへの非同期接続を試行する。
  - `$timeoutMs` 内に接続できれば `$true`、タイムアウトまたは接続失敗時は `$false` を返す。
  - `finally` で `$tcp.Dispose()` を確実に呼ぶ。
- [x] `Test-PortOpen ([int]$port)` を改修する。
  - まず既存の `Get-NetTCPConnection -LocalPort $port -ErrorAction SilentlyContinue | Where-Object State -eq "Listen"` を評価し、存在すれば即座に `$true` を返す。
  - 存在しない場合（WSL2 サービスやプロキシ配下）、フォールバックとして `Test-PortConnect -targetHost "127.0.0.1" -port $port -timeoutMs 500` の結果を返す。
- [x] `Wait-For-Port-Closed ([int]$port, [int]$timeoutSec = 10, [string]$serviceName = "")` を新設する。
  - `Test-PortOpen $port` が `$false` になるまで最大 `$timeoutSec` 秒間、500ms 間隔でポーリング待機する。
  - 閉鎖が確認できれば `$true`、タイムアウトした場合は `$false` を返す。
- [x] `Get-WslComfyPids` を新設する。
  - `$wslPrefix = Get-WslDistroPrefix` を取得。
  - `wsl $($wslPrefix)bash -c "pgrep -f 'python.*main.py'"` を安全に呼び出し、数値 PID の配列を返す（エラー時は空配列 `@()`）。
- [x] `Stop-ComfyUIProcess` を改修・堅牢化する。
  - `pkill -f 'python.*main.py.*8188' || pkill -f 'python.*main.py'` を送信。
  - `Wait-For-Port-Closed -port $cfg.wsl.port -timeoutSec 5 -serviceName "WSL ComfyUI"` でポート閉鎖を待機。
  - 閉鎖しなかった場合は `wsl $($wslPrefix)bash -c "pkill -9 -f 'python.*main.py'"` で SIGKILL 送信し、さらに 2 秒待機。
  - ウィンドウ起動時の残骸 `cmd.exe` を回収: `Get-Process cmd -ErrorAction SilentlyContinue | Where-Object { $_.MainWindowTitle -like "*ComfyUI (WSL2)*" } | ForEach-Object { taskkill.exe /PID $_.Id /T /F 2>$null }`
  - `Stop-PortProcesses $cfg.wsl.port "ComfyUI"` を実行。
- [x] `Restart-ServiceByName` を改修する。
  - 固定 `Start-Sleep -Seconds 1` の代わりに、各サービスのポートに対して `Wait-For-Port-Closed` を実行し、ポートが確実に閉じたことを確認してから `Start-*Process` を呼び出す。
- [x] `Get-StatusObject` の ComfyUI 判定を改修する。
  - `$comfyOpen = Test-PortOpen $cfg.wsl.port` を実行。
  - `$comfyPids = Get-PidsFromPort $cfg.wsl.port` を取得。
  - `$comfyPids.Count -eq 0` かつ `$comfyOpen` の場合、`$comfyPids = Get-WslComfyPids` で WSL 側の PID を補完する。
  - `$comfyRunning = $comfyOpen -or ($comfyPids.Count -gt 0)` で稼働状態を判定する。
  - `status = if ($comfyRunning) { "LISTEN" } else { "STOPPED" }` と設定する。
- [x] `docs/specs/server_manager_guide.md` に WSL2 のポート疎通判定ロジック、停止待機・SIGKILL フォールバック、ウィンドウクリーンアップ仕様を追記する。
