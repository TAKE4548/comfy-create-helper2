    value === 'succeeded' || value === 'failed' || value === 'cancelled' ||
    value === 'unconfirmed' || value === 'released'

  if (value === null || value === undefined) return true

export function decodeReleaseOperation(value: unknown): ReleaseOperation {
  if (!hasRequiredKeys(value, ['items', 'epoch', 'revision']) || !Array.isArray(value.items) ||
      !isString(value.epoch) || !isInteger(value.revision) ||
      !value.items.every((item) => hasRequiredKeys(item, ['job_id', 'prompt_id', 'status', 'outcome', 'message']) &&
        isNullableString(item.job_id) && isNullableString(item.prompt_id) && (item.status === null || isJobStatus(item.status)) &&

        isString(item.message))) {
    throw new Error('Invalid release operation response')
  }
  return value as unknown as ReleaseOperation
}

```ts
/** 全scope・全groupの状態未確認ジョブ。sessionId を渡すとそのsessionのみ。古い順。 */
export function selectUnconfirmedJobs(snapshot: GenerationSnapshot, sessionId: string | null = null): JobView[] {
  return snapshot.jobs.filter((job) => job.status === 'unconfirmed' && (sessionId === null || job.session_id === sessionId))
}

/** 解除済み記録。released_at の新しい順で limit 件（既定10）。 */
export function selectReleasedJobs(snapshot: GenerationSnapshot, sessionId: string | null = null, limit = 10): JobView[] {
  const at = (job: JobView) => Date.parse(job.release?.released_at ?? '') || 0
  return snapshot.jobs
    .filter((job) => job.status === 'released' && (sessionId === null || job.session_id === sessionId))
    .sort((a, b) => at(b) - at(a))
    .slice(0, limit)
}
```
