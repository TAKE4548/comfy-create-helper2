def extract_character_tone_samples(db: Session, target_tag_set_id: Optional[int]) -> list[str]:
    if not target_tag_set_id: return []
    snaps = db.query(Snapshot).order_by(Snapshot.created_at.desc()).limit(50).all()
    samples = []
    for s in snaps:
        if not s.params: continue
        bubbles = s.params.get('speech_bubbles', []) if isinstance(s.params, dict) else json.loads(s.params).get('speech_bubbles', [])
        for b in bubbles:
            if b.get('speaker_tag_set_id') == target_tag_set_id and not b.get('is_se'):
                samples.append(b.get('text', ''))
                if len(samples) >= 3: return samples
    if not samples:
        tone = db.query(SpeechTonePreset).join(SpeechBubblePreset, SpeechBubblePreset.tone_preset_id == SpeechTonePreset.id)\
                 .filter(SpeechBubblePreset.tag_set_id == target_tag_set_id).first()
        if tone: samples.append(tone.prompt)
    return samples

def parse_vlm_json_with_fallback(raw_text: str) -> dict:
    cleaned = re.sub(r'<think>.*?</think>', '', raw_text, flags=re.DOTALL)
    match = re.search(r'```(?:json)?\s*([\s\S]*?)\s*```', cleaned)
    json_str = match.group(1) if match else cleaned
    return json.loads(json_str.strip())
