### 2.1 既存の注入関数 (`frontend/src/pages/DatasetStudioPage/components/MatrixGenerationStep.tsx` L19–L48)
```typescript
export function injectPromptAndSeed(
  workflowPromptJson: Record<string, any> | undefined,
  rawPrompt: string,
  seed: number
): Record<string, any> {
  const payload = workflowPromptJson ? JSON.parse(JSON.stringify(workflowPromptJson)) : {}

  let promptInjected = false
  let seedInjected = false

  for (const nodeId of Object.keys(payload)) {
    const node = payload[nodeId]
    if (!node || typeof node !== 'object') continue

    // Positive text prompt node (CLIPTextEncode)
    if (!promptInjected && (node.class_type === 'CLIPTextEncode' || node.class_type?.includes('CLIPText'))) {
      if (node.inputs) {
        node.inputs.text = rawPrompt
        promptInjected = true
      }
    }

    // Sampler node (KSampler / KSamplerAdvanced)
    if (!seedInjected && (node.class_type === 'KSampler' || node.class_type === 'KSamplerAdvanced')) {
      if (node.inputs) {
        node.inputs.seed = seed
        seedInjected = true
      }
    }
  }
```

## 3. 実装要件
1. `frontend/src/api/types.ts` の `DatasetProject`、`DatasetProjectCreate`、`DatasetProjectUpdate` に `prompt_negative?: string` を追加。
2. `InjectPromptOptions` インターフェースを定義：
   ```typescript
   export interface InjectPromptOptions {
     positiveNodeId?: string | null
     negativeNodeId?: string | null
     negativePrompt?: string
     seedNodeId?: string | null
   }
   ```
3. `injectPromptAndSeed` に `options?: InjectPromptOptions` を追加：
   - `options?.positiveNodeId` が存在すれば該当ノードの `inputs.text` に `rawPrompt` を注入。
   - `options?.negativeNodeId` が存在し `negativePrompt !== undefined` であれば該当ノードの `inputs.text` に `negativePrompt` を注入。
   - 指定ノードが存在しない場合は自動探索フォールバック（最初の CLIPTextEncode / 2番目の CLIPTextEncode またはタイトル判定）で安全に処理。
   - `negativePrompt` が空文字列 `""` の場合も安全に空文字として注入。
