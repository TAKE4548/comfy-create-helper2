### 2.2 既存の対象コード (`frontend/src/hooks/useMultiSelect.ts`)
```typescript
  const getLongPressHandlers = useCallback(
    (id: ID) => {
      let timer: ReturnType<typeof setTimeout> | null = null;
      let startX: number | null = null;
      let startY: number | null = null;

      const globalMove = (e: MouseEvent | TouchEvent) => {
        if (startX === null || startY === null) return;

        let currentX = 0;
        let currentY = 0;
        if ('touches' in e) {
          if (e.touches && e.touches.length > 0) {
            currentX = e.touches[0].clientX;
            currentY = e.touches[0].clientY;
          } else {
            return;
          }
        } else {
          currentX = e.clientX;
          currentY = e.clientY;
        }

        const diffX = currentX - startX;
        const diffY = currentY - startY;
        const distance = Math.sqrt(diffX * diffX + diffY * diffY);

        if (distance >= 5) {
          isDraggingRef.current = true;
          if (timer) {
            clearTimeout(timer);
            timer = null;
          }
          startX = null;
          startY = null;
          document.removeEventListener('mousemove', globalMove);
          document.removeEventListener('touchmove', globalMove);
        }
      };

      const globalClear = () => {
        if (timer) {
          clearTimeout(timer);
          timer = null;
        }
        startX = null;
        startY = null;
        isDraggingRef.current = false;

        document.removeEventListener('mousemove', globalMove);
        document.removeEventListener('touchmove', globalMove);
        document.removeEventListener('mouseup', globalClear);
        document.removeEventListener('touchend', globalClear);
        document.removeEventListener('touchcancel', globalClear);
      };

      const start = (e: React.MouseEvent | React.TouchEvent) => {
        if (isDraggingRef.current) return;
        if ('button' in e && e.button !== 0) return;

        if ('touches' in e) {
          if (e.touches && e.touches.length > 0) {
            startX = e.touches[0].clientX;
            startY = e.touches[0].clientY;
          }
        } else {
          startX = e.clientX;
          startY = e.clientY;
        }

        document.addEventListener('mousemove', globalMove);
        document.addEventListener('touchmove', globalMove, { passive: true });
        document.addEventListener('mouseup', globalClear);
        document.addEventListener('touchend', globalClear);
        document.addEventListener('touchcancel', globalClear);

        const stored = localStorage.getItem(LS_KEYS.LONG_PRESS_DELAY_MULTI_SELECT);
        const parsed = stored !== null ? parseFloat(stored) : NaN;
        const delay = !isNaN(parsed) ? parsed * 1000 : 800;
        timer = setTimeout(() => {
          enterMultiSelectMode(id);
          globalClear();
        }, delay);
      };

      return {
        onMouseDown: start,
        onMouseUp: globalClear,
        onMouseLeave: globalClear,
        onTouchStart: start,
        onTouchEnd: globalClear,
      };
    },
    [enterMultiSelectMode]
  );
```

## 3. 実装要件 (Implementation Logic)
- [x] 1. `frontend/src/hooks/useMultiSelect.ts`:
  - `justLongPressedRef = useRef(false);` を追加し、長押しタイマー完了時に `justLongPressedRef.current = true;` をセットして 400ms 後に `false` へ戻す。
  - `handleItemClick` の先頭で `if (justLongPressedRef.current) return;` を行い、長押し完了直後の合成クリックによる誤トグル（選択解除）を防止する。
  - `globalMove` における移動キャンセル判定閾値を `const MOVE_THRESHOLD = 'touches' in e ? 15 : 8;`（または 15px）に変更し、タッチ時の微小ブレで即座にキャンセルされないようにする。
  - `getLongPressHandlers` の返却オブジェクトに `onContextMenu: (e: React.MouseEvent) => e.preventDefault()` を追加する。
- [x] 2. `frontend/src/pages/HistoryPage.tsx`:
  - スナップショットカード要素に `onContextMenu={(e: React.MouseEvent) => e.preventDefault()}` および `style={{ touchAction: 'pan-y', WebkitTouchCallout: 'none', userSelect: 'none' }}`、`select-none` クラスを付与する。
  - スナップショット画像（`<img>`）に `pointer-events-none` クラスを付与する。
- [x] 3. `frontend/src/hooks/__tests__/useMultiSelect.test.ts`:
  - タッチ微小移動（10px等）で長押しがキャンセルされず、15px以上の移動で正しくキャンセルされるテストを追加。
  - 長押し成立直後のクリックイベントが無視され、選択状態が保持されるテストを追加。
