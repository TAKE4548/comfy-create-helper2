## 2. データ型定義 (Self-Contained Data Contracts & Types)

```typescript
// ─── API Types (frontend/src/api/types.ts) ───────────────────────
export interface SpeechBubblePreset {
  id: number
  name: string
  shape: BubbleShapeType
  orientation: TextOrientation
  font_family: string
  font_size: number
  text_color: string
  background_color: string
  border_color: string
  border_width: number
  sort_order: number
  created_at: string
}

export interface SpeechBubblePresetCreate {
  name: string
  shape?: BubbleShapeType
  orientation?: TextOrientation
  font_family?: string
  font_size?: number
  text_color?: string
  background_color?: string
  border_color?: string
  border_width?: number
  sort_order?: number
}

export interface SpeechBubblePresetUpdate {
  name?: string
  shape?: BubbleShapeType
  orientation?: TextOrientation
  font_family?: string
  font_size?: number
  text_color?: string
  background_color?: string
  border_color?: string
  border_width?: number
  sort_order?: number
}

// ─── API Client (frontend/src/api/client.ts) ─────────────────────
export const speechBubblePresetsApi = {
  list: () => request<SpeechBubblePreset[]>('/speech-bubble-presets'),
  create: (data: SpeechBubblePresetCreate) =>
    request<SpeechBubblePreset>('/speech-bubble-presets', {
      method: 'POST',
      body: JSON.stringify(data),
    }),
  update: (id: number, data: SpeechBubblePresetUpdate) =>
    request<SpeechBubblePreset>(`/speech-bubble-presets/${id}`, {
      method: 'PUT',
      body: JSON.stringify(data),
    }),
  delete: (id: number) =>
    request<void>(`/speech-bubble-presets/${id}`, {
      method: 'DELETE',
    }),
}
```

### 3.3 プリセット保存ロジック (`handleCreatePreset`)
現在の吹き出し（`bubble`）の装飾設定を抽出し、`speechBubblePresetsApi.create` を呼び出す：
```typescript
const handleCreatePreset = async () => {
  if (!newPresetName.trim() || !bubble) return
  setIsSavingPreset(true)
  try {
    const created = await speechBubblePresetsApi.create({
      name: newPresetName.trim(),
      shape: bubble.shape,
      orientation: bubble.orientation,
      font_family: bubble.fontFamily,
      font_size: bubble.fontSize || 0,
      text_color: bubble.textColor,
      background_color: bubble.backgroundColor,
      border_color: bubble.borderColor,
      border_width: bubble.borderWidth,
    })
    setPresets((prev) => [...prev, created])
    setNewPresetName('')
    setShowAddPresetInput(false)
  } catch (err: any) {
    alert('プリセットの保存に失敗しました: ' + err.message)
  } finally {
    setIsSavingPreset(false)
  }
}
```

### 3.4 プリセット更新・削除ロジック
- **上書き更新 (`handleOverwritePreset`)**: 対象プリセットに対して現在の `bubble` の修飾設定を `speechBubblePresetsApi.update` で反映。
- **削除 (`handleDeletePreset`)**: `confirm('プリセット「...」を削除しますか？')` 後、`speechBubblePresetsApi.delete` を呼び出してリストから除外。
