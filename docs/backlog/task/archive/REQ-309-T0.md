import type { Locator, Page } from '@playwright/test'
export type Vp = 'desktop' | 'mobile'
/** desktop(>=768px)=DesktopLayout は `prompt-<role>`、mobile(<768px)=MobileLayout は `prompt-mobile-<role>`（MobileLayout.tsx:767 / DesktopLayout.tsx:532）。 */
export const promptRoot = (page: Page, vp: Vp): Locator =>
  page.locator(vp === 'desktop' ? '#prompt-positive' : '#prompt-mobile-positive')
/** 判定点: 既定の Chips Mode で最初から操作できる入力（TagInput の <input type="text">）。追加操作を要さない。 */
export const promptChipInput = (page: Page, vp: Vp): Locator => promptRoot(page, vp).locator('input[type="text"]')

export async function switchPromptToTextMode(page: Page, vp: Vp): Promise<Locator> {
