### 1. `frontend/src/store/services/systemWsService.ts` (NEW)

```typescript
// 接続定数
export const SYS_BASE_RECONNECT_DELAY_MS = 1000;
export const SYS_MAX_RECONNECT_DELAY_MS = 10000;

export class SystemWsService {
  private ws: WebSocket | null = null;
  private reconnectTimer: ReturnType<typeof setTimeout> | null = null;
  private retryCount: number = 0;
  private suspended: boolean = false;

  // バックエンドの ws URL を構築
  // window.location.origin の http → ws 置換
  private buildUrl(): string {
    return window.location.origin.replace(/^http/, 'ws') + '/api/ws'
  }

  public connect(
    onMessage: (data: Record<string, unknown>) => void,
    onOpen?: () => void,
    onClose?: () => void
  ): void { ... }

  public disconnect(): void { ... }
  public suspend(): void { this.suspended = true; this.disconnect() }
  public resume(callbacks): void { this.suspended = false; this.connect(callbacks) }

  private getReconnectDelay(): number {
    // 指数バックオフ: 1s, 2s, 4s, 8s, 最大10s
    return Math.min(SYS_BASE_RECONNECT_DELAY_MS * Math.pow(2, this.retryCount), SYS_MAX_RECONNECT_DELAY_MS)
  }
}

export const systemWsService = new SystemWsService()
```

### 2. `frontend/src/hooks/useSystemStats.ts` (NEW)

```typescript
import { useEffect, useRef } from 'react'
import { useGenerationStore } from '../store/useGenerationStore'
import { systemWsService } from '../store/services/systemWsService'
import type { SystemStats } from '../store/types'

export function useSystemStats(): void {
  const setSystemStats = useGenerationStore(s => s.setSystemStats)
  const setSystemWsConnected = useGenerationStore(s => s.setSystemWsConnected)

  useEffect(() => {
    const onMessage = (data: Record<string, unknown>) => {
      // type === "SYSTEM_STATS" のみ処理
      if (data && (data as any).type === 'SYSTEM_STATS') {
        setSystemStats((data as any).data as SystemStats)
      }
    }
    const onOpen = () => setSystemWsConnected(true)
    const onClose = () => setSystemWsConnected(false)

    systemWsService.connect(onMessage, onOpen, onClose)

    // Page Visibility API
    const handleVisibility = () => {
      if (document.visibilityState === 'hidden') {
        systemWsService.suspend()
        setSystemWsConnected(false)
      } else {
        systemWsService.resume({ onMessage, onOpen, onClose })
      }
    }
    document.addEventListener('visibilitychange', handleVisibility)

    return () => {
      document.removeEventListener('visibilitychange', handleVisibility)
      systemWsService.disconnect()
      setSystemWsConnected(false)
    }
  }, [setSystemStats, setSystemWsConnected])
}
```

## 完了条件
- フック使用コンポーネントのアンマウント時に WS 切断 + タイマークリアされること
- `document.visibilityState === 'hidden'` 時に `suspend()` が呼ばれること
- 再接続バックオフが 1s → 2s → 4s → 8s → 10s (cap) で動作すること
