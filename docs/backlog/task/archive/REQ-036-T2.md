  const BASE_URL = '/api'
  async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
    const res = await fetch(`${BASE_URL}${path}`, {
      headers: { 'Content-Type': 'application/json', ...options.headers },
      ...options,
    })
    if (!res.ok) {
      const text = await res.text()
      throw new Error(`API error ${res.status}: ${text}`)
    }
    if (res.status === 204) return undefined as T
    return res.json()
  }

  export const lorasApi = {
    list: (category_id?: number) => request(`/loras${category_id ? `?category_id=${category_id}` : ''}`),
    create: (data: unknown) => request('/loras', { method: 'POST', body: JSON.stringify(data) }),
    update: (id: number, data: unknown) => request(`/loras/${id}`, { method: 'PUT', body: JSON.stringify(data) }),
    delete: (id: number) => request(`/loras/${id}`, { method: 'DELETE' }),
  }

## 3. 実装要件 (Implementation Logic)
- [x] `frontend/src/api/types.ts` に `ComfyLoraModel` のインタフェースおよび必要な型を追加する。
- [x] `frontend/src/api/client.ts` の `lorasApi` に `getComfyModels` メソッドを追加し、`/loras/comfy-models` から取得できるようにする（戻り値の型: `Promise<ComfyLoraModel[]>`）。
- [x] `frontend/src/components/LoraModals.tsx` の `LoraFormFields` 内で、React Query (`useQuery`) を用いて `/loras/comfy-models` を取得する（queryKey: `['comfy-lora-models']`）。
- [x] パス入力欄（`form.file_path`）をコンボボックスUIに変更する。
  - テキスト入力エリア (`<input>`) と、フォーカス/クリック時に表示されるドロップダウン選択リスト (`<ul>`/`<li>`) を組み合わせる。
  - 入力されたテキストでドロップダウンの選択肢をリアルタイムフィルタリング（部分一致インクリメンタル検索）する。
  - すでに登録されているモデル（`is_registered: true`）は、ドロップダウン内でグレーアウトし、「登録済み」バッジを表示して `pointer-events-none` などでクリック選択不可にする。
  - 未登録モデルを選択した場合は、`form.file_path` に選択された `file_path` を代入する。
  - 未登録モデル選択時、`form.model_name` が未入力の場合は、パスのベース名（`getBasename` 関数などを利用して抽出した値）を `form.model_name` に自動設定する。
  - リストにない新規パスの直接キーボード入力も許容し、手動入力可能なコンボボックスとして動作する。
  - ドロップダウンリストが開いているときに、入力エリアの外側をクリックしたらドロップダウンが閉じるよう、標準的なReactのイベントハンドリングを適用する。
