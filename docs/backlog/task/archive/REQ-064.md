### 2.2 useMultiSelect 内部ロジック

```typescript
function handleItemClick(id: number, event: React.MouseEvent) {
  if (!isMultiSelectMode) {
    if (event.ctrlKey || event.metaKey) {
      enterMultiSelectMode(id);
    }
    return;
  }

  if (event.shiftKey && lastSelectedIdRef.current !== null) {
    // 範囲選択
    const fromIdx = items.findIndex(i => i.id === lastSelectedIdRef.current);
    const toIdx = items.findIndex(i => i.id === id);
    const [start, end] = [Math.min(fromIdx, toIdx), Math.max(fromIdx, toIdx)];
    const rangeIds = items.slice(start, end + 1).map(i => i.id);
    setSelectedIds(prev => new Set([...prev, ...rangeIds]));
  } else {
    // Ctrl/Meta or 通常クリックでトグル
    setSelectedIds(prev => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  }
  lastSelectedIdRef.current = id;
}

// 長押し検出（500ms）
function getLongPressHandlers(id: number) {
  const timerRef = { current: null as ReturnType<typeof setTimeout> | null };
  return {
    onMouseDown: (e: React.MouseEvent) => {
      if (e.button !== 0) return;
      timerRef.current = setTimeout(() => enterMultiSelectMode(id), 500);
    },
    onMouseUp: () => clearTimeout(timerRef.current ?? undefined),
    onMouseLeave: () => clearTimeout(timerRef.current ?? undefined),
    onTouchStart: (_e: React.TouchEvent) => {
      timerRef.current = setTimeout(() => enterMultiSelectMode(id), 500);
    },
    onTouchEnd: () => clearTimeout(timerRef.current ?? undefined),
  };
}
```

// DragOverlay 用の状態管理パターン（各ページで使用）
const [activeId, setActiveId] = useState<number | null>(null);

function handleDragStart(event: DragStartEvent) {
  setActiveId(event.active.id as number);
}

function handleDragEnd(event: DragEndEvent) {
  const { active, over } = event;
  setActiveId(null);
  if (!over || active.id === over.id) return;

  if (isMultiSelectMode && selectedIds.has(active.id as number) && selectedIds.size > 1) {
    // グループ移動
    const selected = items.filter(i => selectedIds.has(i.id));
    const nonSelected = items.filter(i => !selectedIds.has(i.id));
    const insertAt = nonSelected.findIndex(i => i.id === over.id) + 1;
    const newOrder = [...nonSelected.slice(0, insertAt), ...selected, ...nonSelected.slice(insertAt)];
    setItems(newOrder);
    reorderMutation.mutate(newOrder.map(i => i.id));
  } else {
    const oldIndex = items.findIndex(i => i.id === active.id);
    const newIndex = items.findIndex(i => i.id === over.id);
    const newOrder = arrayMove(items, oldIndex, newIndex);
    setItems(newOrder);
    reorderMutation.mutate(newOrder.map(i => i.id));
  }
}

// DragOverlay の JSX パターン（各ページのカードコンポーネントを渡す）
<DragOverlay>

        {/* 各ページが renderDragOverlayItem(activeId) を実装して渡す */}
        {renderDragOverlayItem(activeId)}
      </div>

        <div className="absolute -bottom-2 -right-2 bg-[#3ECF8E] text-black text-xs font-bold
                        rounded-full w-6 h-6 flex items-center justify-center shadow-lg">
          +{selectedIds.size - 1}

</DragOverlay>
