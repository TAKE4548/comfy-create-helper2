    cancelJob: async (promptId: string): Promise<void> => {
      get().abortSubmission();
      if (!comfyUrl) return
      const activeUrl = resolveComfyUrl(comfyUrl)
      try {
        const isRunning = get().queueRunning.some((job) => job[1] === promptId)
        
        // 1. 実行中の場合は割り込みを投げる
        if (isRunning) {
          await fetch(`${activeUrl}/interrupt`, { method: 'POST' })
        } else {
          // 2. 待機中の場合はキューから削除する
          await fetch(`${activeUrl}/queue`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ delete: [promptId] } as ComfyQueueDeletePayload),
          })
        }

        // 3. 履歴に残らないよう削除を投げる (AC-5)
        await fetch(`${activeUrl}/history`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ delete: [promptId] } as ComfyHistoryDeletePayload),
        })

        // 4. 自セッションの実行中ジョブであった場合はステートをリセット
        if (promptId === get().currentPromptId || get().activeBatch?.some((job) => job.promptId === promptId)) {
          set({
            activeBatch: null,
            isGenerating: false,
            currentPromptId: null,
            activeClientId: null,
            generationStartTime: null,
            currentNode: null,
            nodeProgress: null,
            currentWorkflow: null,
          })
          localStorage.removeItem(LS_KEYS.CURRENT_PROMPT_ID)
        }

        await get().fetchQueue()
      } catch (e) {
        console.error('Failed to cancel job:', e)
      }
    },

## 3. 実装要件 (Implementation Logic)
- [ ] `useGenerationStore.ts`:
  1. `cancelGeneration` / `cancelAllJobs`（一括中断）の実行順序を厳密化:
     - 順序: **① `get().abortSubmission()`** ⇒ **② 待機キュージョブ削除 `POST /queue { delete: pendingIds }`** ⇒ **③ 実行中ジョブ中断 `POST /interrupt`** ⇒ **④ 履歴削除 `POST /history`** ⇒ **⑤ ステートリセット**。
     - 先に待機キューを削除することで、`/interrupt` で現ジョブ終了時に ComfyUI が次の待機ジョブを誤って起動するレースコンディションを完全に防止する。
  2. `cancelJob(promptId)`（個別ジョブ中断）の改修:
     - `const isCurrentRunning = get().currentPromptId === promptId` および `const isInQueueRunning = get().queueRunning.some((job) => job[1] === promptId)` を判定。
     - 実行中の場合は `POST /interrupt` を発行。
     - 待機中の場合は `POST /queue { delete: [promptId] }` を発行。
     - 自セッションのバッチジョブ中断時に対象ジョブを `status: 'failed'` に更新し、未完了ジョブが残っていれば `activeBatch` を維持して `isGenerating: true` のままにする（全件終了時のみ `activeBatch: null`, `isGenerating: false` に遷移）。
- [ ] `useGenerateAction.ts`:
  1. `handleGenerate` に `AbortController` を作成して `setSubmissionAbortController` に渡し、`queuePrompt` に `signal: abortController.signal` を渡す。中断時はループを即座に抜ける。
  2. `handleGenerateSets` で `initialBatch` を作成する際、既存の `activeBatch` が存在すれば `[...existingActiveBatch, ...newBatchJobs]` としてマージし、既存ジョブを保持して母数を合算する。
  3. `currentIdx` のオフセットを既存配列の長さから開始するよう調整。
