# REQ-288-T1: バックエンド WS ルーター + リソース監視サービス

## 目的
FastAPI に `/api/ws` WebSocket エンドポイントと、psutil/rocm-smi によるメトリクス収集サービスを新設する。

```python
# ConnectionManager: active_connections: list[WebSocket]
# async broadcast(message: str) -> None
#   → json.dumps({"type": "SYSTEM_STATS", "data": {...}}) を全クライアントへ

router = APIRouter(prefix="/api", tags=["websocket"])

@router.websocket("/ws")
async def ws_endpoint(websocket: WebSocket):
    await manager.connect(websocket)
    try:
        while True:
            # クライアントからのping受信を待つ (30s タイムアウト)
            await asyncio.wait_for(websocket.receive_text(), timeout=30.0)
    except (WebSocketDisconnect, asyncio.TimeoutError):
        manager.disconnect(websocket)
```

```python
from typing import Callable

class ResourceMonitorService:
    _task: asyncio.Task | None = None
    INTERVAL = 2.5

    async def start(self, broadcast_fn: Callable[[str], None]) -> None:
        self._task = asyncio.create_task(self._run(broadcast_fn))

    async def stop(self) -> None:
        if self._task: self._task.cancel()

    async def _run(self, broadcast_fn) -> None:
        while True:
            data = await self._collect()
            msg = json.dumps({"type": "SYSTEM_STATS", "data": data})
            await broadcast_fn(msg)
            await asyncio.sleep(self.INTERVAL)
```
