### 2.1 型定義と使用する Hook (`frontend/src/hooks/useMultiSelect.ts`)
```typescript
export interface UseMultiSelectReturn<ID extends number | string = number> {
  isMultiSelectMode: boolean;
  selectedIds: Set<ID>;
  enterMultiSelectMode: (firstId?: ID) => void;
  exitMultiSelectMode: () => void;
  handleItemClick: (id: ID, event: React.MouseEvent) => void;
  toggleItem: (id: ID) => void;
  selectAll: () => void;
  clearSelection: () => void;
  isSelected: (id: ID) => boolean;
  getLongPressHandlers: (id: ID) => {
    onMouseDown: (e: React.MouseEvent) => void;
    onMouseUp: () => void;
    onMouseLeave: () => void;
    onTouchStart: (e: React.TouchEvent) => void;
    onTouchEnd: () => void;
    onContextMenu?: (e: React.MouseEvent) => void;
  };
}

export function useMultiSelect<ID extends number | string = number, T extends { id: ID } = { id: ID }>({
  items,
}: { items: T[] }): UseMultiSelectReturn<ID>;
```

### 2.2 実装仕様 (`frontend/src/components/PresetManagerPanel.tsx`)
- `useMultiSelect` Hook を利用して `PresetManagerPanel` の一括選択ロジックをリファクタリング。
```typescript
import { useMultiSelect } from '../hooks/useMultiSelect';

// PresetManagerPanel コンポーネント内:
const multiSelect = useMultiSelect<number>({ items: localPresets });
const bulkMode = multiSelect.isMultiSelectMode;
const bulkSelectedIds = multiSelect.selectedIds;

const toggleBulkMode = () => {
  if (bulkMode) {
    multiSelect.exitMultiSelectMode();
  } else {
    multiSelect.enterMultiSelectMode();
  }
};
```
- `SortablePresetRow` に渡す `onItemClick`:
```typescript
<SortablePresetRow
  key={preset.id}
  preset={preset}
  // ...
  bulkMode={bulkMode}
  isBulkSelected={multiSelect.isSelected(preset.id)}
  onRowClick={(e: React.MouseEvent) => {
    if (bulkMode) {
      multiSelect.handleItemClick(preset.id, e);
    } else if (!editingId && onOpenAsTab) {
      onOpenAsTab(preset.id);
    }
  }}
  onCheckboxClick={(e: React.MouseEvent) => {
    e.stopPropagation();
    multiSelect.handleItemClick(preset.id, e);
  }}
/>
```
- `SortablePresetRow` の JSX 実装:
  - 行クリック時: `onClick={(e) => onRowClick?.(e)}`
  - チェックボックスクリック時: `onClick={(e) => onCheckboxClick?.(e)}`
  - これにより、Shiftキーを押しながら行またはチェックボックスをクリックした際に `multiSelect.handleItemClick` へイベントが伝播し、`useMultiSelect` による範囲選択（`items.slice(start, end + 1)`）が発火する。
