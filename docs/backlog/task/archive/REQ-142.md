export type BubbleShapeType = 'oval' | 'rounded-rect' | 'burst' | 'cloud' | 'none'
export type TextOrientation = 'vertical' | 'horizontal'

export interface BubbleTail {
  x: number // 元画像座標系での尻尾先端X
  y: number // 元画像座標系での尻尾先端Y
}

export interface BubbleStyleConfig {
  shape: BubbleShapeType
  orientation: TextOrientation
  fontFamily: string
  fontSize?: number // 0 または未指定時は自動フィット
  textColor: string // rgba(r,g,b,a)
  backgroundColor: string // rgba(r,g,b,a)
  borderColor: string // rgba(r,g,b,a)
  borderWidth: number // px
}

export const DEFAULT_BUBBLE_STYLE: BubbleStyleConfig = {
  shape: 'oval',
  orientation: 'vertical',
  fontFamily: 'sans-serif',
  fontSize: 0,
  textColor: 'rgba(0, 0, 0, 1)',
  backgroundColor: 'rgba(255, 255, 255, 0.95)',
  borderColor: 'rgba(0, 0, 0, 1)',
  borderWidth: 2,
}

export const DEFAULT_BUBBLE_BOUNDS = {
  w: 160,
  h: 140,
}

export const LAST_USED_STYLE_STORAGE_KEY = 'speech_bubble_last_used_style'

/**
 * 吹き出しオブジェクトから装飾スタイルプロパティのみを抽出する。
 * undefined のプロパティは除外し、明示的に設定された値のみを浅くマージする。
 */
export function extractBubbleStyle(bubble: Partial<SpeechBubble>): Partial<BubbleStyleConfig> {
  const style: Partial<BubbleStyleConfig> = {}
  if (bubble.shape !== undefined) style.shape = bubble.shape
  if (bubble.orientation !== undefined) style.orientation = bubble.orientation
  if (bubble.fontFamily !== undefined) style.fontFamily = bubble.fontFamily
  if (bubble.fontSize !== undefined) style.fontSize = bubble.fontSize
  if (bubble.textColor !== undefined) style.textColor = bubble.textColor
  if (bubble.backgroundColor !== undefined) style.backgroundColor = bubble.backgroundColor
  if (bubble.borderColor !== undefined) style.borderColor = bubble.borderColor
  if (bubble.borderWidth !== undefined) style.borderWidth = bubble.borderWidth
  return style
}

/**
 * localStorage から直前に保存された装飾スタイル設定を読み込む。
 * SSR環境、localStorage無効、JSONパースエラー等の例外時は DEFAULT_BUBBLE_STYLE を安全に返す。
 */
export function loadLastUsedBubbleStyle(): BubbleStyleConfig {
  if (typeof window === 'undefined' || !window.localStorage) {
    return DEFAULT_BUBBLE_STYLE
  }
  try {
    const saved = localStorage.getItem(LAST_USED_STYLE_STORAGE_KEY)
    if (!saved) return DEFAULT_BUBBLE_STYLE
    const parsed = JSON.parse(saved)
    return {
      ...DEFAULT_BUBBLE_STYLE,
      ...extractBubbleStyle(parsed),
    }
  } catch {
    return DEFAULT_BUBBLE_STYLE
  }
}

/**
 * 現在の装飾スタイル設定を localStorage に保存する。
 * 既存の設定に対して渡されたスタイルをマージして永続化する。
 * QuotaExceededError や SecurityError など発生時はエラーを握り潰し安全に終了する。
 */
export function saveLastUsedBubbleStyle(style: Partial<BubbleStyleConfig>): void {
  if (typeof window === 'undefined' || !window.localStorage) return
  try {
    const current = loadLastUsedBubbleStyle()
    const merged = { ...current, ...extractBubbleStyle(style) }
    localStorage.setItem(LAST_USED_STYLE_STORAGE_KEY, JSON.stringify(merged))
  } catch {
    // QuotaExceededError やプライベートモードでの保存エラーは無視
  }
}

export interface SpeechBubble {
  id: string
  x: number // 元画像座標系でのバウンディングボックスX
  y: number // 元画像座標系でのバウンディングボックスY
  w: number // 元画像座標系での幅
  h: number // 元画像座標系での高さ
  text: string
  shape: BubbleShapeType
  orientation: TextOrientation
  fontFamily: string
  fontSize?: number // 0 または未指定時は自動フィット
  textColor: string // rgba(r,g,b,a)
  backgroundColor: string // rgba(r,g,b,a)
  borderColor: string // rgba(r,g,b,a)
  borderWidth: number // px
  tail?: BubbleTail
}

  type: 'bubble' | 'tail' | 'resize'

### 2.4 `SpeechBubbleEditorModal.tsx` 内の完全な実装スニペット
```typescript
export function SpeechBubbleEditorModal({
  snapshot,
  onClose,
  onSaved,
}: SpeechBubbleEditorModalProps) {
  // 1. ステート定義 & Refs
  const [bubbles, setBubbles] = useState<SpeechBubble[]>([])
  const [selectedBubbleId, setSelectedBubbleId] = useState<string | null>(null)
  const [fontList, setFontList] = useState<ServerFont[]>([])
  const [selectedFont, setSelectedFont] = useState<string>('')
  const [lastUsedStyle, setLastUsedStyle] = useState<BubbleStyleConfig>(() => loadLastUsedBubbleStyle())
  const [imageObj, setImageObj] = useState<HTMLImageElement | null>(null)
  const [canvasDimensions, setCanvasDimensions] = useState({ width: 500, height: 500 })
  const [draggingState, setDraggingState] = useState<DraggingState | null>(null)

  const canvasRef = useRef<HTMLCanvasElement | null>(null)
  const scaleX = imageObj ? canvasDimensions.width / imageObj.naturalWidth : 1
  const scaleY = imageObj ? canvasDimensions.height / imageObj.naturalHeight : 1

  // 2. 画像ロード
  useEffect(() => {
    if (!snapshot.image_path) return
    const img = new Image()
    img.src = snapshot.image_path
    img.crossOrigin = 'anonymous'
    img.onload = () => {
      setImageObj(img)
      const maxW = typeof window !== 'undefined' ? Math.min(500, window.innerWidth - 48) : 500
      const maxH = 500
      let w = img.naturalWidth
      let h = img.naturalHeight
      if (w > maxW || h > maxH) {
        const ratio = Math.min(maxW / w, maxH / h)
        w = Math.round(w * ratio)
        h = Math.round(h * ratio)
      }
      setCanvasDimensions({ width: w, height: h })
    }
  }, [snapshot.image_path])

  // 3. フォント取得時の初期化
  useEffect(() => {
    const fetchFonts = async () => {
      try {
        const list = await fontsApi.list()
        setFontList(list)
        if (list.length > 0) {
          const msgothic = list.find((f) => f.file_name.toLowerCase() === 'msgothic.ttc')
          const initialFont = msgothic ? msgothic.file_name : list[0].file_name
          setSelectedFont(initialFont)
          // lastUsedStyleのフォントがデフォルト('sans-serif')のままの場合は初期フォントを同期
          setLastUsedStyle((prev) => (prev.fontFamily === 'sans-serif' ? { ...prev, fontFamily: initialFont } : prev))
        }
      } catch (err) {
        console.error('Failed to fetch font list:', err)
      }
    }
    fetchFonts()
  }, [])

  // 4. 吹き出し新規作成ヘルパー
  const createBubbleFromStyle = (params: {
    id?: string
    x: number
    y: number
    w?: number
    h?: number
    text?: string
    tail?: BubbleTail
  }): SpeechBubble => {
    return {
      id: params.id || `bubble-${Date.now()}-${Math.random().toString(36).substring(2, 6)}`,
      x: params.x,
      y: params.y,
      w: params.w ?? DEFAULT_BUBBLE_BOUNDS.w,
      h: params.h ?? DEFAULT_BUBBLE_BOUNDS.h,
      text: params.text ?? '',
      shape: lastUsedStyle.shape,
      orientation: lastUsedStyle.orientation,
      fontFamily: lastUsedStyle.fontFamily || selectedFont || 'sans-serif',
      fontSize: lastUsedStyle.fontSize ?? 0,
      textColor: lastUsedStyle.textColor,
      backgroundColor: lastUsedStyle.backgroundColor,
      borderColor: lastUsedStyle.borderColor,
      borderWidth: lastUsedStyle.borderWidth,
      tail: params.tail,
    }
  }

  // 5. キャンバスクリック時の吹き出し選択 (handleMouseDown 内)
  const handleMouseDown = (e: React.MouseEvent<HTMLCanvasElement>) => {
    const canvas = canvasRef.current
    if (!canvas) return
    const rect = canvas.getBoundingClientRect()
    const x = e.clientX - rect.left
    const y = e.clientY - rect.top

    const reversedBubbles = [...bubbles].reverse()
    for (const b of reversedBubbles) {
      const bx = b.x * scaleX
      const by = b.y * scaleY
      const bw = b.w * scaleX
      const bh = b.h * scaleY

      if (x >= bx && x <= bx + bw && y >= by && y <= by + bh) {
        setSelectedBubbleId(b.id)
        const style = extractBubbleStyle(b)
        setLastUsedStyle((prev) => ({ ...prev, ...style }))
        saveLastUsedBubbleStyle(style)
        setDraggingState({
          type: 'bubble',
          bubbleId: b.id,
          startX: x,
          startY: y,
          initialBubble: { ...b }
        })
        return
      }
    }
  }

  // 6. SpeechBubbleControlsのonUpdateハンドラ
  const handleBubbleUpdate = (updated: Partial<SpeechBubble>) => {
    if (!selectedBubbleId) return
    setBubbles((prev) =>
      prev.map((b) => (b.id === selectedBubbleId ? { ...b, ...updated } : b))
    )
    const updatedStyle = extractBubbleStyle(updated)
    if (Object.keys(updatedStyle).length > 0) {
      setLastUsedStyle((prev) => {
        const next = { ...prev, ...updatedStyle }
        saveLastUsedBubbleStyle(next)
        return next
      })
    }
  }

  // 7. ダブルクリックによる新規作成
  const handleDoubleClick = (e: React.MouseEvent<HTMLCanvasElement>) => {
    const canvas = canvasRef.current
    if (!canvas) return
    const rect = canvas.getBoundingClientRect()
    const x = e.clientX - rect.left
    const y = e.clientY - rect.top

    const cx = Math.round(x / scaleX)
    const cy = Math.round(y / scaleY)
    const newBubble = createBubbleFromStyle({
      x: Math.max(0, cx - 80),
      y: Math.max(0, cy - 70),
      w: DEFAULT_BUBBLE_BOUNDS.w,
      h: DEFAULT_BUBBLE_BOUNDS.h,
      tail: { x: cx - 30, y: cy + 90 }
    })
    setBubbles((prev) => [...prev, newBubble])
    setSelectedBubbleId(newBubble.id)
  }

  return (
    <div>
      <SpeechBubbleControls
        bubble={bubbles.find((b) => b.id === selectedBubbleId) || null}
        bubblesCount={bubbles.length}
        fontList={fontList}
        isFontLoading={false}
        isDrafting={false}
        onUpdate={handleBubbleUpdate}
        onDelete={() => {
          if (!selectedBubbleId) return
          setBubbles((prev) => prev.filter((b) => b.id !== selectedBubbleId))
          setSelectedBubbleId(null)
        }}
        onGenerateDraft={() => {}}
      />
    </div>
  )
}
```
