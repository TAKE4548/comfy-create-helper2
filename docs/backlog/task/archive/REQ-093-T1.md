```typescript
export const parseToChips = (text: string): TagChip[] => {
  if (!text) return [];
  return text
    .split(',')
    .map((item) => item.trim())
    .filter((item) => item.length > 0)
    .map((item) => ({
      id: Math.random().toString(36).substring(2, 9),
      text: item,
      enabled: true,
    }));
};
```
