```typescript
export const parseToChips = (text: string): TagChip[] => {
  if (!text) return [];
  return text
    .split(',')
    .map((item) => item.trim())
    .filter((item) => item.length > 0)
    .map((item) => ({
      id: generateChipId(),
      text: item,
      enabled: true,
    }));
};

export const chipsToString = (chips: TagChip[]): string => {
  return chips
    .filter((chip) => chip.enabled)
    .map((chip) => chip.text)
    .join(', ');
};
```

## 3. 実装要件 (Implementation Logic)
- [ ] `frontend/src/components/TagChipEditor/useChipParser.ts`
  - `splitTags(text: string, isAnimaMode: boolean = false): string[]` を新規追加・エクスポートする。
    - 括弧深度 `()` `[]` `{}` を管理し、深度が1以上の場合は `,` や `.` で分割しない。
    - 深度0のとき、`,` は常にトークン区切り（カンマ自体はトークンに含めず除外）。
    - 深度0のとき、`isAnimaMode: true` の場合:
      - 直前文字が数字かつ直後文字が数字のピリオド（`/\d$/.test(current) && /\d/.test(text[i + 1])`）は小数点として保護し分割しない。
      - それ以外のピリオドはトークン末尾にピリオドを保持したまま区切りとして扱う（`current += '.'`）。
    - 連続区切り（`..`, `, .` 等）に起因する空トークンやピリオド/カンマ/空白のみのトークンは除外する（`token.replace(/[.,\s]/g, '').length > 0`）。
  - `parseToChips(text: string, isAnimaMode: boolean = false): TagChip[]` を更新し、`splitTags(text, isAnimaMode)` を使用してチップ配列を生成する。
  - `chipsToString(chips: TagChip[], isAnimaMode: boolean = false): string` を更新する:
    - `isAnimaMode: false` の場合は従来通り `, ` で結合。
    - `isAnimaMode: true` の場合、有効チップの連続間において直前のチップ末尾が `.` で終わる場合は半角スペース `" "` で結合し、それ以外は `, ` で結合する。
  - `useChipParser` フックの戻り値に `splitTags` を追加し、シグネチャを整合させる。
