import React, { useState, useRef, useEffect } from 'react';
import { useSortable } from '@dnd-kit/sortable';
import { CSS } from '@dnd-kit/utilities';
import { X, GripVertical } from 'lucide-react';
import { parseTagWeight, formatTagWeight, adjustWeight } from './useChipParser';

export interface TagChipProps {
  id: string;
  text: string;
  enabled: boolean;
  onToggle: (id: string) => void;
  onDelete: (id: string) => void;
  onEdit: (id: string, text: string) => void;
  onDuplicate: (id: string) => void;
}

### 2.2 実装設計スニペット (`TagChip.tsx`)
```tsx
export const TagChip: React.FC<TagChipProps> = ({
  id,
  text,
  enabled,
  onToggle,
  onDelete,
  onEdit,
  onDuplicate,
}) => {
  const [isEditing, setIsEditing] = useState(false);
  const [editValue, setEditValue] = useState(text);
  const [isMenuOpen, setIsMenuOpen] = useState(false);
  const inputRef = useRef<HTMLInputElement>(null);
  const menuRef = useRef<HTMLDivElement>(null);

  const {
    attributes,
    listeners,
    setNodeRef,
    transform,
    transition,
    isDragging,
  } = useSortable({ id });

  // タグ文字列からベース名と現在の強度を取得
  const { baseText, weight } = parseTagWeight(text);
  // UI表示用の強度文字列（未設定時は 1.0）
  const [weightInput, setWeightInput] = useState(weight !== null ? weight.toString() : '1.0');

  useEffect(() => {
    const { weight: currentWeight } = parseTagWeight(text);
    setWeightInput(currentWeight !== null ? currentWeight.toString() : '1.0');
    setEditValue(text);
  }, [text]);

  // サブメニュー外クリック検知
  useEffect(() => {
    if (!isMenuOpen) return;
    const handleClickOutside = (e: MouseEvent) => {
      if (menuRef.current && !menuRef.current.contains(e.target as Node)) {
        setIsMenuOpen(false);
      }
    };
    document.addEventListener('mousedown', handleClickOutside);
    return () => {
      document.removeEventListener('mousedown', handleClickOutside);
    };
  }, [isMenuOpen]);

  // ドラッグ時のインラインスタイル
  const style: React.CSSProperties = {
    transform: isDragging
      ? transform
        ? `${CSS.Transform.toString(transform)} scale(1.05)`
        : 'scale(1.05)'
      : CSS.Transform.toString(transform),
    transition,
    opacity: isDragging ? 0.8 : 1,
    zIndex: isDragging ? 50 : undefined,
    boxShadow: isDragging ? '0 10px 15px -3px rgba(0, 0, 0, 0.3)' : undefined,
  };

  useEffect(() => {
    if (isEditing && inputRef.current) {
      inputRef.current.focus();
      inputRef.current.select();
    }
  }, [isEditing]);

  const handleDoubleClick = (e: React.MouseEvent) => {
    e.stopPropagation();
    setIsMenuOpen(false);
    setIsEditing(true);
  };

  const handleChipClick = (e: React.MouseEvent) => {
    e.stopPropagation();
    if (isEditing) return;
    setIsMenuOpen((prev) => !prev);
  };

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (isEditing) return;

    if (e.key === 'Escape' && isMenuOpen) {
      e.preventDefault();
      setIsMenuOpen(false);
    } else if (e.key === ' ') {
      e.preventDefault();
      onToggle(id);
    } else if (e.key === 'Delete' || e.key === 'Backspace') {
      e.preventDefault();
      onDelete(id);
    } else if ((e.ctrlKey || e.metaKey) && e.key === 'd') {
      e.preventDefault();
      onDuplicate(id);
    }
  };

  const handleStepWeight = (delta: number) => {
    const { baseText: currentBase, weight: currentWeight } = parseTagWeight(text);
    const newWeight = adjustWeight(currentWeight, delta);
    setWeightInput(newWeight.toString());
    const newFormatted = formatTagWeight(currentBase, newWeight);
    onEdit(id, newFormatted);
  };

  const handleDirectWeightChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    setWeightInput(e.target.value);
  };

  const handleWeightSubmit = () => {
    const parsed = parseFloat(weightInput);
    const { baseText: currentBase } = parseTagWeight(text);
    if (!isNaN(parsed)) {
      const clamped = Math.max(0, parseFloat(parsed.toFixed(2)));
      setWeightInput(clamped.toString());
      const newFormatted = formatTagWeight(currentBase, clamped);
      onEdit(id, newFormatted);
    } else {
      const { weight: currentWeight } = parseTagWeight(text);
      setWeightInput(currentWeight !== null ? currentWeight.toString() : '1.0');
    }
  };

  const handleWeightKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
    if (e.key === 'Enter') {
      e.preventDefault();
      handleWeightSubmit();
    } else if (e.key === 'Escape') {
      e.preventDefault();
      setIsMenuOpen(false);
    }
  };

  const handleEditSubmit = () => {
    const trimmed = editValue.trim();
    if (trimmed === '') {
      onDelete(id);
    } else {
      onEdit(id, trimmed);
    }
    setIsEditing(false);
  };

  const handleEditCancel = () => {
    setEditValue(text);
    setIsEditing(false);
  };

  const handleInputKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
    if (e.key === 'Enter' || e.key === 'Tab') {
      e.preventDefault();
      handleEditSubmit();
    } else if (e.key === 'Escape') {
      e.preventDefault();
      handleEditCancel();
    }
  };

  const dragProps = isEditing ? {} : { ...attributes, ...listeners };

  return (
    <div
      ref={setNodeRef}
      style={style}
      tabIndex={isEditing ? -1 : 0}
      onKeyDown={handleKeyDown}
      className={`
        relative flex items-center gap-1.5 px-2.5 py-1 rounded-md border text-sm font-medium transition-colors select-none outline-none
        ${isEditing ? 'bg-[#18181C] border-[#7C6FDD]' : ''}
        ${!isEditing && enabled ? 'bg-[#1F1F25] border-[rgba(255,255,255,0.08)] text-[#E4E4E7] hover:border-[rgba(255,255,255,0.15)] focus:border-[#7C6FDD]' : ''}
        ${!isEditing && !enabled ? 'bg-zinc-900 border-dashed border-[rgba(255,255,255,0.05)] text-[#A1A1AA] opacity-50 line-through' : ''}
      `}
      title="クリックでメニュー / ダブルクリックで編集 / Spaceで切替 / Delで削除 / Ctrl+Dで複製"
      aria-roledescription="ドラッグで並び替え可能"
    >
      {/* ドラッグハンドル */}
      {!isEditing && (
        <div
          {...dragProps}
          className="cursor-grab text-zinc-500 hover:text-zinc-300 active:cursor-grabbing p-0.5"
          title="ドラッグで並び替え"
        >
          <GripVertical size={14} />
        </div>
      )}

      {/* テキスト/編集インプット */}
      {isEditing ? (
        <input
          ref={inputRef}
          type="text"
          value={editValue}
          onChange={(e) => setEditValue(e.target.value)}
          onBlur={handleEditSubmit}
          onKeyDown={handleInputKeyDown}
          className="bg-transparent border-none text-[#E4E4E7] text-sm focus:outline-none w-20 min-w-[60px]"
        />
      ) : (
        <span
          onDoubleClick={handleDoubleClick}
          onClick={handleChipClick}
          className="cursor-pointer flex-grow"
        >
          {text}
        </span>
      )}

      {/* 削除ボタン */}
      <button
        type="button"
        onClick={(e) => {
          e.stopPropagation();
          onDelete(id);
        }}
        className="text-zinc-500 hover:text-zinc-300 rounded p-0.5 transition-colors focus:outline-none focus:ring-1 focus:ring-zinc-400"
        aria-label={`タグ ${text} を削除`}
      >
        <X size={13} />
      </button>

      {/* サブメニュー (ポップオーバー) */}
      {isMenuOpen && !isEditing && (
        <div
          ref={menuRef}
          className="absolute top-full left-0 mt-1.5 z-50 min-w-[200px] p-2.5 bg-[#18181C] border border-[rgba(255,255,255,0.15)] rounded-lg shadow-2xl flex flex-col gap-2 text-xs select-none"
          onClick={(e) => e.stopPropagation()}
        >
          {/* ヘッダー: タグ名 + 有効/無効トグル */}
          <div className="flex items-center justify-between gap-2 border-b border-[rgba(255,255,255,0.08)] pb-1.5">
            <span className="font-semibold text-[#E4E4E7] truncate max-w-[120px]" title={baseText}>
              {baseText}
            </span>
            <button
              type="button"
              onClick={() => onToggle(id)}
              className={`px-2 py-0.5 rounded text-[11px] font-medium transition-colors ${
                enabled
                  ? 'bg-[#3ecf8e] text-zinc-950 font-bold hover:bg-[#3ecf8e]/90'
                  : 'bg-zinc-700 text-zinc-300 hover:bg-zinc-600'
              }`}
            >
              {enabled ? '有効' : '無効'}
            </button>
          </div>

          {/* 強度調整エリア */}
          <div className="flex flex-col gap-1.5">
            <div className="flex items-center justify-between text-[#A1A1AA]">
              <span>強度 (Weight)</span>
              <input
                type="number"
                step="0.1"
                value={weightInput}
                onChange={handleDirectWeightChange}
                onBlur={handleWeightSubmit}
                onKeyDown={handleWeightKeyDown}
                className="w-16 px-1.5 py-0.5 bg-[#1F1F25] border border-[rgba(255,255,255,0.1)] rounded text-right text-[#E4E4E7] focus:outline-none focus:border-[#7C6FDD]"
              />
            </div>

            {/* ステップ増減ボタン */}
            <div className="grid grid-cols-4 gap-1">
              <button
                type="button"
                onClick={() => handleStepWeight(-1.0)}
                className="px-1.5 py-1 bg-[#1F1F25] hover:bg-zinc-700 text-[#E4E4E7] rounded border border-[rgba(255,255,255,0.05)] text-center transition-colors font-mono"
              >
                -1.0
              </button>
              <button
                type="button"
                onClick={() => handleStepWeight(-0.1)}
                className="px-1.5 py-1 bg-[#1F1F25] hover:bg-zinc-700 text-[#E4E4E7] rounded border border-[rgba(255,255,255,0.05)] text-center transition-colors font-mono"
              >
                -0.1
              </button>
              <button
                type="button"
                onClick={() => handleStepWeight(0.1)}
                className="px-1.5 py-1 bg-[#1F1F25] hover:bg-zinc-700 text-[#E4E4E7] rounded border border-[rgba(255,255,255,0.05)] text-center transition-colors font-mono"
              >
                +0.1
              </button>
              <button
                type="button"
                onClick={() => handleStepWeight(1.0)}
                className="px-1.5 py-1 bg-[#1F1F25] hover:bg-zinc-700 text-[#E4E4E7] rounded border border-[rgba(255,255,255,0.05)] text-center transition-colors font-mono"
              >
                +1.0
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};
```

## 3. 実装要件 (Implementation Logic)
- [x] `TagChip.tsx`:
  - テキスト部分のクリックでサブメニュー（ポップオーバー）を開閉（`isMenuOpen`）するように変更。
  - サブメニュー外クリック検知（`useEffect` での `mousedown` リスナー）および `Escape` キーでの閉じる処理を実装。
  - サブメニュー内に有効/無効トグルボタン、強度数値入力フィールド、4つのステップ増減ボタン（`-1.0`, `-0.1`, `+0.1`, `+1.0`）を実装。
  - 強度未設定タグ（`weight === null`）の場合は初期値 `1.0` として扱い、ステップ増減または入力確定時に `formatTagWeight` を通して `onEdit(id, newText)` を呼び出す。
  - チップコンテナに `relative` を付与し、サブメニューが適切にポップオーバー配置されるようにする。
- [x] `TagChip.test.tsx`:
  - チップクリックでサブメニューが表示されることの検証。
  - サブメニュー内のトグルボタンクリックで `onToggle(id)` が発火することの検証。
  - 未設定タグに対して `+0.1` 押下で `onEdit(id, '(tag:1.1)')` が呼ばれることの検証。
  - 数値直接入力で強度更新が行われることの検証。
  - 外側クリックまたは Escape でサブメニューが閉じることの検証。
