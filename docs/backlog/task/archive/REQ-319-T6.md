### 2.3 ComfyUI 直接バッチ投入ロジック (`MatrixGenerationStep.tsx`)
```typescript
const submitDirectBatchToComfy = async (items: DatasetItem[], workflowPromptJson: any) => {
  const comfyUrl = 'http://127.0.0.1:8188';
  for (let i = 0; i < items.length; i++) {
    const item = items[i];
    // Clone workflow, inject item.raw_prompt to positive prompt node, and item.seed to KSampler
    const payload = injectPromptAndSeed(workflowPromptJson, item.raw_prompt, item.seed);
    await fetch(`${comfyUrl}/prompt`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ prompt: payload, client_id: clientId })
    });
    setBatchProgress((i + 1) / items.length * 100);
  }
};
```
