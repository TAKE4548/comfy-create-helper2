@router.post("/scene-generate", response_model=list[schemas.SceneDiffParams])
async def scene_generate(payload: schemas.SceneGenerateRequest, db: Session = Depends(get_db)):
    from app.services.llm_service import run_scene_generate
    prompt_record = db.query(models.LlmPrompt).filter(models.LlmPrompt.key == "scene_generate").first()
    system_prompt_override = prompt_record.content if prompt_record else None
    try:
        scenes_list = [{"id": s.id, "summary": s.summary} for s in payload.scenes]
        results = await run_scene_generate(
            target_fields=payload.target_fields,
            base_params=payload.base_params,
            scenes=scenes_list,
            system_prompt_override=system_prompt_override,
        )
        return results
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

#### 2.2 useSceneGenerator.ts 全体
```typescript
import { llmPromptsApi, workflowPresetsApi } from '../../../api/client'
import type { SceneItem, SceneDiffParams } from '../../../api/types'

interface UseSceneGeneratorParams {
  workflowId: number
  baseParamValues: Record<string, string>
}

export function useSceneGenerator({ workflowId, baseParamValues }: UseSceneGeneratorParams) {

  const generateDrafts = async (
    targetFields: string[],
    scenes: SceneItem[],
    overrideBaseParams?: Record<string, string>,
  ): Promise<SceneDiffParams[]> => {
    const activeBase = overrideBaseParams || baseParamValues
    const baseParams: Record<string, string> = {}
    targetFields.forEach(f => { baseParams[f] = activeBase[f] ?? '' })

    return await llmPromptsApi.sceneGenerate({
      target_fields: targetFields,
      base_params: baseParams,
      scenes: scenes.map(s => ({ id: s.id, summary: s.summary })),
    })
  }

  const saveAsProfiles = async (
    scenes: SceneItem[],
    drafts: SceneDiffParams[],
    overrideBaseParams?: Record<string, string>,
  ): Promise<void> => {
    const activeBase = overrideBaseParams || baseParamValues
    for (const draft of drafts) {
      const scene = scenes.find(s => s.id === draft.scene_id)
      const profileName = scene?.summary ?? `Scene ${draft.scene_id}`
      const finalParams = { ...activeBase, ...draft.diff_params }

      await workflowPresetsApi.create({
        workflow_id: workflowId,
        name: profileName,
        param_values: finalParams,
        applied_assets: {},
        batch_size: 1,
        seed_mode: 'random',
        seed: 42,
      })
    }
  }

  return { generateDrafts, saveAsProfiles }
}
```
