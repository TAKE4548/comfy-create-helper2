### 2.1 `backend/app/middleware/request_logging.py` 実装仕様
```python
import time
import uuid
import logging
from starlette.middleware.base import BaseHTTPMiddleware, RequestResponseEndpoint
from starlette.requests import Request
from starlette.responses import Response

logger = logging.getLogger("app.middleware.request")


class RequestLoggingMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next: RequestResponseEndpoint) -> Response:
        request_id = request.headers.get("X-Request-ID") or f"req-{uuid.uuid4().hex[:12]}"
        request.state.request_id = request_id

        start_time = time.perf_counter()
        status_code = 500
        try:
            response = await call_next(request)
            status_code = response.status_code
            response.headers["X-Request-ID"] = request_id
            return response
        except Exception as exc:
            # Let global exception handler process, but record status_code 500
            status_code = 500
            raise exc
        finally:
            duration_ms = round((time.perf_counter() - start_time) * 1000, 2)
            # Skip noisy static images from normal INFO logging
            path = request.url.path
            is_noise = path.startswith("/api/snapshots/images") or path == "/health"
            log_level = logging.DEBUG if is_noise else logging.INFO

            extra_data = {
                "request_id": request_id,
                "http": {
                    "method": request.method,
                    "path": path,
                    "status_code": status_code,
                    "client_ip": request.client.host if request.client else None,
                    "duration_ms": duration_ms,
                },
                "source": "backend",
            }
            logger.log(
                log_level,
                f"{request.method} {path} completed with {status_code} in {duration_ms}ms",
                extra=extra_data,
            )
```

### 2.2 `backend/app/routers/system_logs.py` 実装仕様
```python
import logging
from typing import Optional, Any
from fastapi import APIRouter, Response, status
from pydantic import BaseModel

router = APIRouter(prefix="/api/system", tags=["system"])
client_logger = logging.getLogger("app.client")


class ClientLogPayload(BaseModel):
    level: str = "error"
    message: str
    stack: Optional[str] = None
    url: Optional[str] = None
    component: Optional[str] = None
    context: Optional[dict[str, Any]] = None


@router.post("/client-logs", status_code=status.HTTP_204_NO_CONTENT)
async def ingest_client_logs(payload: ClientLogPayload):
    level_num = getattr(logging, payload.level.upper(), logging.ERROR)
    error_dict = None
    if payload.stack or payload.message:
        error_dict = {
            "type": "ClientError",
            "message": payload.message,
            "stack_trace": payload.stack,
        }

    extra_data = {
        "source": "frontend",
        "error": error_dict,
        "extra": {
            "url": payload.url,
            "component": payload.component,
            "context": payload.context,
        },
    }

    client_logger.log(
        level_num,
        f"[FE] {payload.component or 'Global'}: {payload.message}",
        extra=extra_data,
    )
    return Response(status_code=status.HTTP_204_NO_CONTENT)
```

### 2.3 `backend/app/main.py` 修正差分スニペット

```python
from app.logging_config import setup_logging
from app.middleware.request_logging import RequestLoggingMiddleware
from app.routers import system_logs
from fastapi.responses import JSONResponse
from fastapi.requests import Request
import traceback

# Initialize logging at startup
setup_logging(log_dir=settings.log_path, log_level=settings.log_level, retention_days=settings.log_retention_days)

# Global unhandled exception handler
@app.exception_handler(Exception)
async def unhandled_exception_handler(request: Request, exc: Exception):
    request_id = getattr(request.state, "request_id", None)
    logger.error(
        f"Unhandled exception on {request.method} {request.url.path}: {str(exc)}",
        exc_info=True,
        extra={
            "request_id": request_id,
            "source": "backend",
            "http": {
                "method": request.method,
                "path": request.url.path,
                "status_code": 500,
                "client_ip": request.client.host if request.client else None,
            },
            "error": {
                "type": type(exc).__name__,
                "message": str(exc),
                "stack_trace": traceback.format_exc(),
            },
        },
    )
    return JSONResponse(
        status_code=500,
        content={"detail": "Internal Server Error", "request_id": request_id},
    )

# Middlewares
app.add_middleware(RequestLoggingMiddleware)

# Routers
app.include_router(system_logs.router)
```
