### Step 2: `ResizableSplitLayout.tsx` の新規作成
`frontend/src/components/layout/ResizableSplitLayout.tsx` に以下の完全な実装コードを配置する。

```typescript
import React, { useState, useEffect, useRef } from 'react'
import { ResizableSplitLayoutProps } from './ResizableSplitLayout'

export function ResizableSplitLayout({
  direction,
  first,
  second,
  initialSize,
  minSize,
  maxSize,
  storageKey,
  className = ''
}: ResizableSplitLayoutProps) {
  const [size, setSize] = useState<number>(() => {
    const stored = localStorage.getItem(storageKey)
    return stored ? parseInt(stored, 10) : initialSize
  })
  
  const isDragging = useRef(false)
  const containerRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    localStorage.setItem(storageKey, size.toString())
  }, [size, storageKey])

  const handlePointerDown = (e: React.PointerEvent) => {
    e.preventDefault()
    isDragging.current = true
    document.body.style.cursor = direction === 'horizontal' ? 'col-resize' : 'row-resize'
    document.body.style.userSelect = 'none'
  }

  useEffect(() => {
    const handlePointerMove = (e: PointerEvent) => {
      if (!isDragging.current || !containerRef.current) return
      const containerRect = containerRef.current.getBoundingClientRect()
      
      let newSize = size
      if (direction === 'horizontal') {
        newSize = e.clientX - containerRect.left
      } else {
        newSize = e.clientY - containerRect.top
      }

      if (newSize < minSize) newSize = minSize
      if (maxSize && newSize > maxSize) newSize = maxSize
      
      const maxBound = direction === 'horizontal' ? containerRect.width : containerRect.height
      if (newSize > maxBound - minSize) newSize = maxBound - minSize

      setSize(newSize)
    }

    const handlePointerUp = () => {
      if (isDragging.current) {
        isDragging.current = false
        document.body.style.cursor = ''
        document.body.style.userSelect = ''
      }
    }

    window.addEventListener('pointermove', handlePointerMove)
    window.addEventListener('pointerup', handlePointerUp)
    return () => {
      window.removeEventListener('pointermove', handlePointerMove)
      window.removeEventListener('pointerup', handlePointerUp)
    }
  }, [size, direction, minSize, maxSize])

  const isH = direction === 'horizontal'

  return (
    <div 
      ref={containerRef} 
      className={`flex ${isH ? 'flex-row' : 'flex-col'} w-full h-full overflow-hidden ${className}`}
    >
      <div 
        style={{ [isH ? 'width' : 'height']: `${size}px` }} 
        className="flex-shrink-0 overflow-hidden"
      >
        {first}
      </div>
      
      {/* 分割線（ドラッグ領域） */}
      <div
        onPointerDown={handlePointerDown}
        className={`flex-shrink-0 relative group transition-colors duration-150 z-30
          ${isH ? 'w-[5px] cursor-col-resize hover:bg-accent-green/50 bg-border' : 'h-[5px] cursor-row-resize hover:bg-accent-green/50 bg-border'}
        `}
      >
        {/* 当たり判定拡張 */}
        <div className={`absolute inset-0 ${isH ? '-left-1 -right-1' : '-top-1 -bottom-1'}`} />
      </div>

      <div className="flex-1 min-w-0 min-h-0 overflow-hidden">
        {second}
      </div>
    </div>
  )
}
```

### Step 3: `SettingsPage.tsx` のアコーディオン化
`expandedSections` 状態変数を定義して各セクションの表示/非表示を制御する：
```tsx
const [expandedSections, setExpandedSections] = useState<ExpandedSectionsState>(() => {
  const stored = localStorage.getItem(LS_KEYS.SETTINGS_SECTIONS_EXPANDED)
  return stored ? JSON.parse(stored) : {
    prompts: true,
    mappingMasters: true,
    longPress: true,
    workflows: true,
    connections: true,
  }
})

useEffect(() => {
  localStorage.setItem(LS_KEYS.SETTINGS_SECTIONS_EXPANDED, JSON.stringify(expandedSections))
}, [expandedSections])

const toggleSection = (key: keyof ExpandedSectionsState) => {
  setExpandedSections(prev => ({ ...prev, [key]: !prev[key] }))
}
```

主要5つのセクション（LLM System Prompts, Node Mapping Masters, 長押し感度設定, ワークフロー管理, Connections）をそれぞれ以下のようにアコーディオンのヘッダー・ボディ構造に変更し、ChevronDown アイコンと回転アニメーション（`rotate-180`）を加える：
```tsx
<section className="border border-[rgba(255,255,255,0.06)] bg-surface-2/20 rounded-xl overflow-hidden">
  <button 
    type="button" 
    onClick={() => toggleSection('prompts')}
    className="w-full flex items-center justify-between px-4 py-3.5 bg-surface hover:bg-white/5 transition-colors text-left"
  >
    <div className="flex items-center gap-2">
      <Wand2 size={15} className="text-[#7C6FDD]" />
      <h2 className="text-sm font-semibold text-zinc-300">LLM System Prompts</h2>
      <span className="text-xs text-zinc-600">— DBに保存（再起動不要）</span>
    </div>
    <ChevronDown size={16} className={`text-zinc-500 transition-transform duration-200 ${expandedSections.prompts ? 'rotate-180' : ''}`} />
  </button>
  {expandedSections.prompts && (
    <div className="p-4 border-t border-[rgba(255,255,255,0.06)] bg-surface/30">
      {/* 既存のフォーム等コンテンツ */}
    </div>
  )}
</section>
```
※他の4つのセクション（`mappingMasters`, `longPress`, `workflows`, `connections`）も同様にアコーディオンに変更する。
