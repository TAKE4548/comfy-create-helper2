# [新規関数] サービス識別子の正規化
function Get-NormalizedServiceName ([string]$name) {
    switch ($name.ToLower()) {
        { $_ -in "fe", "frontend" } { return "frontend" }
        { $_ -in "be", "backend" } { return "backend" }
        { $_ -in "comfy", "wsl" } { return "comfy" }
        default { return "" }
    }
}

# [改修] Stop-ComfyUIProcess
# Windows 側の Stop-PortProcesses 呼び出しを削除し、WSL 仮想マシン内部のプロセスのみを停止する
function Stop-ComfyUIProcess {
    $wslPrefix = Get-WslDistroPrefix
    try {
        Invoke-Expression "wsl $($wslPrefix)bash -c `"pkill -f 'python.*main.py'`" 2>`$null"
    } catch {}

    $closed = Wait-For-Port-Closed -port $cfg.wsl.port -timeoutSec 5 -serviceName "WSL ComfyUI"
    if (-not $closed) {
        try {
            Invoke-Expression "wsl $($wslPrefix)bash -c `"pkill -9 -f 'python.*main.py'`" 2>`$null"
            Start-Sleep -Seconds 2
        } catch {}
    }

    # クリーンアップ: ウィンドウ起動時の残骸 cmd.exe プロセスを回収
    try {
        Get-Process cmd -ErrorAction SilentlyContinue | Where-Object { $_.MainWindowTitle -like "*ComfyUI (WSL2)*" } | ForEach-Object {
            & taskkill.exe /PID $_.Id /T /F 2>$null
        }
    } catch {}

    # 注意: Windows 側の Stop-PortProcesses $cfg.wsl.port は実行しない。
    # WSL2 Mirrored モードでは Windows 側の 8188 は WSL ホストプロキシが LISTEN しており、
    # それを taskkill すると WSL2 との通信ブリッジが破損するため。
}

# [改修] Stop-AllServices
# 停止後に全サービスのポートクローズ完了を待機する
function Stop-AllServices {
    Write-Host "Stopping all services..." -ForegroundColor Yellow
    Stop-ComfyUIProcess
    Stop-BackendProcess
    Stop-FrontendProcess

    Write-Host "Waiting for ports to close..." -ForegroundColor Gray
    Wait-For-Port-Closed -port $cfg.wsl.port -timeoutSec 5 -serviceName "WSL ComfyUI" | Out-Null
    Wait-For-Port-Closed -port $cfg.backend.port -timeoutSec 5 -serviceName "Backend" | Out-Null
    Wait-For-Port-Closed -port $cfg.frontend.port -timeoutSec 5 -serviceName "Frontend" | Out-Null
}

# [改修] Restart-ServiceByName
function Restart-ServiceByName ([string]$name, [bool]$asBackground) {
    $normalized = Get-NormalizedServiceName $name
    switch ($normalized) {
        "comfy" {
            Write-Host "Restarting WSL2 ComfyUI..." -ForegroundColor Yellow
            Stop-ComfyUIProcess
            Wait-For-Port-Closed -port $cfg.wsl.port -timeoutSec 5 -serviceName "WSL ComfyUI" | Out-Null
            Start-ComfyUIProcess $asBackground
            Wait-For-Service "comfy" "WSL ComfyUI" | Out-Null
        }
        "backend" {
            Write-Host "Restarting Backend..." -ForegroundColor Yellow
            Stop-BackendProcess
            Wait-For-Port-Closed -port $cfg.backend.port -timeoutSec 5 -serviceName "Backend" | Out-Null
            Start-BackendProcess $asBackground
            Wait-For-Service "backend" "Backend" | Out-Null
        }
        "frontend" {
            Write-Host "Restarting Frontend..." -ForegroundColor Yellow
            Stop-FrontendProcess
            Wait-For-Port-Closed -port $cfg.frontend.port -timeoutSec 5 -serviceName "Frontend" | Out-Null
            Start-FrontendProcess $asBackground
            Wait-For-Service "frontend" "Frontend" | Out-Null
        }
        default {
            Write-Host "Unknown service: $name (choose: comfy/wsl, backend/be, frontend/fe)" -ForegroundColor Red
        }
    }
    Show-Status
}

# [改修] Main Dispatch switch ($Action.ToLower())
    "start" {
        if ($Service) {
            $normalized = Get-NormalizedServiceName $Service
            switch ($normalized) {
                "comfy" { Start-ComfyUIProcess $Background; Wait-For-Service "comfy" "WSL ComfyUI" | Out-Null; Show-Status }
                "backend" { Start-BackendProcess $Background; Wait-For-Service "backend" "Backend" | Out-Null; Show-Status }
                "frontend" { Start-FrontendProcess $Background; Wait-For-Service "frontend" "Frontend" | Out-Null; Show-Status }
                default { Write-Host "Unknown service: $Service (choose: comfy/wsl, backend/be, frontend/fe)" -ForegroundColor Red }
            }
        } else {
            Start-AllServices $Background
        }
    }
    "stop" {
        if ($Service) {
            $normalized = Get-NormalizedServiceName $Service
            switch ($normalized) {
                "comfy" { Stop-ComfyUIProcess; Show-Status }
                "backend" { Stop-BackendProcess; Show-Status }
                "frontend" { Stop-FrontendProcess; Show-Status }
                default { Write-Host "Unknown service: $Service (choose: comfy/wsl, backend/be, frontend/fe)" -ForegroundColor Red }
            }
        } else {
            Stop-AllServices
            Show-Status
        }

import socket

# [新規関数] ローカル LAN IP 取得
def get_local_ip():
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
            s.connect(("8.8.8.8", 80))
            return s.getsockname()[0]
    except Exception:
        try:
            return socket.gethostbyname(socket.gethostname())
        except Exception:
            return "127.0.0.1"

# [新規クラス] マルチスレッドTCPServer
class ThreadedTCPServer(socketserver.ThreadingMixIn, socketserver.TCPServer):
    daemon_threads = True
    allow_reuse_address = True

# [改修] ManagerHandler.do_POST
    def do_POST(self):
        if self.path == "/api/action":
            length = int(self.headers.get("Content-Length", 0))
            payload = json.loads(self.rfile.read(length).decode("utf-8") or "{}")
            action = payload.get("action", "status")
            service = payload.get("service", "")
            cmd = ["powershell.exe", "-ExecutionPolicy", "Bypass", "-File", str(PS_SCRIPT), "-Action", action]
            if action in ["start", "restart", "reboot-service"]:
                cmd.append("-Background")
            if service:
                cmd += ["-Service", service]
            res = subprocess.run(cmd, capture_output=True, text=True)
            self._send_json({
                "status": "ok",
                "action": action,
                "output": (res.stdout or "") + (res.stderr or "")
            })

# [改修] run_server
def run_server():
    server_address = ("", PORT)
    local_ip = get_local_ip()
    with ThreadedTCPServer(server_address, ManagerHandler) as httpd:
        print(f"Server Manager Web UI running at:")
        print(f"  Local:   http://localhost:{PORT}")
        if local_ip != "127.0.0.1":
            print(f"  Network: http://{local_ip}:{PORT}")
        try:
            httpd.serve_forever()

            print("\nShutting down server.")

## 3. 実装要件 (Implementation Logic)

### 3.1 `tools/server-manager/server-manager.ps1`
1. `Get-NormalizedServiceName`:
   - 入力文字列（小文字化）を受け取り、`fe`/`frontend` は `"frontend"`、`be`/`backend` は `"backend"`、`comfy`/`wsl` は `"comfy"` を返す。未知の場合は空文字を返す。
2. `Action` switch文（`start`, `stop`）および `Restart-ServiceByName`:
   - `$Service` / `$name` を `Get-NormalizedServiceName` で変換した結果に基づいて分岐処理を行う。
   - 不正なサービス名の場合は赤字でエラーメッセージを出力し、何もしない。
3. `Stop-AllServices`:
   - 各停止処理を実行後、`Wait-For-Port-Closed` で 8188, 8000, 5173 の3ポートすべてが閉じるのを待機する。
4. `Stop-ComfyUIProcess`:
   - `Stop-PortProcesses $cfg.wsl.port "ComfyUI"` を削除。
   - `pkill` の正規表現を `'python.*main.py'` に統一（ポート指定に依存しない）。

### 3.2 `tools/server-manager/server_api.py`
1. `get_local_ip()` を追加し、起動時ログにローカルURLとネットワークURLを出力。
2. `ThreadedTCPServer` クラスを定義（`socketserver.ThreadingMixIn`, `socketserver.TCPServer` を継承、`daemon_threads = True`）。
3. `run_server()` で `socketserver.TCPServer` の代わりに `ThreadedTCPServer` をインスタンス化。
4. `do_POST` の `/api/action` 処理において、`action` が `start`, `restart`, `reboot-service` の場合はコマンド引数に `"-Background"` を追加。
5. `HTML_TEMPLATE` 内のヘッダーに `LAN: http://<local_ip>:8899` バッジを表示し、JavaScript に `setControlsDisabled(disabled)` を追加して `triggerAction` の実行中に全ボタンを `disabled` にする。

### 3.3 `tools/server-manager/allow_firewall_8899.bat` (新規)
1. UAC 管理者権限昇格を自動実行。
2. PowerShell の `New-NetFirewallRule` でポート 8899 受信を許可。
