### 2.3 変更後の実装コード (`useGenerateDisplayState.ts`)
```typescript
  const rawImages = useMemo(() => {
    return filterOwnSession ? batchImages : comfyHistoryImages
  }, [filterOwnSession, batchImages, comfyHistoryImages])

  const { data: displayImages = rawImages } = useQuery<string[]>({
    queryKey: ['generateDisplayImages', { filterOwnSession: !!filterOwnSession }],
    queryFn: () => {
      const storeState = useGenerationStore.getState()
      return filterOwnSession ? storeState.batchImages : storeState.comfyHistoryImages
    },
    // rawImages が空配列のときは initialData を設定せず、PersistQueryClientProvider によるキャッシュ復元値を保持させる
    initialData: rawImages.length > 0 ? rawImages : undefined,
    staleTime: Infinity,
  })

  // Synchronize query cache if store rawImages changes (マウント初期の空配列によるキャッシュ消去を防止)
  useEffect(() => {
    if (rawImages.length === 0) return

    queryClient.setQueryData<string[]>(
      ['generateDisplayImages', { filterOwnSession: !!filterOwnSession }],
      (old) => {
        if (!old || old.length === 0) return rawImages
        const oldSet = new Set(old)
        const appended = rawImages.filter((x) => !oldSet.has(x))
        if (appended.length > 0) {
          return [...old, ...appended]
        }
        return old
      }
    )
  }, [rawImages, filterOwnSession, queryClient])

  // ... (addHistoryImage) ...

  // Listen to comfy-job-event for real-time image completion & history sync
  useEffect(() => {
    const handleJobEvent = (e: Event) => {
      const customEvent = e as CustomEvent<{
        type?: string
        images?: string[]
        isOwn?: boolean
        allImages?: string[]
        ownImages?: string[]
      }>
      if (customEvent.detail?.type === 'executed' && customEvent.detail.images) {
        for (const imgUrl of customEvent.detail.images) {
          addHistoryImage(imgUrl, customEvent.detail.isOwn ?? true)
        }
      } else if (customEvent.detail?.type === 'history-fetched') {
        const allImages = customEvent.detail.allImages
        const ownImages = customEvent.detail.ownImages
        if (allImages) {
          queryClient.setQueryData<string[]>(
            ['generateDisplayImages', { filterOwnSession: false }],
            allImages
          )
        }
        if (ownImages) {
          queryClient.setQueryData<string[]>(
            ['generateDisplayImages', { filterOwnSession: true }],
            ownImages
          )
        }
      }
    }
    window.addEventListener('comfy-job-event', handleJobEvent)
    return () => {
      window.removeEventListener('comfy-job-event', handleJobEvent)
    }
  }, [addHistoryImage, queryClient])
```

  describe('REQ-253: reload and history-fetched race condition prevention', () => {
    it('preserves persisted query cache when rawImages is initially empty on reload', () => {
      // 事前にキャッシュにデータが存在する状態（PersistQueryClientProvider 復元シミュレーション）
      queryClient.setQueryData(
        ['generateDisplayImages', { filterOwnSession: false }],
        ['persisted-1.png', 'persisted-2.png']
      )

      // ストアはリロード直後の空状態
      useGenerationStore.setState({
        batchImages: [],
        comfyHistoryImages: [],
        filterOwnSession: false,
      })

      const { result } = renderHook(() => useGenerateDisplayState({ selectedWorkflowId: 1 }), { wrapper })

      // キャッシュが空配列に上書きされず、復元された値が即座に表示されること
      expect(result.current.displayImages).toEqual(['persisted-1.png', 'persisted-2.png'])
    })

    it('directly updates query cache when history-fetched event carries allImages and ownImages', async () => {
      useGenerationStore.setState({
        batchImages: [],
        comfyHistoryImages: [],
        filterOwnSession: false,
      })

      const { result } = renderHook(() => useGenerateDisplayState({ selectedWorkflowId: 1 }), { wrapper })

      await act(async () => {
        window.dispatchEvent(
          new CustomEvent('comfy-job-event', {
            detail: {
              type: 'history-fetched',
              allImages: ['fetched-1.png', 'fetched-2.png'],
              ownImages: ['fetched-1.png'],
            },
          })
        )
      })

      expect(result.current.displayImages).toEqual(['fetched-1.png', 'fetched-2.png'])

      const ownSessionCache = queryClient.getQueryData<string[]>([
        'generateDisplayImages',
        { filterOwnSession: true },
      ])
      expect(ownSessionCache).toEqual(['fetched-1.png'])
    })
  })

## 3. 実装要件 (Implementation Logic)
- [x] `useQuery` において、`initialData` を `rawImages.length > 0 ? rawImages : undefined` に変更し、初期空配列が永続化キャッシュを上書きしないようにする。
- [x] `queryFn` を `() => { const storeState = useGenerationStore.getState(); return filterOwnSession ? storeState.batchImages : storeState.comfyHistoryImages }` に変更し、古いクロージャによる空配列返却を防止する。
- [x] `useEffect` の冒頭に `if (rawImages.length === 0) return` を追加し、リロード直後にキャッシュを `[]` で破壊する処理を根絶する。
- [x] `comfy-job-event` の `history-fetched` 受信時、従来の `queryClient.invalidateQueries` を廃止し、`customEvent.detail.allImages` / `customEvent.detail.ownImages` を直接 `queryClient.setQueryData` で安全かつ確定的に設定する。
- [x] `useGenerateDisplayState.test.tsx` に以下の検証テストを追加する:
  1. ストア初期値が空配列でも、既存キャッシュが存在する場合に `displayImages` が消去されず維持されること。
  2. `history-fetched` イベント受信時に `allImages` / `ownImages` から直接 TanStack Query キャッシュが更新されること。
