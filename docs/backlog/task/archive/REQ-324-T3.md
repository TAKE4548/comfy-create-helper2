# REQ-324-T3: API Specialized Domains Modularization (speech.ts, translation.ts, settings.ts)

## 1. 概要と目的
`frontend/src/api/client.ts` 内の特化サービス API 群をドメイン単位で分割する。
- 音声・セリフ・フキダシ・ストリーム生成系: `domains/speech.ts`
- プロンプト翻訳・LLM 連携系: `domains/translation.ts`
- システム設定・監査ログ・ComfyUI 通信・VRAM 状態系: `domains/settings.ts`

## 2. ターゲット識別
- **対象ファイル**:
  - `frontend/src/api/domains/speech.ts` (新規作成)
  - `frontend/src/api/domains/translation.ts` (新規作成)
  - `frontend/src/api/domains/settings.ts` (新規作成)
- **対象シンボル**:
  - `sceneSpeechApi`, `speechBatchApi`, `speechBubblePresetsApi`, `speechTonePresetsApi`, `fontsApi`
  - `translateApi`, `llmPromptsApi`
  - `settingsApi`, `configApi`, `auditApi`, `comfyApi`, `VramSettings`

export const sceneSpeechApi = {
  generateStream: async (
    payload: SceneDraftRequest,
    options?: SceneSpeechStreamOptions,
  ): Promise<SceneDraftResponse> => {
    const signal = options?.signal
    if (signal?.aborted) throw abortError(signal)
    const response = await fetch(`${BASE_URL}/speech/scene-draft/stream`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Accept: 'application/x-ndjson' },
      body: JSON.stringify(payload),
      signal,
    })
    // ... reader loop & consumeLine
  }
}

// translation.ts 対象:
export const translateApi = {
  translate: (payload: PromptTranslateRequest) =>
    request<PromptTranslateResponse>('/translate/prompt', { method: 'POST', body: JSON.stringify(payload) }),
  test: (payload: TranslateTestRequest) =>
    request<TranslateTestResponse>('/translate/test', { method: 'POST', body: JSON.stringify(payload) }),
  meta: () => request<TranslateMeta>('/translate/meta'),
}

// settings.ts 対象:
export const settingsApi = {
  getVram: () => request<VramSettings>('/settings/vram'),
  updateVram: (data: Partial<VramSettings>) =>
    request<VramSettings>('/settings/vram', { method: 'PATCH', body: JSON.stringify(data) }),
}
