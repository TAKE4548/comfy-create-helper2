  const cancelUnconfirmed = async (jobId: string) => {
    if (busyRef.current) return
    busyRef.current = true; setBusy(true); setResultMessage(null)
    try {
      const result = await cancelJob(jobId)
      setResultMessage(result.outcome === 'cancelled'
        ? '取消を確認しました。'
        : `取消を確認できません（状態未確認のままです）: ${result.message}`)
    } catch (error) {
      setResultMessage(error instanceof Error ? error.message : '取消結果を確認できませんでした。')
    } finally { busyRef.current = false; setBusy(false) }
  }

  const confirmRelease = async () => {
    const ids = releaseConfirmation
    setReleaseConfirmation(null)
    if (!ids || ids.length === 0 || busyRef.current) return
    busyRef.current = true; setBusy(true); setResultMessage(null)
    try {
      const operation = await releaseUnconfirmed(ids)
      const released = operation.items.filter((item) => item.outcome === 'released' || item.outcome === 'already_released').length
      const notReleased = operation.items.length - released
      setResultMessage(`解除しました: ${released}件（上流の取消確認ではありません）` +
        (notReleased > 0 ? `／解除できなかった記録: ${notReleased}件（状態が更新されたか、状態未確認ではありません）` : ''))
    } catch (error) {
      setResultMessage(error instanceof Error ? error.message : '解除結果を確認できませんでした。')
    } finally { busyRef.current = false; setBusy(false) }
  }

## 5. 追加するテスト
**`JobManagementModal.test.tsx`**
- `it('[REQ-335/AC-1] shows 状態未確認 section separately from 待機中 and never labels it 待機中')`: `mocks.state.generationSnapshot.jobs` に unconfirmed(prompt_id=null) を1件、`queuePending=[]` → 見出し「待機中のキュー (0)」と「状態未確認の記録 (1)」が別、行内に「待機中」文言なし、failure.reason 表示。session filter ON で他session行が出ない。
- `it('[REQ-335/AC-2] single cancel of an unconfirmed row shows the unconfirmed message and keeps the row')`: `cancelJob` が `{outcome:'unconfirmed', message:'...'}` → 「取消を確認できません（状態未確認のままです）」、行が残る。
- `it('[REQ-335/AC-2] bulk result with unconfirmed leftovers keeps the modal open with confirmed/unconfirmed counts')`: operation.items に cancelled 1 + unconfirmed 1 → `onClose` 未呼出、「確認済み: 1件、未確認: 1件」と解除導線文言。
- `it('[REQ-335/AC-3] release requires confirmation that states it is not an upstream cancellation and reports per-item outcome')`: 「記録を解除」→ dialog「解除確認」に「ComfyUI側の取消ではありません」。「戻る」で `releaseUnconfirmed` 未呼出。確認で呼出引数が選択ID、結果 `released 1 / not_released 1` → 「解除しました: 1件（上流の取消確認ではありません）／解除できなかった記録: 1件」。busy 中の二重実行なし。released 行が「解除済みの記録」に出る。
**`GenerationProgress.test.tsx`**（Desktop/Mobile 両方、`useGenerationStore.setState({ generationSnapshot, ... })`）
- `it('[REQ-335/AC-1] unconfirmed-only normal group is not shown as waiting or in progress and the queue button shows the unconfirmed count')`: unconfirmed のみ → `displayProgressActive` 相当のバーが非アクティブ、「待機中」「準備中...」が出ない、`data-testid="unconfirmed-count-chip"` が「状態未確認 1件」。
- `it('[REQ-335/AC-1] completed jobs plus an unconfirmed job still show 完了 with the unconfirmed chip')`: succeeded 1 + unconfirmed 1 → 進捗100%/完了、チップ併記。
