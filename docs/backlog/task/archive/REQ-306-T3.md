// frontend/src/components/TagChipEditor/useChipParser.ts:145-172
export function parseTagWeight(text: string): ParsedTagWeight {
  const trimmed = text.trim();
  const match = trimmed.match(/^\((.+):([0-9.]+)\)$/);
  if (match) {
    const baseText = match[1].trim();
    const weightNum = parseFloat(match[2]);
    if (!isNaN(weightNum)) {
      return { baseText, weight: weightNum };
    }
  }
  return { baseText: trimmed, weight: null };
}

export function formatTagWeight(baseText: string, weight: number | null): string {
  const trimmedBase = baseText.trim();
  if (weight === null || isNaN(weight) || weight === 1) {
    return trimmedBase;
  }
  const rounded = parseFloat(weight.toFixed(2));
  if (rounded === 1) {
    return trimmedBase;
  }
  return `(${trimmedBase}:${rounded})`;
}

## 3. 実装要件 (Implementation Logic)
- [ ] `useChipParser.ts`:
  - `parseTagWeight(text: string)` の正規表現を `^\((.+):([0-9.]+)\)\.?$` に拡張し、末尾ピリオド付きの `(tag:x).` や括弧内ピリオドの `(tag.:x)` の双方に対応。
  - ピリオド終端フラグ（`hasPeriod = trimmed.endsWith('.') || rawBase.endsWith('.')`）を判定し、`baseText` にはピリオド付き（または保持された）ベーステキストを返しつつ、内部の不要な末尾ピリオドはサニタイズする。
  - `formatTagWeight(baseText: string, weight: number | null, isAnimaMode: boolean = false)` を実装:
    - `hasPeriod = trimmedBase.endsWith('.') || isAnimaMode`
    - `cleanBase = trimmedBase.replace(/\.+$/, '')`
    - `weight === null || isNaN(weight) || weight === 1 || rounded === 1` の場合: `hasPeriod ? `${cleanBase}.` : cleanBase` を返却。
    - それ以外の場合: `hasPeriod ? `(${cleanBase}:${rounded}).` : `(${cleanBase}:${rounded})`` を返却。

  - `handleStepWeight`, `handleWeightSubmit`, `handleWordSubmit` の各 `formatTagWeight` 呼び出し箇所で、Props の `isAnimaMode` を第3引数として渡す。
