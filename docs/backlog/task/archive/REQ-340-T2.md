#### 3. 変更箇所のサージカルスニペット
```typescript
// インライン編集とコピペ処理
const handlePaste = (e: React.ClipboardEvent<HTMLInputElement>) => {
  e.preventDefault()
  const text = e.clipboardData.getData('text')
  const newTags = text
    .split(/[\r\n,]+/)
    .map((s) => s.replace(/["']/g, '').trim())
    .filter((s) => s.length > 0)
  const merged = Array.from(new Set([...patterns, ...newTags]))
  onChange(merged)
}
```
