```python
async def call_llm(system_prompt: str, user_prompt: str, timeout: float = 180.0) -> str:
    """Call the LLM server (OpenAI-compatible API) asynchronously."""
    payload = {
        "model": settings.llm_model,
        "messages": [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_prompt},
        ],
        "stream": False,
        "temperature": 0.2,
    }
    url = f"{settings.llm_base_url}/chat/completions"
    async with httpx.AsyncClient(timeout=timeout) as client:
        response = await client.post(url, json=payload)
        response.raise_for_status()
        data = response.json()
        return data["choices"][0]["message"]["content"]
```

### 2.4 新規作成ファイルの実装仕様

#### `backend/app/services/llm_service.py` — `run_scene_generate()` 追加

**追加するシステムプロンプト（デフォルト）:**
```
You are a scene description expert for AI image generation.
Given a scene summary in Japanese and a list of candidate fields with their current values,
analyze the scene and generate English prompts ONLY for the fields that are relevant to the scene.
You decide which fields to update based on the scene's natural implication.

Rules:
1. Output MUST be a valid JSON array — no markdown, no explanation outside JSON.
2. Only include fields you want to change; omit unchanged fields.
3. Each field value must be comma-separated English prompt tags (no Japanese, no full sentences).
4. Do NOT hallucinate fields that are not listed in target_fields.
```

**ユーザープロンプト構成:**
```
Target Fields and Current Values:
{target_fields_with_values}

Generate diff_params for each scene below:
{scenes_json}

Return ONLY a JSON array in this format:
[{"scene_id": "...", "diff_params": {"FieldName": "english, tags"}}]
```

**パースロジック（JSON直接パース、XMLタグなし）:**
```python
import json, re

def _parse_scene_generate(raw: str) -> list[dict]:
    """Extract JSON array from LLM response, stripping markdown fences if present."""
    text = raw.strip()
    # Remove ```json ... ``` fences if present
    match = re.search(r'```(?:json)?\s*(\[.*?\])\s*```', text, re.DOTALL)
    if match:
        text = match.group(1)
    else:
        # Find first '[' to last ']'
        start = text.find('[')
        end = text.rfind(']')
        if start != -1 and end != -1:
            text = text[start:end+1]
    return json.loads(text)
```

**`run_scene_generate()` のシグネチャ:**

#### `frontend/src/pages/GeneratePage/hooks/useSceneGenerator.ts` — 新規

```typescript
import { llmPromptsApi, workflowPresetsApi } from '../../../api/client'
import type { SceneItem, SceneDiffParams } from '../../../api/types'

interface UseSceneGeneratorParams {
  workflowId: number
  baseParamValues: Record<string, string>
}

export function useSceneGenerator({ workflowId, baseParamValues }: UseSceneGeneratorParams) {

  const generateDrafts = async (
    targetFields: string[],
    scenes: SceneItem[],
  ): Promise<SceneDiffParams[]> => {
    const baseParams: Record<string, string> = {}
    targetFields.forEach(f => { baseParams[f] = baseParamValues[f] ?? '' })

    return await llmPromptsApi.sceneGenerate({
      target_fields: targetFields,
      base_params: baseParams,
      scenes: scenes.map(s => ({ id: s.id, summary: s.summary })),
    })
  }

  const saveAsProfiles = async (
    scenes: SceneItem[],
    drafts: SceneDiffParams[],
  ): Promise<void> => {
    for (const draft of drafts) {
      const scene = scenes.find(s => s.id === draft.scene_id)
      const profileName = scene?.summary ?? `Scene ${draft.scene_id}`
      const finalParams = { ...baseParamValues, ...draft.diff_params }

      await workflowPresetsApi.create({
        workflow_id: workflowId,
        name: profileName,
        param_values: finalParams,
        applied_assets: {},
        batch_size: 1,
        seed_mode: 'random',
        seed: 42,
      })
    }
  }

  return { generateDrafts, saveAsProfiles }
}
```

**画面フロー:**
```
[input フェーズ]
  - textMappings を map してチェックボックス一覧 (role をキーに使用)
  - scenes 配列を map してテキストエリア一覧
  - シーン追加ボタン: scenes に { id: crypto.randomUUID(), summary: '' } を追加
  - シーン削除ボタン: 対象 id を filter で除去
  - 上へ/下へボタン: 配列の要素を swap
  - Generate Drafts ボタン:
      setIsDrafting(true)
      try {
        const result = await generateDrafts(selectedFields, scenes)
        setDrafts(result)
        setEditedDrafts(JSON.parse(JSON.stringify(result)))  // deep copy
        setPhase('draft')
      } catch(e) {
        setError(e.message)
      } finally {
        setIsDrafting(false)
      }

[draft フェーズ]
  - editedDrafts を map してシーンカード表示
  - 各カードに:
    - シーン概要（scenes から scene_id で lookup）
    - diff_params の各キーに対して label + textarea (編集可能)
      onChange: setEditedDrafts で該当フィールドを更新
  - 「← 戻る」ボタン: setPhase('input')
  - 「プロファイルを作成」ボタン:
      setIsSaving(true)
      try {
        await saveAsProfiles(scenes, editedDrafts)
        onSaved()
      } catch(e) {
        setError(e.message)
      } finally {
        setIsSaving(false)
      }
```

**注意事項:**
- `textMappings` が空の場合（テキスト系フィールドがない）はチェックボックスを表示せずメッセージを出す
- `selectedFields.length === 0` の場合は Generate Drafts ボタンを disabled にする
- `scenes.every(s => s.summary.trim() === '')` の場合も Generate Drafts ボタンを disabled にする
- モーダル背景クリックでも `onClose()` を呼ぶ
