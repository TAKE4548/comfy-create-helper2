# Task Card: REQ-248-T1

## Context & Responsibility
FastAPI アプリケーション定義から、Uvicorn サーバー起動や HTTP 取得を行わずに再現可能な `backend/openapi.json` を出力・検査する CLI スクリプト `scripts/export_openapi.py` を作成する。本スクリプト実行時にアプリの lifespan, DB 初期化, worker, 外部サービス起動等の副作用を生じさせない。また `--check` オプションによる非更新での差分検知（Contract Drift 検知）を実装し、単体テストを整備する (AC-1, AC-4)。

### Change Site: `scripts/export_openapi.py`
```python
import argparse
import difflib
import json
from pathlib import Path
import sys

def main() -> None:
    parser = argparse.ArgumentParser(description="Export or check FastAPI OpenAPI JSON without side effects.")
    parser.add_argument("--output", type=Path, default=Path("backend/openapi.json"), help="Output JSON path")
    parser.add_argument("--check", action="store_true", help="Check if output is up-to-date without modifying file")
    parser.add_argument("--indent", type=int, default=2, help="JSON indentation level")
    args = parser.parse_args()

    # Import app without triggering lifespan
    sys.path.insert(0, str(Path("backend").resolve()))
    from app.main import app

    schema = app.openapi()
    generated_json = json.dumps(schema, indent=args.indent, ensure_ascii=False) + "\n"

    target_path = args.output
    if args.check:
        if not target_path.exists():
            print(f"Error: Target file {target_path} does not exist.")
            sys.exit(1)
        current_json = target_path.read_text(encoding="utf-8")
        if current_json != generated_json:
            print(f"Drift detected in {target_path}!")
            diff = difflib.unified_diff(
                current_json.splitlines(), generated_json.splitlines(),
                fromfile="current", tofile="generated", lineterm=""
            )
            print("\n".join(diff))
            sys.exit(1)
        print(f"OpenAPI schema {target_path} is up-to-date.")
        sys.exit(0)
    else:
        target_path.parent.mkdir(parents=True, exist_ok=True)
        target_path.write_text(generated_json, encoding="utf-8", newline="\n")
        print(f"Successfully exported OpenAPI JSON to {target_path}")

if __name__ == "__main__":
    main()
```

### Change Site: `backend/tests/unit/test_export_openapi.py`
```python
import json
import subprocess
import sys
from pathlib import Path

def test_req248_ac1_export_openapi_no_side_effects(tmp_path: Path) -> None:
    """[REQ-248/AC-1] export_openapi.py exports valid OpenAPI JSON without starting lifespan/DB/workers."""
    out_file = tmp_path / "test_openapi.json"
    cmd = [sys.executable, "scripts/export_openapi.py", "--output", str(out_file)]
    res = subprocess.run(cmd, capture_output=True, text=True, check=True)
    assert out_file.exists()
    data = json.loads(out_file.read_text(encoding="utf-8"))
    assert data.get("openapi") == "3.1.0"
    assert "/api/workflows" in data.get("paths", {})

def test_req248_ac1_export_openapi_idempotent(tmp_path: Path) -> None:
    """[REQ-248/AC-1] Running export twice on identical codebase produces identical bytes."""
    out1 = tmp_path / "out1.json"
    out2 = tmp_path / "out2.json"
    subprocess.run([sys.executable, "scripts/export_openapi.py", "--output", str(out1)], check=True)
    subprocess.run([sys.executable, "scripts/export_openapi.py", "--output", str(out2)], check=True)
    assert out1.read_bytes() == out2.read_bytes()

def test_req248_ac4_export_openapi_check_drift_detection(tmp_path: Path) -> None:
    """[REQ-248/AC-4] export_openapi.py --check exits 1 on schema drift without modifying file."""
    out_file = tmp_path / "openapi.json"
    subprocess.run([sys.executable, "scripts/export_openapi.py", "--output", str(out_file)], check=True)
    # Check passes on match
    res_ok = subprocess.run([sys.executable, "scripts/export_openapi.py", "--output", str(out_file), "--check"])
    assert res_ok.returncode == 0
    # Introduce drift
    out_file.write_text('{"openapi": "3.1.0", "drift": true}\n', encoding="utf-8", newline="\n")
    res_drift = subprocess.run([sys.executable, "scripts/export_openapi.py", "--output", str(out_file), "--check"], capture_output=True, text=True)
    assert res_drift.returncode == 1
    assert "Drift detected" in res_drift.stdout
    # Verify file was not overwritten
    assert "drift" in out_file.read_text(encoding="utf-8")
```
