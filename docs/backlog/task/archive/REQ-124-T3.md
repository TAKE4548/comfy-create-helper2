```typescript
// client.ts L1-15 (抜粋)
const BASE_URL = '/api'

async function request<T>(path: string, options?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE_URL}${path}`, {
    headers: { 'Content-Type': 'application/json' },
    ...options,
  })
  if (!res.ok) throw new Error(`API Error: ${res.status}`)
  if (res.status === 204) return undefined as unknown as T
  return res.json()
}

export const mappingMastersApi = {
  list: () => request<MappingMaster[]>('/mapping-masters'),
  create: (data: MappingMasterCreate) => request<MappingMaster>('/mapping-masters', { method: 'POST', body: JSON.stringify(data) }),
  update: (id: number, data: MappingMasterCreate) => request<MappingMaster>(`/mapping-masters/${id}`, { method: 'PUT', body: JSON.stringify(data) }),
  delete: (id: number) => request<void>(`/mapping-masters/${id}`, { method: 'DELETE' }),
}
```

/**
