export interface PromptMetadataRead {
  prompt_id: string;
  workflow_id: number | null;
  param_values: Record<string, any>;
  applied_assets: Record<string, any> | null;
  created_at: string;

export const promptMetadataApi = {
  create: (data: {
    prompt_id: string;
    workflow_id?: number | null;
    param_values: Record<string, any>;
    applied_assets?: Record<string, any>;

### 2.2 変更仕様詳細

#### 1) `frontend/src/store/slices/jobQueueSlice.ts`
`queuePrompt` 内で `promptId` が発行された際、`isBatchIntermediate` の有無にかかわらず、無条件でローカルストレージに書き込むと同時に、`promptMetadataApi.create` を呼び出してサーバーDBへ永続化する。

```typescript
// L97-L140 周辺:
if (promptId) {
  localStorage.setItem(LS_KEYS.CURRENT_PROMPT_ID, promptId)
  const metaKey = `prompt-surgeon-meta-${promptId}`
  const promptMetadata = {
    paramValues: metadata?.paramValues ?? get().paramValues,
    appliedAssets: metadata?.appliedAssets ?? get().appliedAssets,
    timestamp: Date.now(),
  }
  localStorage.setItem(metaKey, JSON.stringify(promptMetadata))

  // サーバー側DBへ非同期保存 (端末間同期対応)
  promptMetadataApi.create({
    prompt_id: promptId,
    workflow_id: get().currentWorkflowId ?? undefined,
    param_values: promptMetadata.paramValues,
    applied_assets: promptMetadata.appliedAssets,
  }).catch((err) => {
    console.warn('[PromptMetadata] Failed to persist metadata to server:', err)
  })

  // 200件上限のLRUクリーンアップ (ローカルストレージ側)
  setTimeout(() => {
    try {
      const keys = Object.keys(localStorage).filter((k) => k.startsWith('prompt-surgeon-meta-'))
      if (keys.length > 200) {
        const items = keys.map((k) => {
          try {
            const val = JSON.parse(localStorage.getItem(k) || '{}')
            return { key: k, timestamp: val.timestamp || 0 }
          } catch {
            return { key: k, timestamp: 0 }
          }
        })
        items.sort((a, b) => a.timestamp - b.timestamp)
        const toDelete = items.slice(0, items.length - 200)
        toDelete.forEach((item) => localStorage.removeItem(item.key))
      }
    } catch {}
  }, 0)
}
```

#### 2) `frontend/src/pages/GeneratePage/hooks/useSnapshotActions.ts`
スナップショット保存時、画像URLの `t` からまず `localStorage` を調べ、存在しない場合（別端末やシークレットウィンドウ）は `promptMetadataApi.get(t)` を非同期取得してサーバーDBから復元する。
取得できた場合は `resolvedParamValues` と `resolvedAppliedAssets` を確実に更新し、端末の現在の生UI値によるデータ汚染を防止する。

```typescript
// L28-L45 周辺:
try {
  const urlObj = new URL(img, window.location.origin)
  const t = urlObj.searchParams.get('t')
  if (t) {
    let cached = localStorage.getItem(`prompt-surgeon-meta-${t}`)
    if (cached) {
      const parsed = JSON.parse(cached)
      if (parsed.paramValues && parsed.appliedAssets) {
        resolvedParamValues = parsed.paramValues
        resolvedAppliedAssets = parsed.appliedAssets
      }
    } else {
      // 端末跨ぎ・別ブラウザ対応: サーバーDBから取得
      try {
        const serverMeta = await promptMetadataApi.get(t)
        if (serverMeta && serverMeta.param_values) {
          resolvedParamValues = serverMeta.param_values
          resolvedAppliedAssets = serverMeta.applied_assets || {}
        }
      } catch (err) {
        console.warn(`[Snapshot] Could not fetch server metadata for prompt ${t}:`, err)
      }
    }
  }
} catch (e) {
  console.error('Failed to resolve metadata from image URL:', e)
}
```
