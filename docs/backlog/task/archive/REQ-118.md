- `bubble_detector.py`:
  ```python
  import cv2, numpy as np
  from pathlib import Path
  
  def detect_bubble(image_path: str, roi: dict | None = None) -> dict | None:
      """
      Returns {"x": int, "y": int, "w": int, "h": int, "confidence": float} or None.
      roi: {"x": int, "y": int, "w": int, "h": int} or None (全画像検索)
      """
      img = cv2.imread(str(image_path))
      if img is None:
          return None
      if roi:
          x, y, w, h = roi["x"], roi["y"], roi["w"], roi["h"]
          img = img[y:y+h, x:x+w]
          offset = (x, y)
      else:
          offset = (0, 0)
      
      gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
      blurred = cv2.GaussianBlur(gray, (5, 5), 0)
      edges = cv2.Canny(blurred, 50, 150)
      contours, _ = cv2.findContours(edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
      
      if not contours:
          return None
      
      # 白っぽい領域の輪郭から最大のものを選択
      best = None
      best_area = 0
      for cnt in contours:
          area = cv2.contourArea(cnt)
          if area < 1000:  # ノイズ除去
              continue
          x2, y2, w2, h2 = cv2.boundingRect(cnt)
          # 矩形らしさチェック (aspect ratio)
          aspect = w2 / max(h2, 1)
          if aspect < 0.3 or aspect > 10:
              continue
          # 内部の平均輝度チェック (吹き出しは白い)
          roi_region = gray[y2:y2+h2, x2:x2+w2]
          mean_bright = roi_region.mean()
          if mean_bright < 150:
              continue
          if area > best_area:
              best_area = area
              best = (x2 + offset[0], y2 + offset[1], w2, h2)
      
      if best is None:
          return None
      x3, y3, w3, h3 = best
      img_area = img.shape[0] * img.shape[1]
      confidence = min(best_area / img_area * 10, 1.0)
      return {"x": x3, "y": y3, "w": w3, "h": h3, "confidence": round(confidence, 2)}
  ```

- `speech_draft_service.py`:
  ```python
  import base64, json, logging
  from pathlib import Path
  import httpx
  from app.config import settings
  from app import models
  from sqlalchemy.orm import Session
  
  logger = logging.getLogger(__name__)
  
  async def call_vlm(image_base64: str, system_prompt: str, model: str) -> str:
      """VLM (multimodal) 呼び出し — LM Studio /chat/completions"""
      url = f"{settings.llm_base_url}/chat/completions"
      payload = {
          "model": model,
          "messages": [
              {"role": "system", "content": system_prompt},
              {"role": "user", "content": [
                  {"type": "image_url", "image_url": {"url": f"data:image/png;base64,{image_base64}"}},
                  {"type": "text", "text": "この画像の内容を日本語で簡潔に説明してください。"}
              ]}
          ],
          "stream": False,
          "max_tokens": 512
      }
      async with httpx.AsyncClient(timeout=60.0) as client:
          response = await client.post(url, json=payload)
          response.raise_for_status()
          return response.json()["choices"][0]["message"]["content"]
  
  def get_llm_prompt_content(db: Session, key: str, fallback: str = "") -> str:
      record = db.query(models.LlmPrompt).filter(models.LlmPrompt.key == key).first()
      return record.content if record and record.content.strip() else fallback
  
  async def generate_speech_draft(snapshot, db: Session) -> str:
      image_path = Path(settings.image_output_dir) / snapshot.image_path
      with open(image_path, "rb") as f:
          image_base64 = base64.b64encode(f.read()).decode()
      
      vlm_model = get_llm_prompt_content(db, "speech_draft_vlm_model", settings.llm_model)
      vlm_prompt = get_llm_prompt_content(db, "speech_draft_vlm_prompt",
          "この画像のキャラクターや状況を簡潔に日本語で説明してください。")
      
      image_summary = await call_vlm(image_base64, vlm_prompt, vlm_model)
      
      params = json.loads(snapshot.params) if snapshot.params else {}
      generation_prompt = str(params.get("positive", params.get("prompt", "")))
      
      llm_model = get_llm_prompt_content(db, "speech_draft_llm_model")
      llm_prompt = get_llm_prompt_content(db, "speech_draft_llm_prompt",
          "あなたはマンガのセリフ作家です。以下の画像の説明と生成プロンプトを元に、キャラクターが話しそうな自然なセリフを1~2文で生成してください。セリフのみを出力してください。")
      
      user_prompt = f"【画像の説明】\n{image_summary}\n\n【生成プロンプト】\n{generation_prompt}"
      
      from app.services.llm_request import call_llm
      draft = await call_llm(system_prompt=llm_prompt, user_prompt=user_prompt, model=llm_model or None, timeout=60.0)
      return draft.strip()
  ```

#### `SpeechBubbleEditorPage.tsx` コア実装 (react-konva):
- **依存**: `import { Stage, Layer, Image as KonvaImage, Rect, Text } from 'react-konva'`
- **State**:
  - `bubbles: { bbox: BubbleBox; text: string }[]` — 検出された全吹き出し
  - `selectedIdx: number | null` — 編集中の吹き出しインデックス
  - `fontSize: number`, `fontColor: string`

  - `isSaving: boolean`, `isDrafting: boolean`
- **レイアウト**: 2ペイン
  - **左ペイン** (60%): `<Stage>` — 画像に合わせた幅/高さを `useRef` で動的計算
    ```tsx
    <Stage width={stageW} height={stageH} onMouseDown={handleROIStart} onMouseMove={handleROIMove} onMouseUp={handleROIEnd}>
      <Layer>
        <KonvaImage image={img} width={stageW} height={stageH} />
        {bubbles.map((b, i) => (
          <>
            <Rect
              key={`rect-${i}`}
              x={b.bbox.x * scale} y={b.bbox.y * scale}
              width={b.bbox.w * scale} height={b.bbox.h * scale}
              stroke={i === selectedIdx ? '#facc15' : '#60a5fa'}
              strokeWidth={i === selectedIdx ? 3 : 1.5}
              fill={i === selectedIdx ? 'rgba(250,204,21,0.08)' : 'transparent'}
              onClick={() => setSelectedIdx(i)}
            />
            {b.text && <Text key={`text-${i}`} x={...} y={...} text={wrapText(b.text, b.bbox.w * scale, fontSize)} fontSize={fontSize} fill={fontColor} />}
          </>
        ))}
        {roiRect && <Rect {...roiRect} fill="rgba(239,68,68,0.2)" stroke="#ef4444" strokeWidth={1} dash={[4,4]} />}
      </Layer>
    </Stage>
    ```
  - **右ペイン** (40%): 制御エリア
    - 吹き出しリスト（番号 + confidence バッジ）、クリックで `selectedIdx` を切り替え
    - 選択中吹き出しのセリフ入力 `<textarea>`
    - フォントサイズスライダー / フォントカラーピッカー
    - 「ドラフト生成」ボタン (`isDrafting` ディセーブル)
    - 「保存」ボタン (`isSaving` ディセーブル)
    - 「手動補正モード」トグルボタン
- **ページロード時**: `useEffect` で `detectBubble(snapshotId)` を呼び出し `bubbles` を初期化
- **保存処理**:
  1. `stageRef.current.toBlob()` → FormData で `/api/snapshots/upload` へ POST
  2. 取得した `image_path` と元スナップショットの `workflow_id`/`params`/`source_snapshot_id` で `/api/snapshots` POST
  3. `useNavigate` で `/history` へ遷移
- **ページ内上部**: `← History へ戻る` リンク
- **CJK 折り返し**: `wrapText(text, maxWidth, fontSize)` ヘルパー関数を同ファイル内に実装
  ```typescript
  // canvas measureText を使い文字単位で折り返す
  function wrapText(text: string, maxWidth: number, fontSize: number): string {
    const canvas = document.createElement('canvas')
    const ctx = canvas.getContext('2d')!
    ctx.font = `${fontSize}px sans-serif`
    let line = '', result = ''
    for (const char of text) {
      if (ctx.measureText(line + char).width > maxWidth) {
        result += line + '\n'
        line = char
      } else { line += char }
    }
    return result + line
  }
  ```
