~~~text
def detect_bubble(
    image_path: str,
    roi: dict | None = None,
    min_area: int = 1000,
    min_brightness: int = 150
) -> dict | None:
    """
    Returns {"x": int, "y": int, "w": int, "h": int, "confidence": float} or None.
    roi: {"x": int, "y": int, "w": int, "h": int} or None (全画像検索)
    """
    img = cv2.imread(str(image_path))
    if img is None:
        return None

    # 1. カラー画像から直接「白い領域」の二値化マスクを作成 (BGRすべてがmin_brightness以上の領域)
    lower_white = cv2.Mat(np.array([min_brightness] * 3, dtype=np.float64))
    upper_white = cv2.Mat(np.array([255] * 3, dtype=np.float64))
    white_mask = cv2.inRange(img, lower_white, upper_white)

    # 2. 輪郭抽出 (RETR_EXTERNAL を用いて最も外側の輪郭のみを抽出。これで中の文字は無視される)
    contours, _ = cv2.findContours(white_mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

    if not contours:
        return None

~~~
