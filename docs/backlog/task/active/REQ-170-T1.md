export interface BubbleStyleConfig {

export function extractBubbleStyle(bubble: Partial<SpeechBubble>): Partial<BubbleStyleConfig> {
  const style: Partial<BubbleStyleConfig> = {}
  if (bubble.shape !== undefined) style.shape = bubble.shape
  if (bubble.orientation !== undefined) style.orientation = bubble.orientation
  if (bubble.fontFamily !== undefined) style.fontFamily = bubble.fontFamily
  if (bubble.fontSize !== undefined) style.fontSize = bubble.fontSize
  if (bubble.textColor !== undefined) style.textColor = bubble.textColor
  if (bubble.backgroundColor !== undefined) style.backgroundColor = bubble.backgroundColor
  if (bubble.borderColor !== undefined) style.borderColor = bubble.borderColor
  if (bubble.borderWidth !== undefined) style.borderWidth = bubble.borderWidth
  if (bubble.rotation !== undefined) style.rotation = bubble.rotation
  if (bubble.effects !== undefined) style.effects = bubble.effects
  return style
}

### 2.2 frontend/src/utils/speechBubbleRenderer.ts
```typescript
export function drawSpeechBubble(
  ctx: CanvasRenderingContext2D,
  bubble: SpeechBubble,
  isSelected = false,
  scaleX = 1,
  scaleY = 1,
  activeFontFamily?: string
): void {
  const rotation = bubble.rotation || 0
  const cx = (bubble.x + bubble.w / 2) * scaleX
  const cy = (bubble.y + bubble.h / 2) * scaleY

  ctx.save()
  if (rotation !== 0) {
    ctx.translate(cx, cy)
    ctx.rotate((rotation * Math.PI) / 180)
    ctx.translate(-cx, -cy)
  }

  drawBubbleShape(ctx, bubble, scaleX, scaleY)
  drawBubbleText(ctx, bubble, scaleX, scaleY, activeFontFamily)

  if (isSelected) {
    const bx = bubble.x * scaleX
    const by = bubble.y * scaleY
    const bw = bubble.w * scaleX
    const bh = bubble.h * scaleY

    ctx.save()
    ctx.strokeStyle = '#3b82f6'
    ctx.lineWidth = 1.5
    ctx.setLineDash([4, 4])
    ctx.strokeRect(bx, by, bw, bh)

    // 尻尾ハンドル
    if (bubble.tail) {
      const tx = bubble.tail.x * scaleX
      const ty = bubble.tail.y * scaleY

      ctx.setLineDash([])
      ctx.fillStyle = '#3b82f6'
      ctx.strokeStyle = '#ffffff'
      ctx.lineWidth = 2
      ctx.beginPath()
      ctx.arc(tx, ty, 6, 0, Math.PI * 2)
      ctx.fill()
      ctx.stroke()
    }

    // 四隅のリサイズハンドル
    ctx.setLineDash([])
    ctx.fillStyle = '#ffffff'
    ctx.strokeStyle = '#3b82f6'
    ctx.lineWidth = 2

    const hs = RESIZE_HANDLE_SIZE
    const handles = [
      { x: bx - hs / 2, y: by - hs / 2 }, // nw
      { x: bx + bw - hs / 2, y: by - hs / 2 }, // ne
      { x: bx + bw - hs / 2, y: by + bh - hs / 2 }, // se
      { x: bx - hs / 2, y: by + bh - hs / 2 }, // sw
    ]

    handles.forEach(h => {
      ctx.fillRect(h.x, h.y, hs, hs)
      ctx.strokeRect(h.x, h.y, hs, hs)
    })

    ctx.restore()
  }

  ctx.restore()
}
```
