import { request } from '../core/base'
import type { PromptTemplate, PromptTemplateCreate, PromptTemplateUpdate } from '../types'

export const promptTemplatesApi = {
  list: () => request<PromptTemplate[]>('/prompt-templates'),
  create: (data: PromptTemplateCreate) =>
    request<PromptTemplate>('/prompt-templates', { method: 'POST', body: JSON.stringify(data) }),
  update: (id: number, data: PromptTemplateUpdate) =>
    request<PromptTemplate>(`/prompt-templates/${id}`, { method: 'PUT', body: JSON.stringify(data) }),

    request<void>(`/prompt-templates/${id}`, { method: 'DELETE' }),

### 2. 分割・結合純粋関数エンジン (`frontend/src/utils/promptTemplateEngine.ts`)
```typescript
import type {
  PromptOutputGroup,
  PromptTemplateStructure,
  WorkflowPromptTemplateConfig,
} from '../api/types'
import { parsePromptToChips, chipsToPrompt, type TagChip } from '../components/TagChipEditor/useChipParser'

export const PROMPT_TEMPLATE_STATE_KEY = '__prompt_template_state_v1'

export interface PromptTemplateInstanceState {
  instanceId: string
  blocks: Record<string, TagChip[]> // includes enabled: true and enabled: false chips
}

export interface PromptTemplatePresetState {
  version: 1
  templateId: number | null
  sections: Record<string, { instances: PromptTemplateInstanceState[] }>
}

export interface SanitizedPromptTemplateRecord {
  version: 1
  template_id: number | null
  template_snapshot: PromptTemplateStructure
  sections: Array<{
    section_id: string
    section_name: string
    is_repeatable: boolean
    instances: Array<{
      instance_index: number
      blocks: Array<{
        block_id: string
        block_label: string
        value: string
      }>
    }>
  }>
}
```
- **必須提供関数シグネチャ**:
  - `isWorkflowTemplateActive(config?: WorkflowPromptTemplateConfig | null): boolean`: `output_groups.length > 0 && Boolean(config?.template_snapshot?.sections?.length || config?.template_id != null)`。
  - `normalizePromptSegment(raw: string): string`: 前後の空白・カンマを除去し、連続カンマ `, ,` を `, ` に正規化した文字列を返す（空またはカンマのみなら `""`）。
  - `formatEnabledBlockChips(chips: TagChip[] | undefined): string`: `enabled !== false` のチップのみから `chipsToPrompt` 相当の文字列を生成し、`normalizePromptSegment` を通して返す。すべて無効または空なら `""`。
  - `joinOutputGroup(structure: PromptTemplateStructure, state: PromptTemplatePresetState, group: PromptOutputGroup): string`:
    - `group.block_ids` を走査して連続する同一セクションのブロック ID チャンク `Array<{ section: PromptTemplateSection; blockIds: string[] }>` を構築する。
    - 非 Repeat セクションは `instances[0]` の該当 `blockIds` を順に評価。
    - Repeat セクションは `group.repeat_filter.mode === 'index'` なら `instances[(group.repeat_filter.index ?? 1) - 1]` のみ評価（**`AC-10`: 指定番号のインスタンスが未追加で存在しない場合はエラーなく空としてスキップし、余分なカンマを発生させない**）、`'all'` なら `instances` の `0..N-1` を登録順にループし、各インスタンスごとに `blockIds` を順に評価する（`AC-6`, `AC-10`）。
    - 非空のブロック文字列のみを `", "` で結合し、未入力のブロックやチップがすべて `enabled: false` で無効化されたブロックを自動的にスキップし、不要な連続カンマ `, ,` や先頭・末尾の余分なカンマを発生させずに結合文字列を生成する（`AC-6`, `AC-10`）。
  - `computeAllOutputGroupValues(structure: PromptTemplateStructure, state: PromptTemplatePresetState, outputGroups: PromptOutputGroup[]): Record<string, string>`: 各 `group.target_role` に対する結合結果マップを返す。
  - `serializePresetState(state: PromptTemplatePresetState): string` / `parsePresetState(raw: string | undefined, structure?: PromptTemplateStructure | null, fallbackRoleValues?: Record<string, string>, outputGroups?: PromptOutputGroup[]): PromptTemplatePresetState`:
    - `paramValues[PROMPT_TEMPLATE_STATE_KEY]` から `PromptTemplatePresetState` を復元（無効チップ `enabled: false` も完全保持、`AC-7`）。未設定の場合はデフォルト初期状態（各セクション1インスタンス）を生成する。
  - `buildSanitizedTemplateRecord(structure: PromptTemplateStructure, state: PromptTemplatePresetState, templateId?: number | null): SanitizedPromptTemplateRecord`:
    - `enabled: true` のチップのみから構成された有効文字列のみを持つ記録用オブジェクトを生成する（`enabled: false` チップは除外、`AC-8`）。
  - `restorePresetStateFromSanitizedRecord(record: SanitizedPromptTemplateRecord): PromptTemplatePresetState`:
    - スナップショット復元用：保存された有効文字列を `parsePromptToChips`（すべて `enabled: true`）でチップ化して分割状態を復元する（`AC-8`）。

### 3. 生成記録・メタデータビュー拡張 (`frontend/src/utils/generationRecord.ts` & `promptRecord.ts`)
- `GenerationRecord` インターフェース（`generationRecord.ts`）に `prompt_template?: SanitizedPromptTemplateRecord` を追加。
- `buildGenerationRecord` 内で、`paramValues[PROMPT_TEMPLATE_STATE_KEY]` と `workflow?.prompt_template_config` が有効な場合、`buildSanitizedTemplateRecord` を呼び出して `record.prompt_template` に格納する。
- `getMetadataView`（`promptRecord.ts`）の戻り値に `promptTemplate: SanitizedPromptTemplateRecord | null` を追加し、`record?.prompt_template ?? null` を返す。
