```typescript
import { useState, useRef, useCallback } from 'react';

export interface UseMultiSelectOptions<T extends { id: number }> {

export interface UseMultiSelectReturn {

    onTouchStart: (e: React.TouchEvent) => void;

// Hook state and refs:
// const [isMultiSelectMode, setIsMultiSelectMode] = useState(false);
// const [selectedIds, setSelectedIds] = useState<Set<number>>(new Set());
// const lastSelectedIdRef = useRef<number | null>(null);

  const handleItemClick = useCallback(
    (id: number, event: React.MouseEvent) => {
      if (!isMultiSelectMode) {
        if (event.ctrlKey || event.metaKey) {
          enterMultiSelectMode(id);
        }
        return;
      }

      if (event.shiftKey && lastSelectedIdRef.current !== null) {
        const fromIdx = items.findIndex((i) => i.id === lastSelectedIdRef.current);
        const toIdx = items.findIndex((i) => i.id === id);
        if (fromIdx !== -1 && toIdx !== -1) {
          const start = Math.min(fromIdx, toIdx);
          const end = Math.max(fromIdx, toIdx);
          const rangeIds = items.slice(start, end + 1).map((i) => i.id);
          setSelectedIds((prev) => new Set([...prev, ...rangeIds]));
        }
      } else {
        setSelectedIds((prev) => {
          const next = new Set(prev);
          if (next.has(id)) {
            next.delete(id);
          } else {
            next.add(id);
          }
          return next;
        });
      }
      lastSelectedIdRef.current = id;
    },
    [isMultiSelectMode, items, enterMultiSelectMode]
  );
```

## 3. 実装要件 (Implementation Logic)

### 3.1 useMultiSelect.ts の修正
- `handleItemClick` 内でマルチ選択モードではない (`!isMultiSelectMode`) 状態の処理を以下のように更新する：
  - `event.ctrlKey || event.metaKey || event.shiftKey` のいずれかが真の場合に、`enterMultiSelectMode(id)` を実行して選択モードに入る。
  - もし `event.shiftKey` が真であり、かつ `lastSelectedIdRef.current !== null` の場合は、`lastSelectedIdRef.current` からクリックされた `id` までの範囲のすべてのアイテムIDを `selectedIds` に追加（範囲選択）する。
  - 修飾キーがない通常のクリック時は、`lastSelectedIdRef.current` をクリックされた `id` に更新する。

### 3.2 useGeneratePageState.ts の修正
- 過去スナップショット選択ドロップダウンの廃止に伴い：
  - `selectedSnapshotId` / `setSelectedSnapshotId` の状態変数およびその初期化処理、および関連する `useEffect` をすべて削除する。
  - 返却オブジェクトから `selectedSnapshotId` と `setSelectedSnapshotId` を削除する。
- プレビューの追従性向上のため：
  - `selectedIds`、`isMultiSelectMode`、`sortedDisplayImages` に依存する `useEffect` を追加する。
  - `isMultiSelectMode` が真であり、かつ `selectedIds` のサイズが 1 以上のとき、`selectedIds`（Set）の最後のインデックスに対応する画像を `sortedDisplayImages` から取得し、現在の `selectedImg` と異なる場合に `setSelectedImg` を呼び出して更新する。

### 3.3 useMultiSelect.test.ts の更新
- `useMultiSelect` のテストケースに、非マルチ選択モード状態からの Shift クリック、Ctrl クリック、および通常クリックから Shift クリックした際の範囲選択挙動をテストするケースを追加する。
