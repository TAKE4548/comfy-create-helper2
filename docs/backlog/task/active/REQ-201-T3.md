### 2.2 共通タグ抽出ロジック (`ProfileBulkEditModal.tsx`)
```typescript
// 選択中プリセットおよび選択された対象プロンプトロールから共通するタグを抽出する純粋関数
export function computeCommonTags(
  presets: WorkflowPreset[],
  targetRoles: string[]
): string[] {
  if (presets.length === 0 || targetRoles.length === 0) return [];

  // 各プリセットについて、選択された対象ロール内のタグを抽出して Set 化
  const presetTagSets: Set<string>[] = presets.map((preset) => {
    const tags = new Set<string>();
    targetRoles.forEach((role) => {
      const val = preset.param_values?.[role];
      if (typeof val === 'string' && val.trim()) {
        val
          .split(',')
          .map((t) => t.trim())
          .filter(Boolean)
          .forEach((t) => tags.add(t));
      }
    });
    return tags;
  });

  // 全てのプリセットに含まれるタグ（積集合）を算出
  const firstSet = presetTagSets[0];
  if (!firstSet) return [];
  const common: string[] = [];

  firstSet.forEach((tag) => {
    const lowerTag = tag.toLowerCase();
    const isCommonInAll = presetTagSets.every((set) =>
      Array.from(set).some((t) => t.toLowerCase() === lowerTag)
    );
    if (isCommonInAll) {
      common.push(tag);
    }
  });

  return common;
}
```
