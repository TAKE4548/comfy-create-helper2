```typescript
async function request<T>(
  path: string,
  options: RequestInit = {}
): Promise<T> {
  const res = await fetch(`${BASE_URL}${path}`, {
    headers: { 'Content-Type': 'application/json', ...options.headers },
    ...options,

    const text = await res.text()

  if (res.status === 204) return undefined as T
```

### 2.4 `HistoryPage` コンポーネントの構造について
`HistoryPage` はルーティングから直接マウントされる自律型の画面コンポーネントであり、外部からの Props (例えば `HistoryPageProps`) を受け取りません。すべての状態（選択中のアルバム ID など）はコンポーネントの内部ステート（`useState`）として管理されます。

### 2.6 `HistoryModalPreview.tsx` の完全な既存コード
```typescript
import { useRef, useEffect } from 'react'
import { useQuery } from '@tanstack/react-query'
import { X, Loader2 } from 'lucide-react'
import { snapshotsApi, lorasApi, tagSetsApi } from '../api/client'
import type { Snapshot, Lora, TagSet } from '../api/types'

interface HistoryModalPreviewProps {
  isOpen: boolean
  onClose: () => void
  images: string[]
  currentIndex: number
  onChangeIndex: (index: number) => void
  snapshotIds?: (number | null)[]
}

export function HistoryModalPreview({
  isOpen,
  onClose,
  images,
  currentIndex,
  onChangeIndex,
  snapshotIds,
}: HistoryModalPreviewProps) {
  const touchStartX = useRef<number | null>(null)

  const activeSnapshotId = snapshotIds && currentIndex >= 0 && currentIndex < snapshotIds.length
    ? snapshotIds[currentIndex]
    : null

  const { data: snapshot, isLoading } = useQuery<Snapshot>({
    queryKey: ['snapshot', activeSnapshotId],
    queryFn: () => snapshotsApi.get(activeSnapshotId!) as Promise<Snapshot>,
    enabled: !!activeSnapshotId && isOpen,
  })

  // キーボードイベント監視 (ArrowLeft, ArrowRight)
  useEffect(() => {
    if (!isOpen) return
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'ArrowLeft') {
        if (currentIndex > 0) onChangeIndex(currentIndex - 1)
      } else if (e.key === 'ArrowRight') {
        if (currentIndex < images.length - 1) onChangeIndex(currentIndex + 1)
      }
    }
    window.addEventListener('keydown', handleKeyDown)
    return () => window.removeEventListener('keydown', handleKeyDown)
  }, [isOpen, currentIndex, images.length, onChangeIndex])

  if (!isOpen || images.length === 0) return null
  const activeImage = images[currentIndex]

  return (
    <div className="fixed inset-0 z-50 bg-black/95 flex flex-col" onClick={onClose}>
      <div className="relative h-12 flex items-center justify-end px-4" onClick={(e) => e.stopPropagation()}>
        <button onClick={onClose} className="absolute top-4 right-4 z-50 p-2 rounded-full bg-white/10 text-white">
          <X size={20} />
        </button>
      </div>

      <div className="w-full flex items-center justify-center bg-black/40 relative cursor-pointer" style={{ height: '60vh' }}>
        {currentIndex > 0 && (
          <button
            onClick={(e) => { e.stopPropagation(); onChangeIndex(currentIndex - 1) }}
            className="absolute left-4 z-10 p-3 rounded-full bg-black/60 hover:bg-black/80 text-white transition-colors"
          >
            &lt;
          </button>
        )}
        <img src={activeImage} alt="" className="max-w-full max-h-full object-contain" />
        {currentIndex < images.length - 1 && (
          <button
            onClick={(e) => { e.stopPropagation(); onChangeIndex(currentIndex + 1) }}
            className="absolute right-4 z-10 p-3 rounded-full bg-black/60 hover:bg-black/80 text-white transition-colors"
          >
            &gt;
          </button>
        )}
        <div className="absolute bottom-4 left-1/2 transform -translate-x-1/2 bg-black/60 px-3 py-1 rounded-full text-xs text-white">
          {currentIndex + 1} / {images.length}
        </div>
      </div>

      <div className="flex-1 overflow-y-auto p-5 bg-surface border-t border-[rgba(255,255,255,0.06)]" onClick={(e) => e.stopPropagation()}>
        {activeSnapshotId && (
          isLoading ? (
            <div className="flex items-center justify-center py-10 gap-2 text-zinc-400 text-xs">
              <Loader2 size={16} className="animate-spin text-accent-green" />
              <span>Loading snapshot metadata...</span>
            </div>
          ) : (
            <div className="space-y-4 text-xs text-zinc-300">
              <p>Params: {JSON.stringify(snapshot?.params)}</p>
            </div>
          )
        )}
      </div>
    </div>
  )
}
```

```typescript
    transform: CSS.Transform.toString(transform),
    transition,
    zIndex: isDragging ? 10 : 1,
    opacity: isDragging ? 0.8 : 1,
    touchAction: 'none', // モバイルでのスクロールと競合させないための設定
  };

  return (
    <div ref={setNodeRef} style={style}>
      {/* ドラッグ操作を行わせる要素に対して listeners と attributes を適用します */}
      <div {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing">
        {children}
      </div>
    </div>
  );
}
```

## 3. 実装要件 (Implementation Logic)
- [ ] `frontend/src/api/types.ts` にバックエンドとの互換性のための型エイリアス `export type AlbumRead = Album;` および `export type AlbumItemRead = AlbumItem;` を追加する。
- [ ] `frontend/src/components/HistoryModalPreview.tsx` に以下の機能を追加する：
  - **左右移動ボタン**: 画面上の左右両端に円形ナビゲーションボタン（`<` および `>`）を配置。画像の一番最初または最後のときは非表示にする。
  - **キーボードイベントの監視**: `useEffect` を用い、`keydown` イベントで `ArrowLeft` が押されたら前へ、`ArrowRight` が押されたら次へ切り替える。
- [ ] `frontend/src/pages/HistoryPage.tsx` にアルバム詳細確認画面（`selectedAlbumId` が non-null の時のビュー）を実装する：
  - **表示切り替え**: アルバムカードをクリックした際、`selectedAlbumId` に該当するIDを設定し、詳細ビューを描画する。
  - **クロップなし画像一覧**: グリッドレイアウトで、各画像は `object-contain` スタイルを使用し、アスペクト比を保ち切り抜きなしで表示する。画像が `null` の場合は、代替プレースホルダー（`ImageOff` 等）を表示する。
  - **削除機能**: 
    - 画像上にホバーした際、削除アイコンをオーバーレイ表示。クリックで一括削除 API (`deleteItems`) を呼び出す。
    - 「複数選択モード」のトグルボタンを追加。ONのときは、画像をクリック・タップした際にチェック状態をトグルする（画像送りモーダルは開かない）。「一括削除実行」ボタンを配置し、選択中の画像をまとめて除外する。
    - 削除処理完了時は、`useMutation` の `onSuccess` でアルバムのクエリキー `['albums']` （詳細取得クエリがあればそれも）を無効化し、表示を最新化する。削除エラー時はトースト通知やエラーメッセージでユーザーに知らせる。
  - **ドラッグ＆ドロップ (DnD) による順序入れ替え**:
    - `@dnd-kit/core` と `@dnd-kit/sortable` を使用して、グリッドアイテムのドラッグ＆ドロップによる並び替えを実装する。
    - `onDragEnd` が発生した際、順序更新 API (`updateOrder`) を呼び出してバックエンドに永続化する。
    - APIリクエスト中（`useMutation` の `isPending` が true の間）はドラッグ操作を無効化する（`PointerSensor` や `SortableItem` の `disabled` オプションを使用）。
    - ドロップ完了時にローカルステートを即時更新（楽観的更新）し、APIエラー発生時には元の順序にロールバックするロジックを実装する。
  - **ソート機能**:
    - 「カスタム順 (永続順)」「登録順 (古い順)」「登録順 (新しい順)」の3つのソート条件を切り替えるトグルボタンを設置する。
    - ソート順を変更した際、即座に画像一覧の表示順を切り替える（ソートには `created_at` を使用する）。
    - 「カスタム順」以外のソートが選択されているときは、ドラッグ＆ドロップ機能を無効化する。
  - **拡大表示との同期**:
    - 画像クリック（複数選択モードOFF時）で `HistoryModalPreview` を起動する。
    - この際、現在選択されているソート順に従ってソートされた画像およびスナップショットIDの配列をビューワーに渡すことで、ビューワー内でもその順序通りに画像送りができるようにする。
