### 2.1 `frontend/src/pages/SettingsPage/components/ThumbnailSettingsSection.tsx`
```tsx
import React, { useState } from 'react'
import { Image as ImageIcon, Check } from 'lucide-react'
import { useThumbnailAspectRatio } from '../../../hooks/useThumbnailAspectRatio'

export const ThumbnailSettingsSection: React.FC = () => {
  const { presetId, customRatio, presets, aspectRatioStyle, ratioStr, setPreset, setCustomRatio } = useThumbnailAspectRatio()
  const [customInput, setCustomInput] = useState(customRatio)

  const handleCustomChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const val = e.target.value
    setCustomInput(val)
    setCustomRatio(val)
  }

  return (
    <div className="bg-[#18181C] border border-[rgba(255,255,255,0.08)] rounded-xl p-5 shadow-lg space-y-4" data-testid="thumbnail-settings-section">
      <div className="flex items-center gap-2.5 pb-3 border-b border-[rgba(255,255,255,0.06)]">
        <ImageIcon className="text-accent-green" size={18} />
        <div>
          <h3 className="text-sm font-semibold text-zinc-100">サムネイル表示・アスペクト比設定</h3>
          <p className="text-[11px] text-zinc-400">一覧カードやプレビューで表示するサムネイルの標準アスペクト比を設定します。</p>
        </div>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-3 gap-6 items-start">
        {/* Preset Selector */}
        <div className="md:col-span-2 space-y-3">
          <label className="text-xs font-medium text-zinc-300 block">アスペクト比の選択</label>
          <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
            {presets.map((p) => {
              const isSelected = presetId === p.id
              return (
                <button
                  key={p.id}
                  type="button"
                  onClick={() => setPreset(p.id)}
                  className={`px-3 py-2 rounded-lg text-xs font-medium border flex items-center justify-between transition-all ${
                    isSelected
                      ? 'bg-accent-green/15 border-accent-green text-accent-green shadow-sm'
                      : 'bg-surface-2 border-[rgba(255,255,255,0.08)] text-zinc-300 hover:border-zinc-500 hover:text-white'
                  }`}
                  data-testid={`preset-btn-${p.id}`}
                >
                  <span>{p.label}</span>
                  {isSelected && <Check size={14} className="text-accent-green flex-shrink-0" />}
                </button>
              )
            })}
          </div>

          {presetId === 'custom' && (
            <div className="pt-2 space-y-1.5 animate-fade-in">
              <label className="text-xs text-zinc-400 block">カスタム比率入力 (W:H または数値)</label>
              <div className="flex items-center gap-2">
                <input
                  type="text"
                  value={customInput}
                  onChange={handleCustomChange}
                  placeholder="例: 5:7, 7:9, 0.71"
                  className="px-3 py-1.5 text-xs bg-zinc-900 border border-[rgba(255,255,255,0.12)] rounded-lg text-zinc-200 focus:outline-none focus:border-accent-green w-48"
                  data-testid="custom-aspect-ratio-input"
                />
                <span className="text-[11px] text-zinc-500">現在適用中: {ratioStr}</span>
              </div>
            </div>
          )}
        </div>

        {/* Live Preview Box */}
        <div className="flex flex-col items-center justify-center p-4 bg-zinc-950/60 rounded-lg border border-[rgba(255,255,255,0.06)]">
          <span className="text-[10px] text-zinc-500 uppercase tracking-wider font-semibold mb-2">Live Preview</span>
          <div
            className="w-32 bg-zinc-900 border-2 border-dashed border-accent-green/60 rounded-lg flex flex-col items-center justify-center p-2 text-center transition-all duration-300 shadow-inner"
            style={aspectRatioStyle}
            data-testid="aspect-ratio-preview-box"
          >
            <ImageIcon size={20} className="text-accent-green/70 mb-1" />
            <span className="text-[10px] font-mono text-zinc-300 font-bold">{ratioStr}</span>
          </div>
        </div>
      </div>
    </div>
  )
}
```
