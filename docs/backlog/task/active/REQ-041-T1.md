from datetime import datetime
from typing import Optional
from sqlalchemy import Integer, String, Float, Text, DateTime, ForeignKey, UniqueConstraint
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship

class Base(DeclarativeBase):
    pass

class Workflow(Base):
    __tablename__ = "workflows"

    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    name: Mapped[str] = mapped_column(String, nullable=False, unique=True)
    json_body: Mapped[str] = mapped_column(Text, nullable=False)
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)

    mappings: Mapped[list["WorkflowMapping"]] = relationship(
        back_populates="workflow", cascade="all, delete-orphan"
    )
    snapshots: Mapped[list["Snapshot"]] = relationship(
        back_populates="workflow", cascade="all, delete-orphan"
    )

class Snapshot(Base):
    __tablename__ = "snapshots"

    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    workflow_id: Mapped[int] = mapped_column(ForeignKey("workflows.id", ondelete="CASCADE"), nullable=False)
    params: Mapped[str] = mapped_column(Text, nullable=False)  # JSON: {role: value}
    image_path: Mapped[Optional[str]] = mapped_column(String)
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)

    workflow: Mapped["Workflow"] = relationship(back_populates="snapshots")
    album_items: Mapped[list["AlbumItem"]] = relationship(
        back_populates="snapshot", cascade="all, delete-orphan"
    )
    loras: Mapped[list["SnapshotLora"]] = relationship(
        back_populates="snapshot", cascade="all, delete-orphan"
    )
    tag_sets: Mapped[list["SnapshotTagSet"]] = relationship(
        back_populates="snapshot", cascade="all, delete-orphan"
    )

class Album(Base):
    __tablename__ = "albums"

    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    name: Mapped[str] = mapped_column(String, nullable=False, unique=True)
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)

    items: Mapped[list["AlbumItem"]] = relationship(
        back_populates="album", cascade="all, delete-orphan",
        order_by="AlbumItem.sequence_order"
    )

class AlbumItem(Base):
    __tablename__ = "album_items"
    __table_args__ = (UniqueConstraint("album_id", "snapshot_id"),)

    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    album_id: Mapped[int] = mapped_column(ForeignKey("albums.id", ondelete="CASCADE"), nullable=False)
    snapshot_id: Mapped[int] = mapped_column(ForeignKey("snapshots.id", ondelete="CASCADE"), nullable=False)
    sequence_order: Mapped[int] = mapped_column(Integer, nullable=False)

    album: Mapped["Album"] = relationship(back_populates="items")
    snapshot: Mapped["Snapshot"] = relationship(back_populates="album_items")

# backend/app/routers/albums.py
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session
from app.database import get_db
from app import models, schemas

router = APIRouter(prefix="/api/albums", tags=["albums"])

@router.get("", response_model=list[schemas.AlbumRead])
def list_albums(db: Session = Depends(get_db)) -> list[schemas.AlbumRead]:
    albums = db.query(models.Album).order_by(models.Album.name).all()
    return [
        schemas.AlbumRead(
            id=album.id,
            name=album.name,
            created_at=album.created_at,
            items=[
                schemas.AlbumItemRead(
                    snapshot_id=item.snapshot_id,
                    sequence_order=item.sequence_order,
                    image_path=item.snapshot.image_path if item.snapshot else None,
                    created_at=item.snapshot.created_at if item.snapshot else None,
                )
                for item in sorted(album.items, key=lambda x: x.sequence_order)
            ],
        )
        for album in albums
    ]

# バックエンドルーターで使われている get_album 関数定義
@router.get("/{album_id}", response_model=schemas.AlbumRead)
def get_album(album_id: int, db: Session = Depends(get_db)) -> schemas.AlbumRead:
    album = db.query(models.Album).filter(models.Album.id == album_id).first()
    if not album:
        raise HTTPException(status_code=404, detail="Album not found")
    return schemas.AlbumRead(
        id=album.id,
        name=album.name,
        created_at=album.created_at,
        items=[
            schemas.AlbumItemRead(
                snapshot_id=item.snapshot_id,
                sequence_order=item.sequence_order,
                image_path=item.snapshot.image_path if item.snapshot else None,
                created_at=item.snapshot.created_at if item.snapshot else None,
            )
            for item in sorted(album.items, key=lambda x: x.sequence_order)
        ],
    )
```

const BASE_URL = '/api'

async function request<T>(
  path: string,
  options: RequestInit = {}
): Promise<T> {
  const res = await fetch(`${BASE_URL}${path}`, {
    headers: { 'Content-Type': 'application/json', ...options.headers },
    ...options,
  })
  if (!res.ok) {
    const text = await res.text()
    throw new Error(`API error ${res.status}: ${text}`)
  }
  if (res.status === 204) return undefined as T
  return res.json()
}

    @router.post("/{album_id}/items/delete", response_model=schemas.AlbumRead)
    def delete_album_items(
        album_id: int, 
        payload: schemas.AlbumItemsDelete, 
        db: Session = Depends(get_db)
    ) -> schemas.AlbumRead:
        album = db.query(models.Album).filter(models.Album.id == album_id).first()
        if not album:
            raise HTTPException(status_code=404, detail="Album not found")
            
        # 指定された snapshot_ids に一致する AlbumItem レコードを削除
        db.query(models.AlbumItem).filter(
            models.AlbumItem.album_id == album_id,
            models.AlbumItem.snapshot_id.in_(payload.snapshot_ids)
        ).delete(synchronize_session=False)
        
        db.flush()
        
        # 残ったアルバムアイテムを取得し、元の sequence_order でソートしてインデックスを 0 から順に再割り当て
        remaining_items = db.query(models.AlbumItem).filter(
            models.AlbumItem.album_id == album_id
        ).order_by(models.AlbumItem.sequence_order).all()
        
        for idx, item in enumerate(remaining_items):
            item.sequence_order = idx
            
        db.commit()
        db.refresh(album)
        return get_album(album_id, db)
    ```
